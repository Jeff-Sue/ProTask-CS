import html
import json


# ============================================================
# Utils
# ============================================================

def esc(value):
    if value is None:
        return ""
    return html.escape(str(value))


def normalize_case_ids(value):
    if value is None:
        return []

    if isinstance(value, str):
        value = [value]

    if not isinstance(value, list):
        return []

    return list(
        dict.fromkeys(
            str(x).strip()
            for x in value
            if str(x).strip()
        )
    )


def get_case_id(case):
    if not isinstance(case, dict):
        return ""

    return str(
        case.get("case_id")
        or case.get("id")
        or case.get("caseId")
        or ""
    ).strip()


def get_case_title(case):
    if not isinstance(case, dict):
        return ""

    for key in (
        "title",
        "case_title",
        "name",
        "question",
        "problem",
        "content",
        "text",
    ):
        value = case.get(key)

        if value:
            text = str(value).strip()

            if len(text) > 100:
                text = text[:100] + "..."

            return text

    return ""


def get_latency_value(latency, *keys):
    if not isinstance(latency, dict):
        return None

    for key in keys:
        if key in latency:
            return latency[key]

    return None


def fmt_seconds(value):
    if value is None:
        return "-"

    try:
        return f"{float(value):.2f}s"
    except Exception:
        return esc(value)

    return """
    <div class="empty-chat">
        <div class="empty-icon">✦</div>

        <div class="empty-title">
            开始新的 IT 支持会话
        </div>

        <div class="empty-desc">

def render_chat(history):
    if not history:
        return render_empty_chat()

    parts = ['<div class="chat-scroll">']

    for message in history:
        role = message.get("role", "")
        content = esc(message.get("content", ""))

        if role == "user":
            parts.append(
                f"""
                <div class="message-row user-row">

                    <div class="message-bubble user-message">
                        <div class="message-role">用户</div>
                        {content}
                    </div>

                    <div class="avatar user-avatar">
                        U
                    </div>

                </div>
                """
            )

        elif role == "assistant":
            parts.append(
                f"""
                <div class="message-row agent-row">

                    <div class="avatar agent-avatar">
                        AI
                    </div>

                    <div class="message-bubble agent-message">
                        <div class="message-role">IT 智能客服</div>
                        {content}
                    </div>

                </div>
                """
            )

    parts.append("</div>")

    return "".join(parts)

# ============================================================
# Topic Tree
# ============================================================

def render_topic_tree(tree):
    if not tree:
        return """
        <div class="state-text">
            None
        </div>
        """

    parts = ['<div class="topic-tree">']

    def walk(node, depth=0):
        topic = esc(node.get("topic", ""))
        dimension = esc(node.get("dimension", ""))

        case_ids = normalize_case_ids(
            node.get("case_ids")
        )

        leaf = bool(node.get("leaf"))

        cls = "tree-node tree-leaf" if leaf else "tree-node"

        margin = depth * 18

        dimension_html = ""

        if dimension:
            dimension_html = (
                f'<span class="tree-dimension">'
                f'{dimension}'
                f'</span>'
            )

        case_html = ""

        if leaf and case_ids:
            case_html = (
                f'<span class="tree-case">'
                f'[{esc(", ".join(case_ids))}]'
                f'</span>'
            )

        parts.append(
            f"""
            <div class="{cls}"
                 style="margin-left:{margin}px">

                <span class="tree-dot"></span>

                <span class="tree-topic">
                    {topic or "Unknown"}
                </span>

                {dimension_html}
                {case_html}

            </div>
            """
        )

        for child in node.get("children", []) or []:
            walk(child, depth + 1)

    walk(tree)

    parts.append("</div>")

    return "".join(parts)

def render_cases(cases):
    if not cases:
        return """
        <div class="state-text">
            None
        </div>
        """

    parts = []

    for case in cases:
        cid = esc(get_case_id(case))
        title = esc(get_case_title(case))

        parts.append(
            f"""
            <div class="case-item">

                <div class="case-id">
                    {cid or "Unknown Case"}
                </div>

                <div class="case-title">
                    {title or "No preview"}
                </div>

            </div>
            """
        )

    return "".join(parts)

# Latency
# ============================================================

def render_latency(latency):
    latency = latency or {}

    total = get_latency_value(
        latency,
        "total",
        "total_time",
        "total_latency",
    )

    items = [
        (
            "Query",
            get_latency_value(
                latency,
                "query",
                "query_time",
            ),
        ),
        (
            "Trigger",
            get_latency_value(
                latency,
                "trigger",
                "trigger_time",
            ),
        ),
        (
            "RAG",
            get_latency_value(
                latency,
                "rag",
                "rag_time",
                "retrieval",
                "retrieval_time",
            ),
        ),
        (
            "Topic",
            get_latency_value(
                latency,
                "topic",
                "topic_time",
                "topic_structure",
                "topic_structure_time",
            ),
        ),
        (
            "Policy",
            get_latency_value(
                latency,
                "policy",
                "policy_time",
            ),
        ),
        (
            "Response",
            get_latency_value(
                latency,
                "response",
                "response_time",
            ),
        ),
    ]

    item_html = []

    for name, value in items:
        item_html.append(
            f"""
            <div class="latency-item">
                <span>{name}</span>
                <b>{fmt_seconds(value)}</b>
            </div>
            """
        )

    return f"""
    <div class="latency-total">
        {fmt_seconds(total)}
    </div>

    <div class="latency-grid">
        {''.join(item_html)}
    </div>
    """

def render_current_state(state):
    current = state.get("current_turn")

    if not current:
        return """
        <div class="state-empty">

            <div class="state-empty-title">
                Waiting for interaction
            </div>

            <div class="state-empty-desc">
                当前轮执行后，Query、Trigger、Evidence、
                Topic Structure、Policy 和 Latency
                将显示在这里。
            </div>

        </div>
        """

    query_state = current.get("query_state", {}) or {}
    trigger_state = current.get("trigger_state", {}) or {}
    policy_state = current.get("policy_state", {}) or {}

    memory = current.get("memory", {}) or {}

    evidence_state = memory.get(
        "evidence_state",
        {}
    ) or {}

    active_cases = evidence_state.get(
        "active_cases",
        []
    ) or []

    topic_structure = evidence_state.get(
        "topic_structure",
        {}
    ) or {}

    latency = current.get("latency", {}) or {}

    query_mode = esc(
        query_state.get("query_mode", "-")
    )

    query_text = esc(
        query_state.get("query_text", "")
    )

    trigger = esc(
        trigger_state.get("trigger", "-")
    )

    evidence_mode = esc(
        trigger_state.get("evidence_mode", "-")
    )

    policy = esc(
        policy_state.get("label", "-")
    )

    information = policy_state.get("information")

    if information is None:
        information_text = "None"
    elif isinstance(information, (dict, list)):
        information_text = esc(
            json.dumps(
                information,
                ensure_ascii=False,
                indent=2,
            )
        )
    else:
        information_text = esc(information)

    return f"""
    <div class="state-section">

        <div class="section-title">
            Query
        </div>

        <div class="kv-row">
            <span>Mode</span>
            <span class="badge badge-blue">
                {query_mode}
            </span>
        </div>

        <div class="state-label">
            Query Text
        </div>

        <div class="state-text">
            {query_text or "None"}
        </div>

    </div>


    <div class="state-section">

        <div class="section-title">
            Evidence Decision
        </div>

        <div class="kv-row">
            <span>Trigger</span>
            <b>{trigger}</b>
        </div>

        <div class="kv-row">
            <span>Mode</span>
            <span class="badge badge-green">
                {evidence_mode}
            </span>
        </div>

    </div>


    <div class="state-section">

        <div class="section-title">
            Retrieved Cases
        </div>

        {render_cases(active_cases)}

    </div>


    <div class="state-section">

        <div class="section-title">
            Topic Structure
        </div>

        {render_topic_tree(topic_structure)}

    </div>


    <div class="state-section">

        <div class="section-title">
            Policy
        </div>

        <div class="kv-row">
            <span>Action</span>

            <span class="badge badge-blue">
                {policy}
            </span>
        </div>

        <div class="state-label">
            Information
        </div>

        <div class="state-text">
            {information_text}
        </div>

    </div>


    <div class="state-section">

        <div class="section-title">
            Latency
        </div>

        {render_latency(latency)}

    </div>
    """

# Trace
# ============================================================

def render_trace(state):
    trace = state.get("trace", [])

    if not trace:
        return """
        <div class="state-empty">
            <div class="state-empty-title">
                暂无运行轨迹
            </div>
        </div>
        """

    parts = []

    for index, item in enumerate(
        reversed(trace),
        start=1,
    ):
        query = item.get("query_state", {}) or {}
        trigger = item.get("trigger_state", {}) or {}
        policy = item.get("policy_state", {}) or {}
        latency = item.get("latency", {}) or {}

        user = esc(item.get("user", ""))

        query_mode = esc(
            query.get("query_mode", "-")
        )

        evidence_mode = esc(
            trigger.get("evidence_mode", "-")
        )

        policy_label = esc(
            policy.get("label", "-")
        )

        total = get_latency_value(
            latency,
            "total",
            "total_time",
            "total_latency",
        )

        turn_id = item.get(
            "turn_id",
            len(trace) - index + 1,
        )

        parts.append(
            f"""
            <div class="trace-item">

                <div class="trace-head">
                    <b>Turn {esc(turn_id)}</b>
                    <span>{fmt_seconds(total)}</span>
                </div>

                <div class="trace-user">
                    {user}
                </div>

                <div class="trace-flow">

                    <span class="badge badge-blue">
                        {query_mode}
                    </span>

                    <span class="flow-arrow">→</span>

                    <span class="badge badge-green">
                        {evidence_mode}
                    </span>

                    <span class="flow-arrow">→</span>

                    <span class="badge badge-gray">
                        {policy_label}
                    </span>

                </div>

            </div>
            """
        )

    return "".join(parts)
