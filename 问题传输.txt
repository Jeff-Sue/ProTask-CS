def validate_rag_query(query, x, memory, t_id):
    query = query.get("query_text", "") if isinstance(query, dict) else query
    query = "" if query is None else str(query).strip()

    if not query:
        print("\n========== EMPTY RAG QUERY ==========")
        print("turn:", t_id)
        print("dialog_id:", x["dialog_id"])
        print("user:", x["user"])
        print("query_raw:", x.get("query_raw"))
        print("query_state:", x.get("query"))
        print("current_issue:", repr(memory["issue_state"]["current_issue"]))
        print("query_history:", memory["issue_state"]["query_history"])
        print("trigger:", x.get("trigger"))
        print("dialogue_history:", x.get("dialogue_history"))
        print("=====================================\n")
        raise ValueError(f"RAG query is empty: dialog_id={x['dialog_id']}, turn={t_id}")

    return query


def parse_json(text):
    if isinstance(text, dict):
        return text
    if not isinstance(text, str):
        return {}

    text = text.strip()
    if not text:
        return {}

    try:
        return json.loads(text)
    except Exception:
        pass

    for mark in ("```json", "```"):
        if mark in text:
            try:
                content = text.split(mark, 1)[1].split("```", 1)[0].strip()
                return json.loads(content)
            except Exception:
                return {}

    return {}


def parse_llm_output(output):
    parsed = parse_json(output)
    if not parsed:
        return {}
    return parsed if "response" not in parsed else parse_json(parsed["response"])


def prompt_read(path: Path) -> str:
    if not path.exists():
        raise FileNotFoundError(f"Prompt file not found: {path}")
    return path.read_text(encoding="utf-8").strip()


def write_json(path: str | Path, data: Any):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")


def build_messages(system_prompt, data):
    return [
        {"role": "system", "content": system_prompt},
        {"role": "user", "content": json.dumps(data, ensure_ascii=False)}
    ]


def get_case_id(case: Dict) -> str:
    return str(case.get("case_id", "")).strip()


def case_map_from_cases(cases: List[Dict]) -> Dict[str, Dict]:
    return {get_case_id(case): case for case in cases if get_case_id(case)}


def normalize_case_ids(value):
    if value is None:
        return []
    if isinstance(value, str):
        value = [value]
    if not isinstance(value, list):
        return []
    return list(dict.fromkeys(str(x).strip() for x in value if str(x).strip()))

def init_memory():
    return {
        "issue_state": {
            "current_issue": "",
            "query_history": []
        },
        "dialog_state": {
            "compressed_history": []
        },
        "evidence_state": {
            "active_cases": [],
            "topic_structure": {}
        },
        "interaction_state": {
            "stage": "Start",
            "last_action": "",
            "action_history": []
        }
    }


def update_issue_memory(memory, query_state):
    mode = query_state.get("query_mode", "none")
    query = query_state.get("query_text", "")

    if mode in ("rewrite", "refine"):
        memory["issue_state"]["current_issue"] = query
        memory["issue_state"]["query_history"].append(query)

    return memory


def update_evidence_memory(memory, trigger_state, evidence, topic_structure=None):
    if trigger_state.get("evidence_mode") == "retrieve_new_evidence":
        memory["evidence_state"]["active_cases"] = evidence
        memory["evidence_state"]["topic_structure"] = topic_structure or {}

    return memory


def update_interaction_memory(memory, policy_state):
    label = policy_state.get("label", "")
    interaction = memory["interaction_state"]

    interaction["last_action"] = label
    interaction["action_history"].append(label)

    if label in ("AskMissingSlot", "AskClarification"):
        interaction["stage"] = "Clarification"
    elif label in ("ExplainedResponse", "CaseRecommendation"):
        interaction["stage"] = "Resolution"
    elif label == "Handoff":
        interaction["stage"] = "End"

    return memory


def update_dialogue_memory(memory, user, response):
    history = memory["dialog_state"]["compressed_history"]
    history.append({"user": user, "assistant": response})
    memory["dialog_state"]["compressed_history"] = history[-5:]
    return memory

def build_query_input(memory, user):
    return {
        "issue_state": memory["issue_state"],
        "dialog_state": memory["dialog_state"],
        "current": {"u": user}
    }


def build_trigger_input(memory):
    return {
        "issue_state": memory["issue_state"],
        "evidence_state": {
            "active_cases": memory["evidence_state"]["active_cases"]
        },
        "dialog_state": memory["dialog_state"]
    }


def build_policy_input(memory, user):
    return {
        "issue_state": memory["issue_state"],
        "evidence_state": {
            "active_cases": memory["evidence_state"]["active_cases"],
            "topic_structure": memory["evidence_state"]["topic_structure"]
        },
        "interaction_state": memory["interaction_state"],
        "dialog_state": memory["dialog_state"],
        "current": {"u": user}
    }


def build_response_input(memory, user):
    return {
        "issue_state": memory["issue_state"],
        "evidence_state": {
            "active_cases": memory["evidence_state"]["active_cases"]
        },
        "interaction_state": memory["interaction_state"],
        "dialog_state": memory["dialog_state"],
        "current": {
            "u": user,
            "policy": memory["interaction_state"]["last_action"]
        }
    }

def build_topic_split_input(query: str, cases: List[Dict], parent_topic: str, depth: int):
    return {
        "query": query,
        "parent_topic": parent_topic,
        "depth": depth,
        "cases": cases
    }


def make_leaf(case_id: str, topic: Optional[str] = None):
    return {
        "topic": topic or case_id,
        "case_ids": [case_id],
        "leaf": True,
        "case_id": case_id,
        "children": []
    }


def normalize_split(split_result: Dict, expected_case_ids: List[str]) -> Optional[Dict]:
    if not isinstance(split_result, dict):
        return None

    dimension = str(split_result.get("dimension", "")).strip()
    children = split_result.get("children", [])

    if not dimension or not isinstance(children, list):
        return None

    expected_set = set(expected_case_ids)
    normalized_children = []
    observed = []

    for child in children:
        if not isinstance(child, dict):
            continue

        topic = str(child.get("topic", "")).strip()
        case_ids = child.get("case_ids", [])

        if not topic or not isinstance(case_ids, list):
            continue

        clean_ids = [
            str(case_id).strip()
            for case_id in case_ids
            if str(case_id).strip() in expected_set
        ]
        clean_ids = list(dict.fromkeys(clean_ids))

        if not clean_ids:
            continue

        normalized_children.append({
            "topic": topic,
            "case_ids": clean_ids
        })
        observed.extend(clean_ids)

    if len(normalized_children) < 2:
        return None

    if sorted(observed) != sorted(expected_case_ids):
        return None

    if len(observed) != len(set(observed)):
        return None

    return {
        "dimension": dimension,
        "children": normalized_children
    }


def fallback_to_case_leaves(node: Dict, case_ids: List[str]):
    node["leaf"] = False
    node["dimension"] = "案例差异"
    node["children"] = [make_leaf(case_id) for case_id in case_ids]


def generate_topic_structures_batch(
    items: List[Dict],
    topic_prompt: str,
    max_depth: int = 8,
    max_retry: int = 2
):
    trees = []
    case_maps = []
    frontier = []

    # --------------------------------------------------------
    # Root
    # --------------------------------------------------------

    for tree_idx, item in enumerate(items):
        query = item.get("query", "")
        cases = item.get("cases", [])
        case_map = case_map_from_cases(cases)
        case_maps.append(case_map)

        case_ids = list(case_map.keys())

        root = {
            "query": query,
            "topic": query,
            "case_ids": case_ids,
            "leaf": len(case_ids) <= 1,
            "children": []
        }

        trees.append(root)

        if not case_ids:
            continue

        if len(case_ids) == 1:
            root["case_id"] = case_ids[0]
            continue

        frontier.append({
            "tree_idx": tree_idx,
            "node": root,
            "case_ids": case_ids,
            "depth": 0
        })

    # --------------------------------------------------------
    # Recursive refinement
    # --------------------------------------------------------

    round_id = 0

    while frontier:
        print(f"[TopicStructure] Round {round_id}: {len(frontier)} nodes")

        active_frontier = []

        for task in frontier:
            if task["depth"] >= max_depth:
                print(f"[TopicStructure Warning] max_depth reached: {task['case_ids']}")
                fallback_to_case_leaves(task["node"], task["case_ids"])
            else:
                active_frontier.append(task)

        if not active_frontier:
            break

        # ----------------------------------------------------
        # First generation
        # ----------------------------------------------------

        messages = []

        for task in active_frontier:
            tree_idx = task["tree_idx"]
            case_map = case_maps[tree_idx]
            node_cases = [case_map[case_id] for case_id in task["case_ids"]]

            split_input = build_topic_split_input(
                query=items[tree_idx].get("query", ""),
                cases=node_cases,
                parent_topic=task["node"]["topic"],
                depth=task["depth"]
            )

            messages.append(build_messages(topic_prompt, split_input))

        outputs = parallel_inference(messages)

        valid_results = {}
        retry_tasks = []

        for idx, (task, output) in enumerate(zip(active_frontier, outputs)):
            split_result = parse_llm_output(output)
            valid_split = normalize_split(split_result, task["case_ids"])

            if valid_split:
                valid_results[idx] = valid_split
            else:
                retry_tasks.append({
                    "frontier_idx": idx,
                    "task": task,
                    "attempt": 1
                })

        # ----------------------------------------------------
        # Retry invalid split
        # ----------------------------------------------------

        while retry_tasks:
            attempt = retry_tasks[0]["attempt"]

            if attempt > max_retry:
                break

            print(f"[TopicStructure] Retry {attempt}: {len(retry_tasks)} nodes")

            retry_messages = []

            for retry_item in retry_tasks:
                task = retry_item["task"]
                tree_idx = task["tree_idx"]
                case_map = case_maps[tree_idx]
                node_cases = [case_map[case_id] for case_id in task["case_ids"]]

                split_input = build_topic_split_input(
                    query=items[tree_idx].get("query", ""),
                    cases=node_cases,
                    parent_topic=task["node"]["topic"],
                    depth=task["depth"]
                )

                split_input["retry_instruction"] = (
                    "上一次划分结果不合法。请重新划分，并严格保证："
                    "每个输入 case_id 必须且只能出现一次；"
                    "至少划分为两个有意义的子主题；"
                    "不得遗漏或重复任何 case_id。"
                )

                retry_messages.append(build_messages(topic_prompt, split_input))

            retry_outputs = parallel_inference(retry_messages)
            next_retry_tasks = []

            for retry_item, output in zip(retry_tasks, retry_outputs):
                task = retry_item["task"]
                frontier_idx = retry_item["frontier_idx"]

                split_result = parse_llm_output(output)
                valid_split = normalize_split(split_result, task["case_ids"])

                if valid_split:
                    valid_results[frontier_idx] = valid_split
                else:
                    next_attempt = retry_item["attempt"] + 1

                    if next_attempt <= max_retry:
                        next_retry_tasks.append({
                            "frontier_idx": frontier_idx,
                            "task": task,
                            "attempt": next_attempt
                        })

            retry_tasks = next_retry_tasks

        # ----------------------------------------------------
        # Build children
        # ----------------------------------------------------

        next_frontier = []

        for idx, task in enumerate(active_frontier):
            node = task["node"]
            valid_split = valid_results.get(idx)

            if not valid_split:
                print(
                    "[TopicStructure Warning] "
                    f"failed split after retries: {task['case_ids']}"
                )
                fallback_to_case_leaves(node, task["case_ids"])
                continue

            node["leaf"] = False
            node["dimension"] = valid_split["dimension"]
            node["children"] = []

            for child_info in valid_split["children"]:
                child_case_ids = child_info["case_ids"]

                child_node = {
                    "topic": child_info["topic"],
                    "case_ids": child_case_ids,
                    "leaf": len(child_case_ids) == 1,
                    "children": []
                }

                node["children"].append(child_node)

                if len(child_case_ids) == 1:
                    child_node["case_id"] = child_case_ids[0]
                    continue

                # 关键逻辑：
                # 下一轮只对当前 child 自己包含的 Cases 继续细化
                next_frontier.append({
                    "tree_idx": task["tree_idx"],
                    "node": child_node,
                    "case_ids": child_case_ids,
                    "depth": task["depth"] + 1
                })

        frontier = next_frontier
        round_id += 1

    return trees

def format_topic_tree(tree: Dict) -> str:
    if not tree:
        return "None"

    lines = []

    def node_text(node):
        topic = str(node.get("topic", "")).strip()
        dimension = str(node.get("dimension", "")).strip()
        case_id = str(node.get("case_id", "")).strip()

        if node.get("leaf") and case_id:
            return f"{topic}  [{case_id}]" if topic != case_id else f"[{case_id}]"

        if dimension:
            return f"{topic}  <{dimension}>"

        return topic or case_id or "Unknown"

    def walk(node, prefix="", is_last=True, root=False):
        if root:
            lines.append(node_text(node))
        else:
            lines.append(prefix + ("└─ " if is_last else "├─ ") + node_text(node))

        children = node.get("children", [])
        if not children:
            return

        child_prefix = prefix if root else prefix + ("   " if is_last else "│  ")

        for i, child in enumerate(children):
            walk(
                child,
                prefix=child_prefix,
                is_last=(i == len(children) - 1),
                root=False
            )

    walk(tree, root=True)
    return "\n".join(lines)


def print_agent_state(x, memory):
    query_state = x.get("query", {})
    trigger_state = x.get("trigger", {})
    policy_state = x.get("policy", {})

    cases = memory["evidence_state"]["active_cases"]
    case_ids = [get_case_id(case) for case in cases if get_case_id(case)]

    selected_cases = normalize_case_ids(
        policy_state.get("case_id") or policy_state.get("selected_case")
    )

    information = policy_state.get("information")
    if information in (None, "", [], {}):
        information_text = "None"
    elif isinstance(information, (dict, list)):
        information_text = json.dumps(information, ensure_ascii=False)
    else:
        information_text = str(information)

    tree_text = format_topic_tree(
        memory["evidence_state"]["topic_structure"]
    )

    print("\n╔══════════════ AGENT STATE ══════════════╗")
    print(
        f" Query    : {query_state.get('query_mode', '')} | "
        f"{query_state.get('query_text', '')}"
    )
    print(f" Trigger  : {trigger_state.get('trigger', '')}")
    print(f" Evidence : {trigger_state.get('evidence_mode', '')}")
    print(f" Cases    : {', '.join(case_ids) if case_ids else 'None'}")
    print("\n Topic Tree")

    for line in tree_text.splitlines():
        print(f" {line}")

    print(f"\n Policy   : {policy_state.get('label', '')}")
    print(f" Info     : {information_text}")
    print(f" Case     : {selected_cases if selected_cases else 'None'}")
    print("╚══════════════════════════════════════════╝")

def run_memory_agent_batch(data):
    query_prompt = prompt_read(Path("prompts/query.prompt"))
    trigger_prompt = prompt_read(Path("prompts/trigger.prompt"))
    topic_prompt = prompt_read(Path("prompts/topic.prompt"))
    policy_prompt = prompt_read(Path("prompts/policy.prompt"))
    response_prompt = prompt_read(Path("prompts/response.prompt"))

    memories = {dialog_id: init_memory() for dialog_id in data}
    dialogue_history = {dialog_id: "" for dialog_id in data}
    results = {dialog_id: {"turns": []} for dialog_id in data}

    max_turn = max(len(item["text"]) for item in data.values())

    for t_id in range(max_turn):
        print(f"\n========== TURN {t_id} ==========")

        current_batch = []

        # ====================================================
        # Current Users
        # ====================================================

        for dialog_id, item in data.items():
            if t_id >= len(item["text"]):
                continue

            msg = item["text"][t_id]

            if "用户" in msg:
                dialogue_history[dialog_id] += f"用户: {msg['用户']}\n"
            elif "客服" in msg:
                dialogue_history[dialog_id] += f"客服: {msg['客服']}\n"

            if "用户" not in msg:
                continue

            golden_response = ""
            if t_id + 1 < len(item["text"]):
                golden_response = item["text"][t_id + 1].get("客服", "")

            current_batch.append({
                "dialog_id": dialog_id,
                "user": msg["用户"],
                "golden_response": golden_response,
                "dialogue_history": dialogue_history[dialog_id]
            })

        if not current_batch:
            continue

        # ====================================================
        # 1. Query
        # ====================================================

        print(f"[Turn {t_id}] Query Batch: {len(current_batch)}")

        query_messages = [
            build_messages(
                query_prompt,
                build_query_input(memories[x["dialog_id"]], x["user"])
            )
            for x in current_batch
        ]

        query_outputs = parallel_inference(query_messages)

        for x, output in zip(current_batch, query_outputs):
            query_state = parse_llm_output(output)
            x["query_raw"] = output
            x["query"] = query_state

            dialog_id = x["dialog_id"]
            memories[dialog_id] = update_issue_memory(
                memories[dialog_id],
                query_state
            )

        # ====================================================
        # 2. Trigger
        # ====================================================

        print(f"[Turn {t_id}] Trigger Batch")

        trigger_messages = [
            build_messages(
                trigger_prompt,
                build_trigger_input(memories[x["dialog_id"]])
            )
            for x in current_batch
        ]

        trigger_outputs = parallel_inference(trigger_messages)

        for x, output in zip(current_batch, trigger_outputs):
            x["trigger"] = parse_llm_output(output)

        # ====================================================
        # 3. RAG
        # ====================================================

        print(f"[Turn {t_id}] RAG Batch")

        rag_dialogs = []
        rag_queries = []
        rag_indices = []

        for idx, x in enumerate(current_batch):
            memory = memories[x["dialog_id"]]

            if x["trigger"].get("evidence_mode") == "retrieve_new_evidence":
                query = validate_rag_query(
                    memory["issue_state"]["current_issue"],
                    x,
                    memory,
                    t_id
                )

                rag_dialogs.append(x["dialogue_history"])
                rag_queries.append(query)
                rag_indices.append(idx)

            else:
                x["evidence"] = memory["evidence_state"]["active_cases"]
                x["topic_structure"] = copy.deepcopy(
                    memory["evidence_state"]["topic_structure"]
                )

        if rag_queries:
            print(f"[Turn {t_id}] RAG Requests: {len(rag_queries)}")

            rag_outputs = batch_zefeng_rag(rag_dialogs, rag_queries)

            for idx, evidence in zip(rag_indices, rag_outputs):
                current_batch[idx]["evidence"] = evidence

        # ====================================================
        # 3.5 Topic Structure
        # ====================================================

        topic_items = []
        topic_indices = []

        for idx in rag_indices:
            x = current_batch[idx]
            memory = memories[x["dialog_id"]]

            topic_items.append({
                "query": memory["issue_state"]["current_issue"],
                "cases": x["evidence"]
            })
            topic_indices.append(idx)

        if topic_items:
            print(f"[Turn {t_id}] Topic Structure: {len(topic_items)}")

            structures = generate_topic_structures_batch(
                topic_items,
                topic_prompt
            )

            for idx, structure in zip(topic_indices, structures):
                current_batch[idx]["topic_structure"] = structure

        # ====================================================
        # Update Evidence Memory
        # ====================================================

        for x in current_batch:
            dialog_id = x["dialog_id"]

            memories[dialog_id] = update_evidence_memory(
                memories[dialog_id],
                x["trigger"],
                x["evidence"],
                x.get("topic_structure")
            )

        # ====================================================
        # 4. Policy
        # ====================================================

        print(f"[Turn {t_id}] Policy Batch")

        policy_messages = [
            build_messages(
                policy_prompt,
                build_policy_input(
                    memories[x["dialog_id"]],
                    x["user"]
                )
            )
            for x in current_batch
        ]

        policy_outputs = parallel_inference(policy_messages)

        for x, output in zip(current_batch, policy_outputs):
            policy_state = parse_llm_output(output)
            x["policy"] = policy_state

            dialog_id = x["dialog_id"]
            memories[dialog_id] = update_interaction_memory(
                memories[dialog_id],
                policy_state
            )

            # 中间状态展示
            print_agent_state(
                x,
                memories[dialog_id]
            )

        # ====================================================
        # 5. Response
        # ====================================================

        print(f"[Turn {t_id}] Response Batch")

        response_messages = [
            build_messages(
                response_prompt,
                build_response_input(
                    memories[x["dialog_id"]],
                    x["user"]
                )
            )
            for x in current_batch
        ]

        response_outputs = parallel_inference(response_messages)

        for x, output in zip(current_batch, response_outputs):
            x["response"] = parse_llm_output(output)

            dialog_id = x["dialog_id"]

            memories[dialog_id] = update_dialogue_memory(
                memories[dialog_id],
                x["user"],
                x["golden_response"]
            )

            x["topic_structure"] = copy.deepcopy(
                memories[dialog_id]["evidence_state"]["topic_structure"]
            )

            results[dialog_id]["turns"].append({
                "turn_id": int(t_id / 2),
                **x
            })

    for dialog_id in data:
        results[dialog_id]["final_memory"] = memories[dialog_id]

    return results


# ============================================================
# Main
# ============================================================

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--data_path", required=True)
    parser.add_argument("--output_path", required=True)
    parser.add_argument("--limit", type=int, default=1)
    parser.add_argument("--start", type=int, default=240)
    args = parser.parse_args()

    with open(args.data_path, "r", encoding="utf-8") as f:
        data = json.load(f)

    if args.limit is not None:
        data = dict(
            list(data.items())[args.start:args.start + args.limit]
        )

    start_time = time.time()

    outputs = run_memory_agent_batch(data)

    print(f"\nIt cost {time.time() - start_time:.2f} s.")

    write_json(args.output_path, outputs)

    print(f"Saved {len(outputs)} dialogues to {args.output_path}")


if __name__ == "__main__":
    main()
