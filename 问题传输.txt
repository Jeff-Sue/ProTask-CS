# ============================================================
# Session
# ============================================================

def new_session():
    return {
        "session_id": f"web_{uuid.uuid4().hex[:10]}",
        "history": [],
        "agent": MemoryAgent(
            show_state=False,
        ),
        "trace": [],
        "current_turn": None,
        "view": "user",
    }


def ensure_session(state):
    if not isinstance(state, dict):
        return new_session()

    if "agent" not in state:
        return new_session()

    return state


# ============================================================
# Agent State Extraction
# ============================================================

def get_agent_last_turn(agent):
    """
    Compatible with the latency-enabled run.py.

    Recommended run.py fields:
        agent.last_turn
        or
        agent.last_trace

    If neither exists, we still expose memory.
    """

    if hasattr(agent, "last_turn"):
        value = getattr(agent, "last_turn")

        if isinstance(value, dict):
            return copy.deepcopy(value)

    if hasattr(agent, "last_trace"):
        value = getattr(agent, "last_trace")

        if isinstance(value, dict):
            return copy.deepcopy(value)

    return {
        "memory": copy.deepcopy(agent.memory),
        "latency": {},
    }


# ============================================================
# Prefix Parser
# ============================================================

def parse_prefix(text):
    history = []

    for raw_line in str(text or "").splitlines():
        line = raw_line.strip()

        if not line:
            continue

        if line.startswith("用户:"):
            content = line.split(":", 1)[1].strip()

            if content:
                history.append({
                    "role": "user",
                    "content": content,
                })

        elif line.startswith("用户："):
            content = line.split("：", 1)[1].strip()

            if content:
                history.append({
                    "role": "user",
                    "content": content,
                })

        elif line.startswith("客服:"):
            content = line.split(":", 1)[1].strip()

            if content:
                history.append({
                    "role": "assistant",
                    "content": content,
                })

        elif line.startswith("客服："):
            content = line.split("：", 1)[1].strip()

            if content:
                history.append({
                    "role": "assistant",
                    "content": content,
                })

    return history


def rebuild_agent_memory(agent, history):
    """
    Prefix is treated as existing dialogue context.

    We do NOT rerun the Agent for old turns.
    We only rebuild dialogue memory so the next user turn
    can use the loaded context.
    """

    agent.reset()

    recent = []

    pending_user = None

    for message in history:
        role = message.get("role")
        content = str(
            message.get("content", "")
        ).strip()

        if role == "user":
            pending_user = content

        elif role == "assistant" and pending_user:
            recent.append({
                "user": pending_user,
                "assistant": content,
            })

            pending_user = None

    agent.memory["dialogue_state"]["recent_history"] = (
        recent[-5:]
    )


# ============================================================
# View
# ============================================================

def switch_to_user(state):
    state = ensure_session(state)
    state["view"] = "user"

    return (
        gr.update(visible=True),
        gr.update(visible=False),
        state,
    )


def switch_to_developer(state):
    state = ensure_session(state)
    state["view"] = "developer"

    return (
        gr.update(visible=False),
        gr.update(visible=True),
        state,
    )


# ============================================================
# Chat
# ============================================================

def chat(user_input, state):
    state = ensure_session(state)

    user_input = str(
        user_input or ""
    ).strip()

    if not user_input:
        return (
            "",
            render_chat(state["history"]),
            render_current_state(state),
            render_trace(state),
            render_session_bar(state),
            state,
        )

    agent = state["agent"]

    # --------------------------------------------------------
    # Add current user
    # --------------------------------------------------------

    dialogue_history = (
        state["history"]
        + [{
            "role": "user",
            "content": user_input,
        }]
    )

    try:
        # ----------------------------------------------------
        # Run Agent
        # ----------------------------------------------------

        response = agent.reply(
            dialogue_history
        )

        # ----------------------------------------------------
        # Save chat
        # ----------------------------------------------------

        state["history"].append({
            "role": "user",
            "content": user_input,
        })

        state["history"].append({
            "role": "assistant",
            "content": response,
        })

        # ----------------------------------------------------
        # Get execution state
        # ----------------------------------------------------

        turn = get_agent_last_turn(agent)

        turn["turn_id"] = (
            len(state["trace"]) + 1
        )

        turn["user"] = user_input
        turn["response"] = response

        if "memory" not in turn:
            turn["memory"] = copy.deepcopy(
                agent.memory
            )

        state["current_turn"] = turn
        state["trace"].append(turn)

    except Exception as exc:
        error_text = (
            f"Agent 执行失败："
            f"{type(exc).__name__}: {exc}"
        )

        state["history"].append({
            "role": "user",
            "content": user_input,
        })

        state["history"].append({
            "role": "assistant",
            "content": error_text,
        })

    return (
        "",
        render_chat(state["history"]),
        render_current_state(state),
        render_trace(state),
        render_session_bar(state),
        state,
    )


# ============================================================
# Reset
# ============================================================

def reset_chat():
    state = new_session()

    return (
        "",
        render_chat([]),
        render_current_state(state),
        render_trace(state),
        render_session_bar(state),
        state,
        "",
    )


# ============================================================
# Load Prefix
# ============================================================

def load_prefix(prefix_text, state):
    state = ensure_session(state)

    history = parse_prefix(
        prefix_text
    )

    if not history:
        return (
            render_chat(state["history"]),
            render_current_state(state),
            render_trace(state),
            render_session_bar(state),
            state,
            render_prefix_status(
                "没有识别到有效对话。"
                "请使用“用户:”和“客服:”格式。",
                False,
            ),
        )

    state["history"] = history
    state["trace"] = []
    state["current_turn"] = None

    rebuild_agent_memory(
        state["agent"],
        history,
    )

    return (
        render_chat(state["history"]),
        render_current_state(state),
        render_trace(state),
        render_session_bar(state),
        state,
        render_prefix_status(
            f"已加载 {len(history)} 条历史消息。",
            True,
        ),
    )
