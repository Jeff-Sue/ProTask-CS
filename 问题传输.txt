def new_session():
    return {
        "session_id": "web_" + uuid.uuid4().hex[:10],

        "agent": MemoryAgent(
            mode="single",
            show_state=False,
        ),

        "history": [],
        "trace": [],
        "turns": 0,
        "mode": "user",
    }

# Agent Snapshot
# ============================================================

def get_last_run(agent):
    """
    MemoryAgent.reply() 当前实际暴露的是：

    agent.last_turn_state = {
        "query": ...,
        "trigger": ...,
        "evidence": ...,
        "policy": ...,
        "response": ...,
        "latency": ...
    }

    UI 内部统一转换成：
        query_state
        trigger_state
        evidence_state
        policy_state
        response
        latency
    """

    turn_state = getattr(
        agent,
        "last_turn_state",
        None,
    )

    if turn_state:
        return {
            "query_state": turn_state.get(
                "query",
                {},
            ) or {},

            "trigger_state": turn_state.get(
                "trigger",
                {},
            ) or {},

            "evidence_state": turn_state.get(
                "evidence",
                {},
            ) or {},

            "policy_state": turn_state.get(
                "policy",
                {},
            ) or {},

            "response": turn_state.get(
                "response",
                "",
            ),

            "latency": turn_state.get(
                "latency",
                {},
            ) or {},
        }

    # fallback
    memory = getattr(
        agent,
        "memory",
        {},
    ) or {}

    return {
        "query_state": {},
        "trigger_state": {},

        "evidence_state": memory.get(
            "evidence_state",
            {},
        ) or {},

        "policy_state": (
            memory
            .get("interaction_state", {})
            .get("last_policy", {})
            or {}
        ),

        "response": "",
        "latency": {},
    }

def parse_prefix(text):
    history = []

    if not text:
        return history

    for raw_line in text.splitlines():
        line = raw_line.strip()

        if not line:
            continue

        if line.startswith("用户:"):
            content = line.split(
                "用户:",
                1,
            )[1].strip()

            history.append({
                "role": "user",
                "content": content,
            })

        elif line.startswith("用户："):
            content = line.split(
                "用户：",
                1,
            )[1].strip()

            history.append({
                "role": "user",
                "content": content,
            })

        elif line.startswith("客服:"):
            content = line.split(
                "客服:",
                1,
            )[1].strip()

            history.append({
                "role": "assistant",
                "content": content,
            })

        elif line.startswith("客服："):
            content = line.split(
                "客服：",
                1,
            )[1].strip()

            history.append({
                "role": "assistant",
                "content": content,
            })

    return history

def session_html(state):
    return render_session(
        state["session_id"],
        state["turns"],
    )


def developer_chat_html(state):
    return render_chat(
        state["history"],
        user_mode=False,
    )


def user_chat_html(state):
    return render_chat(
        state["history"],
        user_mode=True,
    )


def current_decision_html(state):
    if not state["trace"]:
        return render_decision(None)

    return render_decision(
        state["trace"][-1]
    )


def trace_html(state):
    return render_trace(
        state["trace"]
    )

def chat(user_input, state):

    if state is None:
        state = new_session()

    user_input = str(
        user_input or ""
    ).strip()

    if not user_input:
        return (
            "",
            user_chat_html(state),
            developer_chat_html(state),
            current_decision_html(state),
            trace_html(state),
            session_html(state),
            state,
        )

    history = list(
        state["history"]
    )

    history.append({
        "role": "user",
        "content": user_input,
    })

    try:
        response = state["agent"].reply(
            history
        )

    except Exception as exc:
        response = (
            "Agent 执行失败："
            f"{type(exc).__name__}: {exc}"
        )

    response = str(
        response or ""
    ).strip()

    history.append({
        "role": "assistant",
        "content": response,
    })

    state["history"] = history
    state["turns"] += 1

    snapshot = get_last_run(
        state["agent"]
    )

    snapshot = dict(snapshot)

    snapshot["turn"] = state["turns"]
    snapshot["user"] = user_input
    snapshot["response"] = response

    state["trace"].append(
        snapshot
    )

    return (
        "",
        user_chat_html(state),
        developer_chat_html(state),
        current_decision_html(state),
        trace_html(state),
        session_html(state),
        state,
    )

# ============================================================
# Load Prefix
# ============================================================

def load_prefix(prefix_text, state):

    if state is None:
        state = new_session()

    history = parse_prefix(
        prefix_text
    )

    if not history:
        return (
            developer_chat_html(state),
            current_decision_html(state),
            trace_html(state),
            session_html(state),
            state,
            render_prefix_result(
                False,
                "没有识别到有效对话。",
            ),
        )

    state["agent"] = MemoryAgent(
        mode="single",
        show_state=False,
    )

    state["history"] = history
    state["trace"] = []

    state["turns"] = sum(
        1
        for item in history
        if item.get("role") == "user"
    )

    return (
        developer_chat_html(state),
        current_decision_html(state),
        trace_html(state),
        session_html(state),
        state,
        render_prefix_result(
            True,
            f"已加载 {len(history)} 条历史消息。",
        ),
    )

def escape(value):
    if value is None:
        return ""

    return html.escape(
        str(value)
    )


def json_text(value):
    try:
        return json.dumps(
            value,
            ensure_ascii=False,
            indent=2,
        )

    except Exception:
        return str(value)

    if text is None:

    return (
        f'<span class="badge badge-{kind}">'
        f'{escape(text)}'

def render_chat(history, user_mode=False):

    if not history:
        return render_empty_chat(
            user_mode=user_mode
        )

    rows = []

    for message in history:

        role = message.get(
            "role"
        )

        content = escape(
            message.get(
                "content",
                "",
            )
        )

        if role == "user":

            rows.append(
                f"""
                <div class="message-row user-row">

                    <div class="message-wrap user-wrap">

                        <div class="message-role user-role">
                            用户
                        </div>

                        <div class="message-bubble user-message">
                            {content}
                        </div>

                    </div>

                    <div class="avatar user-avatar">
                        U
                    </div>

                </div>
                """
            )

        elif role == "assistant":

            rows.append(
                f"""
                <div class="message-row agent-row">

                    <div class="avatar agent-avatar">
                        AI
                    </div>

                    <div class="message-wrap agent-wrap">

                        <div class="message-role">
                            智能客服
                        </div>

                        <div class="message-bubble agent-message">
                            {content}
                        </div>

                    </div>

                </div>
                """
            )

    return f"""
    <div class="chat-surface">

        <div class="chat-scroll">
            {''.join(rows)}
        </div>

    </div>
    """

def render_topic_tree(tree):

    if not tree:

        return """
        <div class="empty-small">
            暂无 Topic Structure
        </div>
        """

    def walk(node, depth=0):

        topic = escape(
            node.get(
                "topic",
                "",
            )
        )

        dimension = escape(
            node.get(
                "dimension",
                "",
            )
        )

        case_ids = (
            node.get("case_ids")
            or []
        )

        if isinstance(
            case_ids,
            str,
        ):
            case_ids = [case_ids]

        case_text = ", ".join(
            escape(x)
            for x in case_ids
        )

        leaf = bool(
            node.get("leaf")
        )

        node_class = (
            "tree-node tree-leaf"
            if leaf
            else "tree-node"
        )

        margin = depth * 20

        dimension_html = ""

        if dimension:
            dimension_html = (
                f'<span class="tree-dimension">'
                f'{dimension}'
                f'</span>'
            )

        case_html = ""

        if case_text:
            case_html = (
                f'<span class="tree-case">'
                f'{case_text}'
                f'</span>'
            )

        current = f"""
        <div
            class="{node_class}"
            style="margin-left:{margin}px"
        >

            <span class="tree-dot"></span>

            <span class="tree-topic">
                {topic}
            </span>

            {dimension_html}

            {case_html}

        </div>
        """

        children = (
            node.get("children")
            or []
        )

        return current + "".join(
            walk(
                child,
                depth + 1,
            )
            for child in children
        )

    return f"""
    <div class="topic-tree">
        {walk(tree)}
    </div>
    """

def render_cases(cases):

    if not cases:
        return """
        <div class="empty-small">
            当前没有 Active Cases
        </div>
        """

    items = []

    for index, case in enumerate(cases):
        if not isinstance(case, dict):
            continue
        case_id = (
            case.get("case_id")
            or case.get("id")
            or case.get("caseId")
            or f"Case {index + 1}"
        )

        # ----------------------------------------------------
        # 尝试寻找适合作为卡片摘要的字段
        # ----------------------------------------------------

        title = (
            case.get("title")
            or case.get("name")
            or case.get("question")
            or case.get("text")
            or case.get("content")
            or case.get("problem")
            or ""
        )

        title = str(title).strip()

        # 卡片外面只显示摘要
        preview = title

        if len(preview) > 100:
            preview = preview[:100] + "..."

        # ----------------------------------------------------
        # 完整内容
        # ----------------------------------------------------
        detail_rows = []
        for key, value in case.items():
            if value is None:
                continue
            if isinstance(value, (dict, list)):
                value_text = json_text(value)
            else:
                value_text = str(value)
            if not value_text.strip():
                continue
            detail_rows.append(
                f"""
                <div class="case-detail-row">
                    <div class="case-detail-key">
                        {escape(key)}
                    </div>
                    <div class="case-detail-value">
                        {escape(value_text)}
                    </div>
                </div>
                """
            )
        details_html = "".join(detail_rows)
        items.append(
            f"""
            <details class="case-card">
                <summary class="case-card-summary">
                    <div class="case-card-main">
                        <div class="case-card-id">
                            {escape(case_id)}
                        </div>
                        <div class="case-card-preview">
                            {escape(preview) if preview else "点击查看案例详情"}
                        </div>
                    </div>
                    <div class="case-card-action">
                        <span class="case-expand-text">
                            查看详情
                        </span>
                        <span class="case-chevron">
                            ›
                        </span>
                    </div>
                </summary>
                <div class="case-detail">
                    <div class="case-detail-header">
                        Case Detail
                    </div>
                    {details_html}
                </div>
            </details>
            """
        )
    if not items:
        return """
        <div class="empty-small">
            当前没有可显示的 Active Cases
        </div>
        """
    return "".join(items)

def render_latency(latency):

    latency = latency or {}

    def ms_value(name):
        value = latency.get(
            name,
            0,
        )

        try:
            return float(value)
        except Exception:
            return 0.0

# ============================================================

def render_trace(trace):

    if not trace:

        return """
        <div class="trace-surface">

            <div class="decision-empty">

                <div class="decision-empty-title">
                    暂无运行轨迹
                </div>

                <div class="decision-empty-desc">
                    每完成一轮交互，执行轨迹会显示在这里。
                </div>

            </div>

        </div>
        """

    items = []

    for index, item in enumerate(
        reversed(trace),
        start=1,
    ):

        turn = item.get(
            "turn",
            len(trace) - index + 1,
        )

        user = escape(
            item.get(
                "user",
                "",
            )
        )

        query = (
            item.get("query_state")
            or {}
        )

        trigger = (
            item.get("trigger_state")
            or {}
        )

        policy = (
            item.get("policy_state")
            or {}
        )

        latency = (
            item.get("latency")
            or {}
        )

        try:
            total = (
                float(
                    latency.get(
                        "total_ms",
                        0,
                    )
                )
                / 1000.0
            )
        except Exception:
            total = 0.0

        items.append(
            f"""
            <div class="trace-item">

                <div class="trace-head">

                    <b>
                        Turn {turn}
                    </b>

                    <span>
                        {total:.2f}s
                    </span>

                </div>

                <div class="trace-user">
                    {user}
                </div>

                <div class="trace-flow">

                    {badge(
                        query.get(
                            "query_mode",
                            "none",
                        ),
                        "blue"
                    )}

                    <span class="flow-arrow">
                        →
                    </span>

                    {badge(
                        trigger.get(
                            "evidence_mode",
                            "",
                        ),
                        "gray"
                    )}

                    <span class="flow-arrow">
                        →
                    </span>

                    {badge(
                        policy.get(
                            "label",
                            "",
                        ),
                        "green"
                    )}

                </div>

            </div>
            """
        )

    return f"""
    <div class="trace-surface">
        {''.join(items)}
    </div>
    """

def parse_json(text):
    if isinstance(text, dict):
        return text
    if not isinstance(text, str):
        return {}

    text = text.strip()
    if not text:
        return {}

    try:
        return json.loads(text)
    except Exception:
        pass

    if "```json" in text:
        try:
            text = text.split("```json", 1)[1].split("```", 1)[0].strip()
            return json.loads(text)
        except Exception:
            return {}

    if "```" in text:
        try:
            text = text.split("```", 1)[1].split("```", 1)[0].strip()
            return json.loads(text)
        except Exception:
            return {}

    return {}

def normalize_split(split_result, expected_case_ids):
    if not isinstance(split_result, dict):
        return None, "output is not a dict"

    split = split_result.get("split")
    if not isinstance(split, bool):
        return None, "missing or invalid split"

    if not split:
        return {
            "split": False,
            "dimension": "",
            "children": [],
        }, ""

    dimension = str(split_result.get("dimension", "")).strip()
    assignments = split_result.get("assignments", [])

    if not dimension:
        return None, "missing dimension"

    if not isinstance(assignments, list):
        return None, "assignments must be a list"

    expected_set = set(expected_case_ids)
    assignment_map = {}

    for item in assignments:
        if not isinstance(item, dict):
            return None, "assignment is not a dict"

        case_id = str(item.get("case_id", "")).strip()
        topic = str(item.get("topic", "")).strip()

        if not case_id or not topic:
            return None, "assignment missing case_id or topic"

        if case_id not in expected_set:
            return None, f"unknown case_id: {case_id}"

        if case_id in assignment_map:
            return None, f"duplicated case_id: {case_id}"

        assignment_map[case_id] = topic

    missing = [
        case_id
        for case_id in expected_case_ids
        if case_id not in assignment_map
    ]

    if missing:
        return None, f"missing case_ids: {missing}"

    groups = {}

    for case_id in expected_case_ids:
        topic = assignment_map[case_id]
        groups.setdefault(topic, []).append(case_id)

    if len(groups) < 2:
        return None, "fewer than 2 valid topic groups"

    children = [
        {
            "topic": topic,
            "case_ids": case_ids,
        }
        for topic, case_ids in groups.items()
    ]

    return {
        "split": True,
        "dimension": dimension,
        "children": children,
    }, ""


def stop_topic_refinement(node, split_status):
    node["leaf"] = True
    node["dimension"] = ""
    node["children"] = []
    node["split_status"] = split_status


def normalize_case_ids(value):
    if value is None:
        return []
    if isinstance(value, str):
        value = [value]
    if not isinstance(value, list):
        return []
    return list(dict.fromkeys(str(x).strip() for x in value if str(x).strip()))


def format_topic_tree(tree):
    if not tree:
        return "None"

    lines = []

    def node_text(node):
        topic = str(node.get("topic", "")).strip()
        dimension = str(node.get("dimension", "")).strip()
        case_ids = normalize_case_ids(node.get("case_ids"))

        if node.get("leaf"):
            if case_ids:
                ids = ", ".join(case_ids)
                return f"{topic}  [{ids}]" if topic and topic not in case_ids else f"[{ids}]"
            return topic or "Unknown"

        if dimension:
            return f"{topic}  <{dimension}>"

        return topic or "Unknown"

    def walk(node, prefix="", is_last=True, root=False):
        if root:
            lines.append(node_text(node))
        else:
            lines.append(prefix + ("└─ " if is_last else "├─ ") + node_text(node))

        children = node.get("children", [])
        if not children:
            return

        child_prefix = prefix if root else prefix + ("   " if is_last else "│  ")

        for i, child in enumerate(children):
            walk(child, child_prefix, i == len(children) - 1, False)

    walk(tree, root=True)
    return "\n".join(lines)

class MemoryAgent:
    """
    Stateful customer-service agent.

    One instance corresponds to one dialogue.
    Call reset() before starting a new dialogue if reusing it.
    """

    def __init__(
        self,
        prompt_dir="Agent/promptsv2_3",
        mode="single",
        max_topic_depth=8,
        show_state=True,
    ):
        prompt_dir = Path(prompt_dir)

        self.query_prompt = prompt_read(prompt_dir / "query.prompt")
        self.trigger_prompt = prompt_read(prompt_dir / "trigger.prompt")
        self.topic_prompt = prompt_read(prompt_dir / "topic.prompt")
        self.policy_prompt = prompt_read(prompt_dir / "policy.prompt")
        self.response_prompt = prompt_read(prompt_dir / "response.prompt")

        if mode not in {"single", "parallel"}:
            raise ValueError(f"Unsupported mode: {mode}")

        self.mode = mode
        self.max_topic_depth = max_topic_depth
        self.show_state = show_state

        self.memory = init_memory()

        # 对外暴露当前轮状态，供 Project / Web UI 使用。
        # 不参与 Agent 内部决策。
        self.last_turn_state = {}

    # ========================================================
    # LLM
    # ========================================================

    def call_llm(self, prompt, data):
        messages = build_messages(prompt, data)

        if self.mode == "single":
            output = single_inference(messages)
        else:
            output = parallel_inference([messages])[0]

        parsed = parse_llm_output(output)

        return parsed

    def call_llm_batch(self, messages):
        if not messages:
            return []

        if self.mode == "single":
            outputs = [single_inference(message) for message in messages]
        else:
            outputs = parallel_inference(messages)

        return [parse_llm_output(output) for output in outputs]

    # ========================================================
    # Memory Update
    # ========================================================

    def update_issue(self, query_state):
        mode = str(query_state.get("query_mode", "none")).strip()
        query = str(query_state.get("query_text", "")).strip()

        if mode in {"rewrite", "refine"} and query:
            self.memory["issue_state"]["current_issue"] = query
            self.memory["issue_state"]["query_history"].append(query)

    def update_evidence(self, trigger_state, evidence, topic_structure=None):
        if trigger_state.get("evidence_mode") != "retrieve_new_evidence":
            return

        self.memory["evidence_state"]["active_cases"] = evidence
        self.memory["evidence_state"]["topic_structure"] = topic_structure or {}

    def update_interaction(self, policy_state):
        label = str(policy_state.get("label", "")).strip()
        information = policy_state.get("information")

        policy = {
            "label": label,
            "information": information,
        }

        interaction = self.memory["interaction_state"]
        interaction["last_policy"] = policy
        interaction["policy_history"].append(policy)

        if label in {"AskMissingSlot", "AskClarification"}:
            interaction["stage"] = "Clarification"
        elif label in {"ExplainedResponse", "CaseRecommendation"}:
            interaction["stage"] = "Resolution"
        elif label == "Handoff":
            interaction["stage"] = "End"

    def update_dialogue(self, user, response):
        history = self.memory["dialogue_state"]["recent_history"]

        history.append({
            "user": user,
            "assistant": response,
        })

        self.memory["dialogue_state"]["recent_history"] = history[-5:]

    # ========================================================
    # Topic Structure Generation
    # ========================================================

    def generate_topic_structure(self, query, cases):
        case_map = case_map_from_cases(cases)
        case_ids = list(case_map)

        root = {
            "query": query,
            "topic": query,
            "case_ids": case_ids,
            "leaf": len(case_ids) <= 1,
            "children": [],
        }

        if not case_ids:
            return root

        if len(case_ids) == 1:
            root["case_id"] = case_ids[0]
            root["split_status"] = "single_case"
            return root

        frontier = [{
            "node": root,
            "case_ids": case_ids,
            "depth": 0,
            "parent_path": [],
        }]

        round_id = 0

        while frontier:
            print(f"[TopicStructure] Round {round_id}: {len(frontier)} nodes")

            active_frontier = []

            for task in frontier:
                if task["depth"] >= self.max_topic_depth:
                    print(f"[TopicStructure Warning] max_depth reached: {task['case_ids']}")
                    stop_topic_refinement(task["node"], "max_depth")
                else:
                    active_frontier.append(task)

            if not active_frontier:
                break

            messages = []

            for task in active_frontier:
                node_cases = [case_map[case_id] for case_id in task["case_ids"]]

                split_input = build_topic_split_input(
                    query=query,
                    cases=node_cases,
                    parent_path=task["parent_path"],
                    depth=task["depth"],
                )

                messages.append(build_messages(self.topic_prompt, split_input))

            outputs = self.call_llm_batch(messages)

            valid_results = {}

            for idx, (task, output) in enumerate(zip(active_frontier, outputs)):
                valid_split, error = normalize_split(output, task["case_ids"])

                if valid_split:
                    valid_results[idx] = valid_split
                else:
                    print(f"[TopicStructure Invalid] {error}")
                    print("output:", output, flush=True)

            next_frontier = []

            for idx, task in enumerate(active_frontier):
                node = task["node"]
                valid_split = valid_results.get(idx)

                if not valid_split:
                    print(
                        "[TopicStructure Warning] "
                        f"invalid output, stop refinement: {task['case_ids']}"
                    )
                    stop_topic_refinement(node, "invalid_output")
                    continue

                if valid_split["split"] is False:
                    print(
                        "[TopicStructure] "
                        f"no meaningful split: {task['case_ids']}"
                    )
                    stop_topic_refinement(node, "no_meaningful_split")
                    continue

                node["leaf"] = False
                node["dimension"] = valid_split["dimension"]
                node["children"] = []
                node["split_status"] = "valid"

                for child_info in valid_split["children"]:
                    child_case_ids = child_info["case_ids"]
                    child_topic = child_info["topic"]

                    child_node = {
                        "topic": child_topic,
                        "case_ids": child_case_ids,
                        "leaf": len(child_case_ids) == 1,
                        "children": [],
                    }

                    node["children"].append(child_node)

                    if len(child_case_ids) == 1:
                        child_node["case_id"] = child_case_ids[0]
                        child_node["split_status"] = "single_case"
                        continue

                    next_frontier.append({
                        "node": child_node,
                        "case_ids": child_case_ids,
                        "depth": task["depth"] + 1,
                        "parent_path": task["parent_path"] + [child_topic],
                    })

            frontier = next_frontier
            round_id += 1

        return root

    # ========================================================
    # State Display
    # ========================================================

    def print_state(self, query_state, trigger_state, policy_state):
        if not self.show_state:
            return

        query_mode = query_state.get("query_mode", "")
        query_text = query_state.get("query_text", "")
        trigger = trigger_state.get("trigger", "")
        evidence_mode = trigger_state.get("evidence_mode", "")

        active_cases = self.memory["evidence_state"]["active_cases"]
        case_ids = [get_case_id(case) for case in active_cases if get_case_id(case)]

        policy = policy_state.get("label", "")
        information = policy_state.get("information")
        topic_tree = format_topic_tree(self.memory["evidence_state"]["topic_structure"])

        print("\n╔══════════════ AGENT STATE ══════════════╗")
        print(f" Query    : {query_mode} | {query_text}")
        print(f" Trigger  : {trigger}")
        print(f" Evidence : {evidence_mode}")
        print(f" Cases    : {', '.join(case_ids) if case_ids else 'None'}")

        print("\n Topic Tree")
        for line in topic_tree.splitlines():
            print(f" {line}")

        print(f"\n Policy   : {policy}")
        print(f" Info     : {information if information is not None else 'None'}")
        print("╚══════════════════════════════════════════╝")

    # ========================================================
    # Reply
    # ========================================================

    def reply(self, dialogue_history):
        """
        dialogue_history:
        [
            {"role": "user", "content": "..."},
            {"role": "assistant", "content": "..."},
            ...
            {"role": "user", "content": "..."},
        ]

        Returns:
            str: customer-service response
        """

        if not dialogue_history:
            raise ValueError("dialogue_history is empty")

        last_msg = dialogue_history[-1]

        if last_msg.get("role") != "user":
            raise ValueError("The last message in dialogue_history must be from user.")

        user = str(last_msg.get("content", "")).strip()

        if not user:
            raise ValueError("Current user message is empty")

        # 每轮开始时重置对外状态。
        self.last_turn_state = {}

        latency = {
            "query_ms": 0,
            "trigger_ms": 0,
            "rag_ms": 0,
            "topic_ms": 0,
            "policy_ms": 0,
            "response_ms": 0,
            "total_ms": 0,
        }

        total_start = time.perf_counter()

        # ----------------------------------------------------
        # 1. Query
        # ----------------------------------------------------

        start = time.perf_counter()

        query_state = self.call_llm(
            self.query_prompt,
            {
                "issue_state": self.memory["issue_state"],
                "dialogue_state": self.memory["dialogue_state"],
                "current": {"u": user},
            },
        )

        latency["query_ms"] = round(
            (time.perf_counter() - start) * 1000
        )

        self.update_issue(query_state)

        # ----------------------------------------------------
        # 2. Trigger
        # ----------------------------------------------------

        start = time.perf_counter()

        trigger_state = self.call_llm(
            self.trigger_prompt,
            {
                "query": {
                    "query_mode": query_state.get("query_mode", "none"),
                    "query_text": query_state.get("query_text", ""),
                },
                "active_cases": self.memory["evidence_state"]["active_cases"],
            },
        )

        latency["trigger_ms"] = round(
            (time.perf_counter() - start) * 1000
        )

        # ----------------------------------------------------
        # 3. RAG + Topic
        # ----------------------------------------------------

        evidence_mode = trigger_state.get("evidence_mode", "")

        if evidence_mode == "retrieve_new_evidence":
            query = str(query_state.get("query_text", "")).strip()

            if not query:
                query = self.memory["issue_state"]["current_issue"].strip()

            if not query:
                raise ValueError("RAG query is empty")

            # ---------------- RAG ----------------

            start = time.perf_counter()

            evidence = batch_zefeng_rag(
                [history_to_text(dialogue_history)],
                [query],
            )[0]

            latency["rag_ms"] = round(
                (time.perf_counter() - start) * 1000
            )

            # ---------------- Topic ----------------

            start = time.perf_counter()

            topic_structure = self.generate_topic_structure(
                query=query,
                cases=evidence,
            )

            latency["topic_ms"] = round(
                (time.perf_counter() - start) * 1000
            )

        else:
            evidence = self.memory["evidence_state"]["active_cases"]
            topic_structure = self.memory["evidence_state"]["topic_structure"]

        self.update_evidence(trigger_state, evidence, topic_structure)

        # ----------------------------------------------------
        # 4. Policy
        # ----------------------------------------------------

        if evidence_mode == "no_evidence_needed":
            policy_evidence = {
                "active_cases": [],
                "topic_structure": {},
            }
        else:
            policy_evidence = {
                "active_cases": self.memory["evidence_state"]["active_cases"],
                "topic_structure": self.memory["evidence_state"]["topic_structure"],
            }

        start = time.perf_counter()

        policy_state = self.call_llm(
            self.policy_prompt,
            {
                "issue_state": self.memory["issue_state"],
                "evidence_state": policy_evidence,
                "interaction_state": self.memory["interaction_state"],
                "dialogue_state": self.memory["dialogue_state"],
                "current": {"u": user},
            },
        )

        latency["policy_ms"] = round(
            (time.perf_counter() - start) * 1000
        )

        self.update_interaction(policy_state)

        # ----------------------------------------------------
        # 5. Response
        # ----------------------------------------------------

        selected_evidence = get_selected_evidence(
            policy_state,
            self.memory["evidence_state"]["active_cases"],
        )

        response_dialogue = {
            "recent_history": self.memory["dialogue_state"]["recent_history"],
            "current": {"u": user},
        }

        response_policy = {
            "label": policy_state.get("label", ""),
            "information": policy_state.get("information"),
        }

        start = time.perf_counter()

        response_state = self.call_llm(
            self.response_prompt,
            {
                "dialogue_state": response_dialogue,
                "policy": response_policy,
                "evidence": selected_evidence,
            },
        )

        latency["response_ms"] = round(
            (time.perf_counter() - start) * 1000
        )

        response = (
            response_state.get("response")
            or response_state.get("text")
            or response_state.get("content")
            or ""
        )

        response = str(response).strip()

        self.update_dialogue(user, response)

        # ----------------------------------------------------
        # 6. Latency + External Turn State
        # ----------------------------------------------------

        latency["total_ms"] = round(
            (time.perf_counter() - total_start) * 1000
        )

        self.last_turn_state = {
            "query": query_state,
            "trigger": trigger_state,
            "evidence": {
                "active_cases": self.memory["evidence_state"]["active_cases"],
                "topic_structure": self.memory["evidence_state"]["topic_structure"],
            },
            "policy": policy_state,
            "response": response,
            "latency": latency,
        }

        # ----------------------------------------------------
        # Display
        # ----------------------------------------------------

        self.print_state(
            query_state=query_state,
            trigger_state=trigger_state,
            policy_state=policy_state,
        )

        return response

    def reset(self):
        self.memory = init_memory()
        self.last_turn_state = {}
