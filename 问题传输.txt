# ============================================================
# Utils
# ============================================================

def now_iso():
    return datetime.now().astimezone().isoformat(
        timespec="seconds"
    )


# ============================================================
# Session
# ============================================================

def new_session():

    return {
        "session_id":
            "web_" + uuid.uuid4().hex[:10],

        "created_at":
            now_iso(),

        "agent": MemoryAgent(
            mode="single",
            show_state=False,
        ),

        "history": [],

        # 系统真实运行产生的 turn
        "trace": [],

        # 用户加载的已有历史
        "prefix_history": [],

        "turns": 0,

        "mode": "user",

        # {
        #     turn_id: {
        #         rating,
        #         reason,
        #         comment,
        #         created_at
        #     }
        # }
        "feedback": {},
    }


# ============================================================
# Agent Snapshot
# ============================================================

def get_last_run(agent):

    turn_state = getattr(
        agent,
        "last_turn_state",
        None,
    )

    if turn_state:

        return {
            "query_state":
                turn_state.get(
                    "query",
                    {},
                ) or {},

            "trigger_state":
                turn_state.get(
                    "trigger",
                    {},
                ) or {},

            "evidence_state":
                turn_state.get(
                    "evidence",
                    {},
                ) or {},

            "policy_state":
                turn_state.get(
                    "policy",
                    {},
                ) or {},

            "response":
                turn_state.get(
                    "response",
                    "",
                ),

            "latency":
                turn_state.get(
                    "latency",
                    {},
                ) or {},
        }

    # fallback
    memory = getattr(
        agent,
        "memory",
        {},
    ) or {}

    return {
        "query_state": {},

        "trigger_state": {},

        "evidence_state":
            memory.get(
                "evidence_state",
                {},
            ) or {},

        "policy_state":
            (
                memory
                .get(
                    "interaction_state",
                    {},
                )
                .get(
                    "last_policy",
                    {},
                )
                or {}
            ),

        "response": "",

        "latency": {},
    }


# ============================================================
# History Parser
# ============================================================

def parse_prefix(text):

    history = []

    if not text:
        return history

    for raw_line in text.splitlines():

        line = raw_line.strip()

        if not line:
            continue

        if line.startswith("用户:"):

            content = line.split(
                "用户:",
                1,
            )[1].strip()

            history.append({
                "role": "user",
                "content": content,
            })

        elif line.startswith("用户："):

            content = line.split(
                "用户：",
                1,
            )[1].strip()

            history.append({
                "role": "user",
                "content": content,
            })

        elif line.startswith("客服:"):

            content = line.split(
                "客服:",
                1,
            )[1].strip()

            history.append({
                "role": "assistant",
                "content": content,
            })

        elif line.startswith("客服："):

            content = line.split(
                "客服：",
                1,
            )[1].strip()

            history.append({
                "role": "assistant",
                "content": content,
            })

    return history


# ============================================================
# Render Helpers
# ============================================================

def session_html(state):

    return render_session(
        state["session_id"],
        state["turns"],
    )


def developer_chat_html(state):

    return render_chat(
        state["history"],
        user_mode=False,
    )


def user_chat_html(state):

    return render_chat(
        state["history"],
        user_mode=True,
    )


def current_decision_html(state):

    if not state["trace"]:
        return render_decision(
            None
        )

    return render_decision(
        state["trace"][-1]
    )


def trace_html(state):

    return render_trace(
        state["trace"],
        state.get("feedback", {}),
    )



def feedback_hint_html(state):

    if not state:
        return render_feedback_hint(
            None
        )

    turn = state.get(
        "turns",
        0,
    )

    if turn <= 0:
        return render_feedback_hint(
            None
        )

    return render_feedback_hint(
        turn
    )

# Chat
# ============================================================

def chat(
    user_input,
    state,
):

    if state is None:
        state = new_session()

    user_input = str(
        user_input or ""
    ).strip()

    if not user_input:

        return (
            "",
            user_chat_html(state),
            developer_chat_html(state),
            current_decision_html(state),
            trace_html(state),
            session_html(state),
            state,
            feedback_hint_html(state),
            "",
            gr.update(visible=False),
        )

    history = list(
        state["history"]
    )

    history.append({
        "role": "user",
        "content": user_input,
    })

    try:

        response = state[
            "agent"
        ].reply(
            history
        )

    except Exception as exc:

        response = (
            "Agent 执行失败："
            f"{type(exc).__name__}: {exc}"
        )

    response = str(
        response or ""
    ).strip()

    history.append({
        "role": "assistant",
        "content": response,
    })

    state["history"] = history

    state["turns"] += 1

    snapshot = get_last_run(
        state["agent"]
    )

    snapshot = dict(
        snapshot
    )

    snapshot["turn"] = (
        state["turns"]
    )

    snapshot["user"] = (
        user_input
    )

    snapshot["response"] = (
        response
    )

    snapshot["created_at"] = (
        now_iso()
    )

    state["trace"].append(
        snapshot
    )

    # --------------------------------------------------------
    # 每轮立即保存
    # --------------------------------------------------------

    save_session(
        state
    )

    return (
        "",
        user_chat_html(state),
        developer_chat_html(state),
        current_decision_html(state),
        trace_html(state),
        session_html(state),
        state,
        feedback_hint_html(state),
        "",
        gr.update(visible=False),
    )

def submit_positive(
    state,
):

    if state is None:

        return (
            state,

            render_feedback_result(
                False,
                "当前没有可反馈的回复。",
            ),

            gr.update(
                visible=False
            ),

            render_trace(
                [],
                {},
            ),
        )

    turn = state.get(
        "turns",
        0,
    )

    if turn <= 0:

        return (
            state,

            render_feedback_result(
                False,
                "当前没有可反馈的回复。",
            ),

            gr.update(
                visible=False
            ),

            trace_html(state),
        )

    state.setdefault(
        "feedback",
        {},
    )

    state["feedback"][turn] = {
        "rating": "positive",
        "reason": None,
        "reason_text": None,
        "comment": "",
        "created_at": now_iso(),
    }

    # 保存到后台日志
    save_session(
        state
    )

    return (
        state,

        render_feedback_result(
            True,
            "感谢您的反馈。",
        ),

        gr.update(
            visible=False
        ),

        # 立即刷新开发者运行轨迹
        trace_html(state),
    )



def show_negative_feedback(
    state,
):

    if (
        state is None
        or state.get("turns", 0) <= 0
    ):

        return (
            gr.update(
                visible=False
            ),

            render_feedback_result(
                False,
                "当前没有可反馈的回复。",
            ),
        )

    return (
        gr.update(
            visible=True
        ),

        "",
    )


def submit_negative(
    reason_text,
    comment,
    state,
):

    if state is None:

        return (
            state,

            render_feedback_result(
                False,
                "当前没有可反馈的回复。",
            ),

            gr.update(
                visible=True
            ),

            reason_text,

            comment,

            render_trace(
                [],
                {},
            ),
        )

    turn = state.get(
        "turns",
        0,
    )

    if turn <= 0:

        return (
            state,

            render_feedback_result(
                False,
                "当前没有可反馈的回复。",
            ),

            gr.update(
                visible=True
            ),

            reason_text,

            comment,

            trace_html(state),
        )

    reason_text = str(
        reason_text or ""
    ).strip()

    if not reason_text:

        return (
            state,

            render_feedback_result(
                False,
                "请选择一个不满意原因。",
            ),

            gr.update(
                visible=True
            ),

            reason_text,

            comment,

            trace_html(state),
        )

    reason_code = FEEDBACK_REASONS.get(
        reason_text,
        "other",
    )

    comment = str(
        comment or ""
    ).strip()

    state.setdefault(
        "feedback",
        {},
    )

    state["feedback"][turn] = {
        "rating": "negative",
        "reason": reason_code,
        "reason_text": reason_text,
        "comment": comment,
        "created_at": now_iso(),
    }

    # 保存到后台日志
    save_session(
        state
    )

    return (
        state,

        render_feedback_result(
            True,
            "感谢您的反馈。",
        ),

        gr.update(
            visible=False
        ),

        None,

        "",

        # 立即刷新开发者运行轨迹
        trace_html(state),
    )

# ============================================================
# Load Prefix
# ============================================================

def load_prefix(
    prefix_text,
    state,
):

    if state is None:
        state = new_session()

    history = parse_prefix(
        prefix_text
    )

    if not history:

        return (
            developer_chat_html(state),
            current_decision_html(state),
            trace_html(state),
            session_html(state),
            state,

            render_prefix_result(
                False,
                "没有识别到有效对话。",
            ),
        )

    state["agent"] = MemoryAgent(
        mode="single",
        show_state=False,
    )

    state["history"] = history

    # 这里不把加载历史伪装成真实 Agent trace
    state["trace"] = []

    state["prefix_history"] = list(
        history
    )

    # 真实运行 turn 从 0 开始重新记录
    state["turns"] = 0

    state["feedback"] = {}

    save_session(
        state
    )

    return (
        developer_chat_html(state),
        current_decision_html(state),
        trace_html(state),
        session_html(state),
        state,

        render_prefix_result(
            True,
            f"已加载 {len(history)} 条历史消息。",
        ),
    )

def now_iso():

        timespec="seconds"

def new_session():

    return {
        "session_id":
            "web_" + uuid.uuid4().hex[:10],

        "created_at":
            now_iso(),

        "agent": MemoryAgent(
            mode="single",
            show_state=False,
        ),

        # 系统真实运行产生的 turn
        "trace": [],

        # 用户加载的已有历史
        "prefix_history": [],

        "turns": 0,

        "mode": "user",

        # {
        #     turn_id: {

        #         reason,
        #         comment,
        #         created_at
        #     }

        "feedback": {},

# ============================================================
# Agent Snapshot
# ============================================================

def get_last_run(agent):

    turn_state = getattr(
        agent,
        "last_turn_state",
        None,
    )

    if turn_state:

        return {
            "query_state":
                turn_state.get(
                    "query",
                    {},
                ) or {},

            "trigger_state":
                turn_state.get(
                    "trigger",
                    {},
                ) or {},

            "evidence_state":
                turn_state.get(
                    "evidence",
                    {},
                ) or {},

            "policy_state":
                turn_state.get(
                    "policy",
                    {},
                ) or {},

            "response":
                turn_state.get(
                    "response",
                    "",
                ),

            "latency":
                turn_state.get(
                    "latency",
                    {},
                ) or {},
        }

    # fallback
    memory = getattr(
        agent,
        "memory",
        {},
    ) or {}

    return {
        "query_state": {},

        "trigger_state": {},

        "evidence_state":
            memory.get(
                "evidence_state",
                {},
            ) or {},

        "policy_state":
            (
                memory
                .get(
                    "interaction_state",
                    {},
                )
                .get(
                    "last_policy",
                    {},
                )
                or {}
            ),

        "response": "",

        "latency": {},
    }


# ============================================================
# History Parser
# ============================================================

def parse_prefix(text):

    history = []

    if not text:
        return history

    for raw_line in text.splitlines():

        line = raw_line.strip()

        if not line:
            continue

        if line.startswith("用户:"):

            content = line.split(
                "用户:",
                1,
            )[1].strip()

            history.append({
                "role": "user",
                "content": content,
            })

        elif line.startswith("用户："):

            content = line.split(
                "用户：",
                1,
            )[1].strip()

            history.append({
                "role": "user",
                "content": content,
            })

        elif line.startswith("客服:"):

            content = line.split(
                "客服:",
                1,
            )[1].strip()

            history.append({
                "role": "assistant",
                "content": content,
            })

        elif line.startswith("客服："):

            content = line.split(
                "客服：",
                1,
            )[1].strip()

            history.append({
                "role": "assistant",
                "content": content,
            })

    return history

def chat(
    user_input,
    state,
):

    if state is None:
        state = new_session()

    user_input = str(
        user_input or ""
    ).strip()

    if not user_input:

        return (
            "",
            user_chat_html(state),
            developer_chat_html(state),
            current_decision_html(state),
            trace_html(state),
            session_html(state),
            state,
            feedback_hint_html(state),
            "",
            gr.update(visible=False),
        )

    history = list(
        state["history"]
    )

    history.append({
        "role": "user",
        "content": user_input,
    })

    try:

        response = state[
            "agent"
        ].reply(
            history
        )

    except Exception as exc:

        response = (
            "Agent 执行失败："
            f"{type(exc).__name__}: {exc}"
        )

    response = str(
        response or ""
    ).strip()

    history.append({
        "role": "assistant",
        "content": response,
    })

    state["history"] = history

    state["turns"] += 1

    snapshot = get_last_run(
        state["agent"]
    )

    snapshot = dict(
        snapshot
    )

    snapshot["turn"] = (
        state["turns"]
    )

    snapshot["user"] = (
        user_input
    )

    snapshot["response"] = (
        response
    )

    snapshot["created_at"] = (
        now_iso()
    )

    state["trace"].append(
        snapshot
    )

    # --------------------------------------------------------
    # 每轮立即保存
    # --------------------------------------------------------

    save_session(
        state
    )

    return (
        "",
        user_chat_html(state),
        developer_chat_html(state),
        current_decision_html(state),
        trace_html(state),
        session_html(state),
        state,
        feedback_hint_html(state),
        "",
        gr.update(visible=False),
    )

def submit_positive(
    state,
):

    if state is None:

        return (
            state,

            render_feedback_result(
                False,
                "当前没有可反馈的回复。",
            ),

            gr.update(
                visible=False
            ),

            render_trace(
                [],
                {},
            ),
        )

    turn = state.get(
        "turns",
        0,
    )

    if turn <= 0:

        return (
            state,

            render_feedback_result(
                False,
                "当前没有可反馈的回复。",
            ),

            gr.update(
                visible=False
            ),

            trace_html(state),
        )

    state.setdefault(
        "feedback",
        {},
    )

    state["feedback"][turn] = {
        "rating": "positive",
        "reason": None,
        "reason_text": None,
        "comment": "",
        "created_at": now_iso(),
    }

    # 保存到后台日志
    save_session(
        state
    )

    return (
        state,

        render_feedback_result(
            True,
            "感谢您的反馈。",
        ),

        gr.update(
            visible=False
        ),

        # 立即刷新开发者运行轨迹
        trace_html(state),
    )

        ),

    )


def submit_negative(
    reason_text,
    comment,
    state,
):

    if state is None:

        return (
            state,

            render_feedback_result(
                False,
                "当前没有可反馈的回复。",
            ),

            gr.update(
                visible=True
            ),

            reason_text,

            comment,

            render_trace(
                [],
                {},
            ),
        )

    turn = state.get(
        "turns",
        0,
    )

    if turn <= 0:

        return (
            state,

            render_feedback_result(
                False,
                "当前没有可反馈的回复。",
            ),

            gr.update(
                visible=True
            ),

            reason_text,

            comment,

            trace_html(state),
        )

    reason_text = str(
        reason_text or ""
    ).strip()

    if not reason_text:

        return (
            state,

            render_feedback_result(
                False,
                "请选择一个不满意原因。",
            ),

            gr.update(
                visible=True
            ),

            reason_text,

            comment,

            trace_html(state),
        )

    reason_code = FEEDBACK_REASONS.get(
        reason_text,
        "other",
    )

    comment = str(
        comment or ""
    ).strip()

    state.setdefault(
        "feedback",
        {},
    )

    state["feedback"][turn] = {
        "rating": "negative",
        "reason": reason_code,
        "reason_text": reason_text,
        "comment": comment,
        "created_at": now_iso(),
    }

    # 保存到后台日志
    save_session(
        state
    )

    return (
        state,

        render_feedback_result(
            True,
            "感谢您的反馈。",
        ),

        gr.update(
            visible=False
        ),

        None,

        "",

        # 立即刷新开发者运行轨迹
        trace_html(state),
    )

# Load Prefix
# ============================================================

def load_prefix(
    prefix_text,
    state,
):

    if state is None:
        state = new_session()

    history = parse_prefix(
        prefix_text
    )

    if not history:

        return (
            developer_chat_html(state),
            current_decision_html(state),
            trace_html(state),
            session_html(state),
            state,

            render_prefix_result(
                False,
                "没有识别到有效对话。",
            ),
        )

    state["agent"] = MemoryAgent(
        mode="single",
        show_state=False,
    )

    state["history"] = history

    # 这里不把加载历史伪装成真实 Agent trace
    state["trace"] = []

    state["prefix_history"] = list(
        history
    )

    # 真实运行 turn 从 0 开始重新记录
    state["turns"] = 0

    state["feedback"] = {}

    save_session(
        state
    )

    return (
        developer_chat_html(state),
        current_decision_html(state),
        trace_html(state),
        session_html(state),
        state,

        render_prefix_result(
            True,
            f"已加载 {len(history)} 条历史消息。",
        ),
    )
