def print_state(
    self,
    query_state,
    trigger_state,
    evidence,
    policy_state,
):
    # Query
    query_mode = query_state.get("query_mode", "")
    query_text = query_state.get("query_text", "")

    # Trigger / Evidence
    trigger = trigger_state.get("trigger", "")
    evidence_mode = trigger_state.get("evidence_mode", "")

    # Top-K Case IDs
    case_ids = []

    for case in evidence or []:
        if not isinstance(case, dict):
            continue

        case_id = (
            case.get("case_id")
            or case.get("id")
            or case.get("caseId")
        )

        if case_id:
            case_ids.append(str(case_id))

    # Policy
    policy = policy_state.get("label", "")

    selected_case = (
        policy_state.get("case_id")
        or policy_state.get("selected_case")
        or ""
    )

    # Print
    print("\n╔══════════════ AGENT STATE ══════════════╗")
    print(f" Query    : {query_mode} | {query_text}")
    print(f" Trigger  : {trigger}")
    print(f" Evidence : {evidence_mode}")
    print(
        f" Cases    : "
        f"{', '.join(case_ids) if case_ids else 'None'}"
    )
    print(f" Policy   : {policy}")
    print(f" Case     : {selected_case or 'None'}")
    print("╚══════════════════════════════════════════╝")
