def run_refinement_pe_batch(
    items,
    memories,
    refinement_prompt,
    max_depth=4,
):
    """
    Dynamic Batched Difference-Guided Refinement

    items:
        当前需要重新构建 topic structure 的 current_batch 子集

    每个 depth：
        1. 对所有 active samples 一起调用 refinement.prompt
        2. 每个 sample 自己判断 decision_sufficient
        3. insufficient 的更新一层 structure
        4. 只保留仍需 refinement 的 samples 进入下一 depth

    样本内 sequential，样本间 parallel。
    """

    # ========================================================
    # 1. Initialize Structures
    # ========================================================

    active_items = []

    for x in items:

        dialog_id = x["dialog_id"]
        memory = memories[dialog_id]

        knowledge_state = memory[
            "evidence_state"
        ].get("knowledge_state", {})

        if not knowledge_state:
            x["topic_structure"] = {}
            continue

        structure = init_topic_structure(
            knowledge_state
        )

        if not structure.get("topic"):
            x["topic_structure"] = {}
            continue

        # 当前样本自己的运行状态
        x["_refinement_structure"] = structure

        x["_refinement_frontier"] = [
            {
                "topic": structure["topic"],
                "case_ids": structure["case_ids"],
            }
        ]

        x["_refinement_history"] = []

        x["_refinement_done"] = False

        active_items.append(x)

    # ========================================================
    # 2. Depth-wise Dynamic Batch
    # ========================================================

    for depth in range(max_depth):

        if not active_items:
            break

        print(
            f"[Refinement Depth {depth}] "
            f"Active Samples: {len(active_items)}"
        )

        refinement_messages = []
        valid_items = []

        # ----------------------------------------------------
        # Build current-depth batch
        # ----------------------------------------------------

        for x in active_items:

            dialog_id = x["dialog_id"]
            memory = memories[dialog_id]

            structure = x[
                "_refinement_structure"
            ]

            frontier = x[
                "_refinement_frontier"
            ]

            if not frontier:
                # 没有 unresolved branch
                structure["decision_sufficient"] = True
                structure["final_depth"] = depth

                x["_refinement_done"] = True
                continue

            refinement_input = \
                build_refinement_input(
                    memory,
                    structure,
                    frontier,
                )

            refinement_messages.append(
                build_messages(
                    refinement_prompt,
                    refinement_input,
                )
            )

            valid_items.append(x)

        if not refinement_messages:
            break

        # ----------------------------------------------------
        # One batch inference at this depth
        # ----------------------------------------------------

        refinement_outputs = parallel_inference(
            refinement_messages
        )

        next_active_items = []

        # ----------------------------------------------------
        # Parse each sample
        # ----------------------------------------------------

        for x, out in zip(
            valid_items,
            refinement_outputs,
        ):

            result = parse_llm_output(out)

            structure = x[
                "_refinement_structure"
            ]

            frontier_before = copy.deepcopy(
                x["_refinement_frontier"]
            )

            x["_refinement_history"].append(
                {
                    "depth": depth,
                    "input_frontier":
                        frontier_before,
                    "output":
                        copy.deepcopy(result),
                }
            )

            # =================================================
            # A. Invalid output
            # =================================================

            if not result:

                structure["decision_sufficient"] = False
                structure["stop_reason"] = \
                    "invalid_refinement_output"

                structure["final_depth"] = depth

                x["_refinement_done"] = True
                continue

            # =================================================
            # B. Difference already sufficient
            # =================================================

            if result.get(
                "decision_sufficient",
                False,
            ):

                structure["decision_sufficient"] = True
                structure["stop_reason"] = \
                    "decision_sufficient"

                structure["final_depth"] = depth

                x["_refinement_done"] = True
                continue

            # =================================================
            # C. One-Level Refinement
            # =================================================

            refinements = result.get(
                "refinements",
                [],
            )

            if not refinements:

                structure["decision_sufficient"] = False
                structure["stop_reason"] = \
                    "no_valid_refinement"

                structure["final_depth"] = depth

                x["_refinement_done"] = True
                continue

            structure = apply_refinements(
                structure,
                refinements,
            )

            x[
                "_refinement_structure"
            ] = structure

            # =================================================
            # D. Get unresolved branches for next depth
            # =================================================

            next_frontier = result.get(
                "unresolved_branches",
                [],
            )

            x[
                "_refinement_frontier"
            ] = next_frontier

            # 如果模型已经没有 unresolved branch，
            # 则当前 refinement 后自然到达 frontier
            if not next_frontier:

                structure["decision_sufficient"] = True
                structure["stop_reason"] = \
                    "no_unresolved_branch"

                structure["final_depth"] = depth + 1

                x["_refinement_done"] = True
                continue

            # 仍需下一层 refinement
            next_active_items.append(x)

        active_items = next_active_items

    # ========================================================
    # 3. Max Depth
    # ========================================================

    for x in active_items:

        structure = x[
            "_refinement_structure"
        ]

        structure["decision_sufficient"] = False
        structure["stop_reason"] = "max_depth"
        structure["final_depth"] = max_depth

        x["_refinement_done"] = True

    # ========================================================
    # 4. Finalize
    # ========================================================

    for x in items:

        if "_refinement_structure" not in x:
            continue

        structure = x[
            "_refinement_structure"
        ]

        structure[
            "refinement_history"
        ] = x[
            "_refinement_history"
        ]

        x["topic_structure"] = structure

        dialog_id = x["dialog_id"]

        memories[
            dialog_id
        ] = update_topic_structure_memory(
            memories[dialog_id],
            structure,
        )

        # 清理运行时字段，不写进最终结果
        x.pop(
            "_refinement_structure",
            None,
        )

        x.pop(
            "_refinement_frontier",
            None,
        )

        x.pop(
            "_refinement_history",
            None,
        )

        x.pop(
            "_refinement_done",
            None,
        )






# ====================================================
# 5. Difference-Guided Adaptive Refinement Batch
# ====================================================

print(
    f"[Turn {t_id}] "
    f"Topic Refinement Batch"
)

refinement_items = []

for x in current_batch:

    memory = memories[
        x["dialog_id"]
    ]

    # -----------------------------------------------
    # 当前 Query / Evidence 没变化，复用旧 Structure
    # -----------------------------------------------

    if not need_structure_update(x):

        x["topic_structure"] = \
            copy.deepcopy(
                memory[
                    "evidence_state"
                ][
                    "topic_structure"
                ]
            )

        continue

    # -----------------------------------------------
    # 没有 Knowledge State
    # -----------------------------------------------

    if not memory[
        "evidence_state"
    ]["knowledge_state"]:

        x["topic_structure"] = {}
        continue

    # -----------------------------------------------
    # 加入本轮 Dynamic Refinement Batch
    # -----------------------------------------------

    refinement_items.append(x)


# 一次处理所有需要重新 refinement 的 samples
if refinement_items:

    run_refinement_pe_batch(
        items=refinement_items,
        memories=memories,
        refinement_prompt=
            refinement_prompt,
        max_depth=
            max_refinement_depth,
    )
