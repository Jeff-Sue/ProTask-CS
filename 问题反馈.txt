def format_topic_tree(
    topic_structure,
    max_depth=None,
):
    """
    将 topic_structure 转成简单易读的文本树。

    示例：

    Outlook邮箱最大扩容限制  <扩容场景>
    ├─ 最大容量限制
    │  ├─ 个人邮箱最大容量  [KT00270912]
    │  └─ 特殊邮箱容量限制  [KT00191614]
    └─ 邮箱扩容操作
       └─ 邮件小助手扩容  [KT00270348]
    """

    if not topic_structure:
        return "None"

    lines = []

    def node_label(node):
        topic = str(
            node.get("topic", "")
        ).strip()

        dimension = str(
            node.get("dimension", "")
        ).strip()

        case_id = str(
            node.get("case_id", "")
        ).strip()

        leaf = node.get(
            "leaf",
            False,
        )

        # fallback 节点可能 topic 本身就是 case id
        if leaf:

            if case_id:

                # 避免：
                # KT000123 [KT000123]
                if topic == case_id:
                    return case_id

                return (
                    f"{topic}  "
                    f"[{case_id}]"
                )

            return topic

        # 非叶子节点显示当前划分维度
        if dimension:
            return (
                f"{topic}  "
                f"<{dimension}>"
            )

        return topic

    def walk(
        node,
        prefix="",
        depth=0,
    ):
        children = node.get(
            "children",
            [],
        )

        if not isinstance(
            children,
            list,
        ):
            return

        if (
            max_depth is not None
            and depth >= max_depth
        ):
            if children:
                lines.append(
                    prefix + "└─ ..."
                )
            return

        for i, child in enumerate(
            children
        ):

            is_last = (
                i == len(children) - 1
            )

            connector = (
                "└─ "
                if is_last
                else "├─ "
            )

            lines.append(
                prefix
                + connector
                + node_label(child)
            )

            child_prefix = (
                prefix
                + (
                    "   "
                    if is_last
                    else "│  "
                )
            )

            walk(
                child,
                prefix=child_prefix,
                depth=depth + 1,
            )

    # Root
    lines.append(
        node_label(
            topic_structure
        )
    )

    walk(
        topic_structure
    )

    return "\n".join(lines)




def print_state(
    self,
    query_state,
    trigger_state,
    policy_state,
):
    if not self.show_state:
        return

    query_mode = (
        query_state.get(
            "query_mode",
            "",
        )
    )

    query_text = (
        query_state.get(
            "query_text",
            "",
        )
    )

    trigger = (
        trigger_state.get(
            "trigger",
            "",
        )
    )

    evidence_mode = (
        trigger_state.get(
            "evidence_mode",
            "",
        )
    )

    cases = self.memory[
        "evidence_state"
    ][
        "active_cases"
    ]

    case_ids = [
        get_case_id(case)
        for case in cases
        if get_case_id(case)
    ]

    topic_structure = self.memory[
        "evidence_state"
    ][
        "topic_structure"
    ]

    policy = policy_state.get(
        "label",
        "",
    )

    selected_case = (
        policy_state.get(
            "case_id"
        )
        or policy_state.get(
            "selected_case"
        )
        or ""
    )

    print(
        "\n"
        "╔══════════════ AGENT STATE ══════════════╗"
    )

    print(
        f" Query    : "
        f"{query_mode} | {query_text}"
    )

    print(
        f" Trigger  : {trigger}"
    )

    print(
        f" Evidence : {evidence_mode}"
    )

    print(
        " Cases    : "
        + (
            ", ".join(case_ids)
            if case_ids
            else "None"
        )
    )

    # ====================================================
    # Topic Tree
    # ====================================================

    print("\n Topic Tree")

    tree_text = format_topic_tree(
        topic_structure
    )

    for line in tree_text.splitlines():
        print(
            f" {line}"
        )

    # ====================================================

    print(
        f"\n Policy   : {policy}"
    )

    print(
        f" Case     : "
        f"{selected_case or 'None'}"
    )

    print(
        "╚══════════════════════════════════════════╝"
    )
