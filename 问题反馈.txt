def print_failure_summary(
    failure_summary,
    failure_details
):

    print(
        "\n========== "
        "案例推荐失败原因 "
        "=========="
    )

    total_failure = sum(
        failure_summary.values()
    )

    print(
        f"失败案例轮总数: "
        f"{total_failure}"
    )

    failure_order = [
        FAILURE_NO_TRIGGER,
        FAILURE_TOP5_MISS,
        FAILURE_WRONG_POLICY,
        FAILURE_WRONG_CASE
    ]

    for reason in failure_order:

        count = failure_summary.get(
            reason,
            0
        )

        ratio = (
            count / total_failure
            if total_failure
            else 0
        )

        print(
            f"\n{reason}: "
            f"{count} "
            f"({ratio:.2%})"
        )

        records = failure_details.get(
            reason,
            []
        )

        for record in records:

            print(
                f"  {record['golden_case']} "
                f"| Dialog={record['dialog_id']} "
                f"| Turn={record['turn_id']}"
            )
