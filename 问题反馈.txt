def run_refinement_pe_batch(
    items,
    memories,
    judgement_prompt,
    refinement_prompt,
    max_depth=4
):
    active_items = []

    for x in items:
        memory = memories[x["dialog_id"]]
        knowledge_state = memory["evidence_state"]["knowledge_state"]

        if not knowledge_state:
            x["topic_structure"] = {}
            continue

        structure = init_topic_structure(knowledge_state)

        if not structure.get("topic"):
            x["topic_structure"] = {}
            continue

        x["_structure"] = structure
        x["_frontier"] = [make_frontier_item(structure)]
        x["_judgement_history"] = []
        x["_refinement_history"] = []

        active_items.append(x)

    for depth in range(max_depth + 1):
        if not active_items:
            break

        # ====================================================
        # 1. Judgement Batch
        # ====================================================

        print(
            f"[Refinement Depth {depth}] "
            f"Judgement Batch: {len(active_items)}"
        )

        judgement_messages = []

        for x in active_items:
            memory = memories[x["dialog_id"]]

            judgement_input = build_judgement_input(
                memory,
                x["_structure"],
                x["_frontier"]
            )

            judgement_messages.append(
                build_messages(
                    judgement_prompt,
                    judgement_input
                )
            )

        judgement_outputs = parallel_inference(
            judgement_messages
        )

        need_refinement = []

        for x, out in zip(
            active_items,
            judgement_outputs
        ):
            judgement = parse_llm_output(out)
            structure = x["_structure"]

            x["_judgement_history"].append({
                "depth": depth,
                "frontier": copy.deepcopy(
                    x["_frontier"]
                ),
                "output": copy.deepcopy(
                    judgement
                )
            })

            if not validate_judgement(
                judgement,
                x["_frontier"]
            ):
                structure["decision_sufficient"] = False
                structure["evidence_status"] = "invalid"
                structure["stop_reason"] = "invalid_judgement"
                structure["final_depth"] = depth
                continue

            decision_sufficient = judgement.get(
                "decision_sufficient",
                False
            )

            matched_case_ids = judgement.get(
                "matched_case_ids",
                []
            )

            reason = judgement.get(
                "reason",
                ""
            )

            # --------------------------------------------
            # 已经足够
            # --------------------------------------------

            if decision_sufficient:
                structure["decision_sufficient"] = True
                structure["evidence_status"] = "sufficient"
                structure["matched_case_ids"] = matched_case_ids
                structure["judgement_reason"] = reason
                structure["stop_reason"] = "decision_sufficient"
                structure["final_depth"] = depth
                continue

            refine_node_ids = judgement.get(
                "refine_node_ids",
                []
            )

            # --------------------------------------------
            # 没有可继续展开节点
            # --------------------------------------------

            if not refine_node_ids:
                structure["decision_sufficient"] = False
                structure["evidence_status"] = "insufficient"
                structure["matched_case_ids"] = matched_case_ids
                structure["judgement_reason"] = reason
                structure["stop_reason"] = "no_refinable_branch"
                structure["final_depth"] = depth
                continue

            if depth >= max_depth:
                structure["decision_sufficient"] = False
                structure["evidence_status"] = "insufficient"
                structure["matched_case_ids"] = matched_case_ids
                structure["judgement_reason"] = reason
                structure["stop_reason"] = "max_depth"
                structure["final_depth"] = depth
                continue

            refine_nodes = []

            for node_id in refine_node_ids:
                node = find_node_by_id(
                    structure,
                    node_id
                )

                if node is None:
                    continue

                if len(
                    node.get("case_ids", [])
                ) <= 1:
                    node["status"] = "terminal"
                    continue

                refine_nodes.append(
                    make_frontier_item(node)
                )

            if not refine_nodes:
                structure["decision_sufficient"] = False
                structure["evidence_status"] = "insufficient"
                structure["matched_case_ids"] = matched_case_ids
                structure["judgement_reason"] = reason
                structure["stop_reason"] = "no_refinable_branch"
                structure["final_depth"] = depth
                continue

            x["_refine_nodes"] = refine_nodes
            need_refinement.append(x)

        if not need_refinement:
            break

        # ====================================================
        # 2. Refinement Batch
        # ====================================================

        print(
            f"[Refinement Depth {depth}] "
            f"Refinement Batch: {len(need_refinement)}"
        )

        refinement_messages = []

        for x in need_refinement:
            memory = memories[x["dialog_id"]]

            refinement_input = build_refinement_input(
                memory,
                x["_structure"],
                x["_refine_nodes"]
            )

            refinement_messages.append(
                build_messages(
                    refinement_prompt,
                    refinement_input
                )
            )

        refinement_outputs = parallel_inference(
            refinement_messages
        )

        next_active_items = []

        for x, out in zip(
            need_refinement,
            refinement_outputs
        ):
            refinement_result = parse_llm_output(
                out
            )

            structure = x["_structure"]

            x["_refinement_history"].append({
                "depth": depth,
                "refine_nodes": copy.deepcopy(
                    x["_refine_nodes"]
                ),
                "output": copy.deepcopy(
                    refinement_result
                )
            })

            refinements = refinement_result.get(
                "refinements",
                []
            )

            if not refinements:
                structure["decision_sufficient"] = False
                structure["evidence_status"] = "insufficient"
                structure["stop_reason"] = "no_valid_refinement"
                structure["final_depth"] = depth
                continue

            structure, new_frontier, created_count = apply_refinements(
                structure,
                refinements
            )

            x["_structure"] = structure

            if created_count == 0 or not new_frontier:
                structure["decision_sufficient"] = False
                structure["evidence_status"] = "insufficient"
                structure["stop_reason"] = "no_new_child"
                structure["final_depth"] = depth + 1
                continue

            x["_frontier"] = new_frontier
            next_active_items.append(x)

        active_items = next_active_items

    # ========================================================
    # Finalize
    # ========================================================

    for x in items:
        if "_structure" not in x:
            continue

        structure = x["_structure"]

        if "evidence_status" not in structure:
            structure["decision_sufficient"] = False
            structure["evidence_status"] = "unresolved"
            structure["stop_reason"] = "unfinished"

        structure["judgement_history"] = x["_judgement_history"]
        structure["refinement_history"] = x["_refinement_history"]

        x["topic_structure"] = structure

        dialog_id = x["dialog_id"]

        memories[dialog_id] = update_topic_structure_memory(
            memories[dialog_id],
            structure
        )

        memories[dialog_id] = mark_pe_version(
            memories[dialog_id]
        )

        x.pop("_structure", None)
        x.pop("_frontier", None)
        x.pop("_refine_nodes", None)
        x.pop("_judgement_history", None)
        x.pop("_refinement_history", None)






# ====================================================
# 6. Difference-Guided Adaptive Refinement Batch
# ====================================================

print(
    f"[Turn {t_id}] "
    f"Topic Refinement Batch"
)

refinement_items = []

for x in current_batch:
    memory = memories[
        x["dialog_id"]
    ]

    if not x["structure_update"]:
        x["topic_structure"] = copy.deepcopy(
            memory["evidence_state"]["topic_structure"]
        )
        continue

    if not memory["evidence_state"]["knowledge_state"]:
        x["topic_structure"] = {}
        memory["evidence_state"]["topic_structure"] = {}
        memories[x["dialog_id"]] = mark_pe_version(
            memory
        )
        continue

    refinement_items.append(x)


if refinement_items:
    run_refinement_pe_batch(
        items=refinement_items,
        memories=memories,
        judgement_prompt=judgement_prompt,
        refinement_prompt=refinement_prompt,
        max_depth=max_refinement_depth
    )
