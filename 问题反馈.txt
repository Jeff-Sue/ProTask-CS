def trace_html(state):

    return render_trace(
        state["trace"],
        state.get("feedback", {}),
    )




def submit_positive(
    state,
):

    if state is None:

        return (
            state,

            render_feedback_result(
                False,
                "当前没有可反馈的回复。",
            ),

            gr.update(
                visible=False
            ),

            render_trace(
                [],
                {},
            ),
        )

    turn = state.get(
        "turns",
        0,
    )

    if turn <= 0:

        return (
            state,

            render_feedback_result(
                False,
                "当前没有可反馈的回复。",
            ),

            gr.update(
                visible=False
            ),

            trace_html(state),
        )

    state.setdefault(
        "feedback",
        {},
    )

    state["feedback"][turn] = {
        "rating": "positive",
        "reason": None,
        "reason_text": None,
        "comment": "",
        "created_at": now_iso(),
    }

    # 保存到后台日志
    save_session(
        state
    )

    return (
        state,

        render_feedback_result(
            True,
            "感谢您的反馈。",
        ),

        gr.update(
            visible=False
        ),

        # 立即刷新开发者运行轨迹
        trace_html(state),
    )








def submit_negative(
    reason_text,
    comment,
    state,
):

    if state is None:

        return (
            state,

            render_feedback_result(
                False,
                "当前没有可反馈的回复。",
            ),

            gr.update(
                visible=True
            ),

            reason_text,

            comment,

            render_trace(
                [],
                {},
            ),
        )

    turn = state.get(
        "turns",
        0,
    )

    if turn <= 0:

        return (
            state,

            render_feedback_result(
                False,
                "当前没有可反馈的回复。",
            ),

            gr.update(
                visible=True
            ),

            reason_text,

            comment,

            trace_html(state),
        )

    reason_text = str(
        reason_text or ""
    ).strip()

    if not reason_text:

        return (
            state,

            render_feedback_result(
                False,
                "请选择一个不满意原因。",
            ),

            gr.update(
                visible=True
            ),

            reason_text,

            comment,

            trace_html(state),
        )

    reason_code = FEEDBACK_REASONS.get(
        reason_text,
        "other",
    )

    comment = str(
        comment or ""
    ).strip()

    state.setdefault(
        "feedback",
        {},
    )

    state["feedback"][turn] = {
        "rating": "negative",
        "reason": reason_code,
        "reason_text": reason_text,
        "comment": comment,
        "created_at": now_iso(),
    }

    # 保存到后台日志
    save_session(
        state
    )

    return (
        state,

        render_feedback_result(
            True,
            "感谢您的反馈。",
        ),

        gr.update(
            visible=False
        ),

        None,

        "",

        # 立即刷新开发者运行轨迹
        trace_html(state),
    )

def render_trace(
    trace,
    feedback_map=None,
):

    feedback_map = (
        feedback_map
        or {}
    )

    if not trace:

        return """
        <div class="trace-surface">

            <div class="decision-empty">

                <div class="decision-empty-title">
                    暂无运行轨迹
                </div>

                <div class="decision-empty-desc">
                    每完成一轮交互，执行轨迹会显示在这里。
                </div>

            </div>

        </div>
        """

    items = []

    for index, item in enumerate(
        reversed(trace),
        start=1,
    ):

        turn = item.get(
            "turn",
            len(trace) - index + 1,
        )

        user = escape(
            item.get(
                "user",
                "",
            )
        )

        query = (
            item.get("query_state")
            or {}
        )

        trigger = (
            item.get("trigger_state")
            or {}
        )

        policy = (
            item.get("policy_state")
            or {}
        )

        latency = (
            item.get("latency")
            or {}
        )

        try:
            total = (
                float(
                    latency.get(
                        "total_ms",
                        0,
                    )
                )
                / 1000.0
            )

        except Exception:
            total = 0.0

        # ====================================================
        # Feedback
        # ====================================================

        feedback = (
            feedback_map.get(turn)
            or feedback_map.get(str(turn))
        )

        if not feedback:

            feedback_html = """
            <div class="trace-feedback trace-feedback-empty">
                <span class="trace-feedback-label">
                    Feedback
                </span>

                <span>
                    —
                </span>
            </div>
            """

        elif feedback.get(
            "rating"
        ) == "positive":

            feedback_html = """
            <div class="trace-feedback trace-feedback-positive">

                <span class="trace-feedback-label">
                    Feedback
                </span>

                <span>
                    👍 满意
                </span>

            </div>
            """

        else:

            reason = escape(
                feedback.get(
                    "reason_text",
                    "",
                )
                or "不满意"
            )

            comment = escape(
                feedback.get(
                    "comment",
                    "",
                )
            )

            comment_html = ""

            if comment:

                comment_html = f"""
                <div class="trace-feedback-comment">
                    {comment}
                </div>
                """

            feedback_html = f"""
            <div class="trace-feedback trace-feedback-negative">

                <div class="trace-feedback-main">

                    <span class="trace-feedback-label">
                        Feedback
                    </span>

                    <span>
                        👎 不满意 · {reason}
                    </span>

                </div>

                {comment_html}

            </div>
            """

        # ====================================================
        # Turn
        # ====================================================

        items.append(
            f"""
            <div class="trace-item">

                <div class="trace-head">

                    <b>
                        Turn {turn}
                    </b>

                    <span>
                        {total:.2f}s
                    </span>

                </div>

                <div class="trace-user">
                    {user}
                </div>

                <div class="trace-flow">

                    {badge(
                        query.get(
                            "query_mode",
                            "none",
                        ),
                        "blue"
                    )}

                    <span class="flow-arrow">
                        →
                    </span>

                    {badge(
                        trigger.get(
                            "evidence_mode",
                            "",
                        ),
                        "gray"
                    )}

                    <span class="flow-arrow">
                        →
                    </span>

                    {badge(
                        policy.get(
                            "label",
                            "",
                        ),
                        "green"
                    )}

                </div>

                {feedback_html}

            </div>
            """
        )

    return f"""
    <div class="trace-surface">
        {''.join(items)}
    </div>
    """
