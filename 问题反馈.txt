def render_cases(cases):

    if not cases:
        return """
        <div class="empty-small">
            当前没有 Active Cases
        </div>
        """

    items = []

    for index, case in enumerate(cases):

        if not isinstance(case, dict):
            continue

        case_id = (
            case.get("case_id")
            or case.get("id")
            or case.get("caseId")
            or f"Case {index + 1}"
        )

        # ----------------------------------------------------
        # 尝试寻找适合作为卡片摘要的字段
        # ----------------------------------------------------

        title = (
            case.get("title")
            or case.get("name")
            or case.get("question")
            or case.get("text")
            or case.get("content")
            or case.get("problem")
            or ""
        )

        title = str(title).strip()

        # 卡片外面只显示摘要
        preview = title

        if len(preview) > 100:
            preview = preview[:100] + "..."

        # ----------------------------------------------------
        # 完整内容
        # ----------------------------------------------------

        detail_rows = []

        for key, value in case.items():

            if value is None:
                continue

            if isinstance(value, (dict, list)):
                value_text = json_text(value)
            else:
                value_text = str(value)

            if not value_text.strip():
                continue

            detail_rows.append(
                f"""
                <div class="case-detail-row">

                    <div class="case-detail-key">
                        {escape(key)}
                    </div>

                    <div class="case-detail-value">
                        {escape(value_text)}
                    </div>

                </div>
                """
            )

        details_html = "".join(detail_rows)

        items.append(
            f"""
            <details class="case-card">

                <summary class="case-card-summary">

                    <div class="case-card-main">

                        <div class="case-card-id">
                            {escape(case_id)}
                        </div>

                        <div class="case-card-preview">
                            {escape(preview) if preview else "点击查看案例详情"}
                        </div>

                    </div>

                    <div class="case-card-action">
                        <span class="case-expand-text">
                            查看详情
                        </span>

                        <span class="case-chevron">
                            ›
                        </span>
                    </div>

                </summary>


                <div class="case-detail">

                    <div class="case-detail-header">
                        Case Detail
                    </div>

                    {details_html}

                </div>

            </details>
            """
        )

    if not items:
        return """
        <div class="empty-small">
            当前没有可显示的 Active Cases
        </div>
        """

    return "".join(items)
