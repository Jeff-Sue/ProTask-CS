import json
from pathlib import Path

from Agent.utils.scu_rag import batch_zefeng_rag
from Agent.utils.qwen_api_multi import single_inference, parallel_inference

# Utils

def parse_json(text):
    if isinstance(text, dict):
        return text
    if not isinstance(text, str):
        return {}

    text = text.strip()
    if not text:
        return {}

    try:
        return json.loads(text)
    except Exception:
        pass

    if "```json" in text:
        try:
            text = text.split("```json", 1)[1].split("```", 1)[0].strip()
            return json.loads(text)
        except Exception:
            return {}

    if "```" in text:
        try:
            text = text.split("```", 1)[1].split("```", 1)[0].strip()
            return json.loads(text)
        except Exception:
            return {}

    return {}


def parse_llm_output(output):
    parsed = parse_json(output)
    if not parsed:
        return {}
    if "response" not in parsed:
        return parsed
    return parse_json(parsed.get("response", ""))


def prompt_read(path):
    path = Path(path)
    if not path.exists():
        raise FileNotFoundError(f"Prompt file not found: {path}")
    return path.read_text(encoding="utf-8").strip()


def build_messages(system_prompt, data):
    return [
        {"role": "system", "content": system_prompt},
        {"role": "user", "content": json.dumps(data, ensure_ascii=False)},
    ]


def history_to_text(history):
    lines = []
    for msg in history:
        if msg.get("role") == "user":
            lines.append(f"用户: {msg.get('content', '')}")
        elif msg.get("role") == "assistant":
            lines.append(f"客服: {msg.get('content', '')}")
    return "\n".join(lines)


def get_case_id(case):
    if not isinstance(case, dict):
        return ""
    return str(
        case.get("case_id")
        or case.get("id")
        or case.get("caseId")
        or ""
    ).strip()


def case_map_from_cases(cases):
    return {
        get_case_id(case): case
        for case in cases or []
        if get_case_id(case)
    }


def normalize_case_ids(value):
    if value is None:
        return []

    if isinstance(value, str):
        value = [value]

    if not isinstance(value, list):
        return []

    return list(dict.fromkeys(
        str(x).strip()
        for x in value
        if str(x).strip()
    ))

def init_memory():
    return {
        "issue_state": {
            "current_issue": "",
            "query_history": [],
        },
        "dialog_state": {
            "compressed_history": [],
        },
        "evidence_state": {
            "active_cases": [],
            "topic_structure": {},
        },
        "interaction_state": {
            "stage": "Start",
            "last_action": "",
            "last_case_ids": [],
            "action_history": [],
        },
    }


# ============================================================
# Topic Structure
# ============================================================

def build_topic_split_input(query, cases, parent_topic, depth):
    return {
        "query": query,
        "parent_topic": parent_topic,
        "depth": depth,
        "cases": cases,
    }


def make_leaf(case_id, topic=None):
    return {
        "topic": topic or case_id,
        "case_ids": [case_id],
        "leaf": True,
        "case_id": case_id,
        "children": [],
    }


def normalize_split(split_result, expected_case_ids):
    if not isinstance(split_result, dict):
        return None

    dimension = str(split_result.get("dimension", "")).strip()
    children = split_result.get("children", [])

    if not dimension or not isinstance(children, list):
        return None

    expected_set = set(expected_case_ids)
    normalized_children = []
    observed = []

    for child in children:
        if not isinstance(child, dict):
            continue

        topic = str(child.get("topic", "")).strip()
        case_ids = child.get("case_ids", [])

        if not topic or not isinstance(case_ids, list):
            continue

        clean_ids = []
        for case_id in case_ids:
            case_id = str(case_id).strip()
            if case_id and case_id in expected_set:
                clean_ids.append(case_id)

        clean_ids = list(dict.fromkeys(clean_ids))

        if not clean_ids:
            continue

        normalized_children.append({
            "topic": topic,
            "case_ids": clean_ids,
        })
        observed.extend(clean_ids)

    if len(normalized_children) < 2:
        return None

    if sorted(observed) != sorted(expected_case_ids):
        return None

    if len(observed) != len(set(observed)):
        return None

    return {
        "dimension": dimension,
        "children": normalized_children,
    }


def fallback_to_case_leaves(node, case_ids):
    node["leaf"] = False
    node["dimension"] = "案例差异"
    node["children"] = [
        make_leaf(case_id)
        for case_id in case_ids
    ]


def format_topic_tree(tree):
    if not tree:
        return "None"

    lines = []

    def node_text(node):
        topic = str(node.get("topic", "")).strip()
        dimension = str(node.get("dimension", "")).strip()
        case_id = str(node.get("case_id", "")).strip()

        if node.get("leaf") and case_id:
            if topic and topic != case_id:
                return f"{topic}  [{case_id}]"
            return f"[{case_id}]"

        if dimension:
            return f"{topic}  <{dimension}>"

        return topic or case_id or "Unknown"

    def walk(node, prefix="", is_last=True, root=False):
        if root:
            lines.append(node_text(node))
        else:
            lines.append(
                prefix
                + ("└─ " if is_last else "├─ ")
                + node_text(node)
            )

        children = node.get("children", [])

        if not children:
            return

        child_prefix = (
            prefix
            if root
            else prefix + ("   " if is_last else "│  ")
        )

        for i, child in enumerate(children):
            walk(
                child,
                prefix=child_prefix,
                is_last=(i == len(children) - 1),
                root=False,
            )

    walk(tree, root=True)
    return "\n".join(lines)

# Memory Agent

class MemoryAgent:
    """
    Stateful customer-service agent.

    One instance corresponds to one dialogue.
    Call reset() before starting a new dialogue if reusing it.
    """

    def __init__(
        self,
        prompt_dir="Agent/promptsv2_2",
        mode="single",
        max_topic_depth=8,
        max_topic_retry=2,
        show_state=True,
    ):
        prompt_dir = Path(prompt_dir)

        self.query_prompt = prompt_read(prompt_dir / "query.prompt")
        self.trigger_prompt = prompt_read(prompt_dir / "trigger.prompt")
        self.topic_prompt = prompt_read(prompt_dir / "topic.prompt")
        self.policy_prompt = prompt_read(prompt_dir / "policy.prompt")
        self.response_prompt = prompt_read(prompt_dir / "response.prompt")

        if mode not in {"single", "parallel"}:
            raise ValueError(f"Unsupported mode: {mode}")

        self.mode = mode
        self.max_topic_depth = max_topic_depth
        self.max_topic_retry = max_topic_retry
        self.show_state = show_state
        self.memory = init_memory()

    # ========================================================
    # LLM
    # ========================================================

    def call_llm(self, prompt, data):
        messages = build_messages(prompt, data)

        if self.mode == "single":
            output = single_inference(messages)
        else:
            output = parallel_inference([messages])[0]

        return parse_llm_output(output)

    def call_llm_batch(self, messages):
        if not messages:
            return []

        if self.mode == "single":
            outputs = [
                single_inference(message)
                for message in messages
            ]
        else:
            outputs = parallel_inference(messages)

        return [
            parse_llm_output(output)
            for output in outputs
        ]

    # ========================================================
    # Memory Update
    # ========================================================

    def update_issue(self, query_state):
        mode = query_state.get("query_mode", "none")
        query = str(query_state.get("query_text", "")).strip()

        if mode in {"rewrite", "refine"} and query:
            self.memory["issue_state"]["current_issue"] = query
            self.memory["issue_state"]["query_history"].append(query)

    def update_evidence(
        self,
        trigger_state,
        evidence,
        topic_structure=None,
    ):
        if trigger_state.get("evidence_mode") == "retrieve_new_evidence":
            self.memory["evidence_state"]["active_cases"] = evidence
            self.memory["evidence_state"]["topic_structure"] = (
                topic_structure or {}
            )

    def update_interaction(self, policy_state):
        label = str(policy_state.get("label", "")).strip()
        case_ids = normalize_case_ids(
            policy_state.get("case_id")
            or policy_state.get("selected_case")
        )

        interaction = self.memory["interaction_state"]

        interaction["last_action"] = label
        interaction["last_case_ids"] = case_ids
        interaction["action_history"].append({
            "label": label,
            "case_ids": case_ids,
        })

        if label in {"AskMissingSlot", "AskClarification"}:
            interaction["stage"] = "Clarification"
        elif label in {"ExplainedResponse", "CaseRecommendation"}:
            interaction["stage"] = "Resolution"
        elif label == "Handoff":
            interaction["stage"] = "End"

    def update_dialogue(self, user, response):
        history = self.memory["dialog_state"]["compressed_history"]

        history.append({
            "user": user,
            "assistant": response,
        })

        self.memory["dialog_state"]["compressed_history"] = history[-5:]

    # ========================================================
    # Topic Structure Generation
    # ========================================================

    def generate_topic_structure(self, query, cases):
        case_map = case_map_from_cases(cases)
        case_ids = list(case_map.keys())

        root = {
            "query": query,
            "topic": query,
            "case_ids": case_ids,
            "leaf": len(case_ids) <= 1,
            "children": [],
        }

        if not case_ids:
            return root

        if len(case_ids) == 1:
            root["case_id"] = case_ids[0]
            return root

        frontier = [{
            "node": root,
            "case_ids": case_ids,
            "depth": 0,
        }]

        round_id = 0

        while frontier:
            print(
                f"[TopicStructure] Round {round_id}: "
                f"{len(frontier)} nodes"
            )

            active_frontier = []

            for task in frontier:
                if task["depth"] >= self.max_topic_depth:
                    print(
                        "[TopicStructure Warning] "
                        f"max_depth reached: {task['case_ids']}"
                    )
                    fallback_to_case_leaves(
                        task["node"],
                        task["case_ids"],
                    )
                else:
                    active_frontier.append(task)

            if not active_frontier:
                break

            # -----------------------------------------------
            # First generation
            # -----------------------------------------------

            messages = []

            for task in active_frontier:
                node_cases = [
                    case_map[case_id]
                    for case_id in task["case_ids"]
                ]

                split_input = build_topic_split_input(
                    query=query,
                    cases=node_cases,
                    parent_topic=task["node"]["topic"],
                    depth=task["depth"],
                )

                messages.append(
                    build_messages(
                        self.topic_prompt,
                        split_input,
                    )
                )

            outputs = self.call_llm_batch(messages)

            valid_results = {}
            retry_tasks = []

            for idx, (task, output) in enumerate(
                zip(active_frontier, outputs)
            ):
                valid_split = normalize_split(
                    output,
                    task["case_ids"],
                )

                if valid_split:
                    valid_results[idx] = valid_split
                else:
                    retry_tasks.append({
                        "frontier_idx": idx,
                        "task": task,
                        "attempt": 1,
                    })

            # -----------------------------------------------
            # Retry
            # -----------------------------------------------

            while retry_tasks:
                attempt = retry_tasks[0]["attempt"]

                if attempt > self.max_topic_retry:
                    break

                print(
                    f"[TopicStructure] Retry {attempt}: "
                    f"{len(retry_tasks)} nodes"
                )

                retry_messages = []

                for retry_item in retry_tasks:
                    task = retry_item["task"]

                    node_cases = [
                        case_map[case_id]
                        for case_id in task["case_ids"]
                    ]

                    split_input = build_topic_split_input(
                        query=query,
                        cases=node_cases,
                        parent_topic=task["node"]["topic"],
                        depth=task["depth"],
                    )

                    split_input["retry_instruction"] = (
                        "上一次划分结果不合法。请重新划分，并严格保证："
                        "每个输入 case_id 必须且只能出现一次；"
                        "至少划分为两个有意义的子主题；"
                        "不得遗漏或重复任何 case_id。"
                    )

                    retry_messages.append(
                        build_messages(
                            self.topic_prompt,
                            split_input,
                        )
                    )

                retry_outputs = self.call_llm_batch(
                    retry_messages
                )

                next_retry_tasks = []

                for retry_item, output in zip(
                    retry_tasks,
                    retry_outputs,
                ):
                    task = retry_item["task"]

                    valid_split = normalize_split(
                        output,
                        task["case_ids"],
                    )

                    if valid_split:
                        valid_results[
                            retry_item["frontier_idx"]
                        ] = valid_split
                    else:
                        next_attempt = (
                            retry_item["attempt"] + 1
                        )

                        if next_attempt <= self.max_topic_retry:
                            next_retry_tasks.append({
                                "frontier_idx": (
                                    retry_item["frontier_idx"]
                                ),
                                "task": task,
                                "attempt": next_attempt,
                            })

                retry_tasks = next_retry_tasks

            # -----------------------------------------------
            # Build next level
            # -----------------------------------------------

            next_frontier = []

            for idx, task in enumerate(active_frontier):
                node = task["node"]
                valid_split = valid_results.get(idx)

                if not valid_split:
                    print(
                        "[TopicStructure Warning] "
                        "failed split after retries: "
                        f"{task['case_ids']}"
                    )

                    fallback_to_case_leaves(
                        node,
                        task["case_ids"],
                    )
                    continue

                node["leaf"] = False
                node["dimension"] = valid_split["dimension"]
                node["children"] = []

                for child_info in valid_split["children"]:
                    child_case_ids = child_info["case_ids"]

                    child_node = {
                        "topic": child_info["topic"],
                        "case_ids": child_case_ids,
                        "leaf": len(child_case_ids) == 1,
                        "children": [],
                    }

                    node["children"].append(child_node)

                    if len(child_case_ids) == 1:
                        child_node["case_id"] = (
                            child_case_ids[0]
                        )
                        continue

                    # 只继续细化当前 child 自己的 cases
                    next_frontier.append({
                        "node": child_node,
                        "case_ids": child_case_ids,
                        "depth": task["depth"] + 1,
                    })

            frontier = next_frontier
            round_id += 1

        return root

    # ========================================================
    # State Display
    # ========================================================

    def print_state(
        self,
        query_state,
        trigger_state,
        policy_state,
    ):
        if not self.show_state:
            return

        query_mode = query_state.get("query_mode", "")
        query_text = query_state.get("query_text", "")

        trigger = trigger_state.get("trigger", "")
        evidence_mode = trigger_state.get(
            "evidence_mode",
            "",
        )

        evidence = self.memory[
            "evidence_state"
        ]["active_cases"]

        case_ids = [
            get_case_id(case)
            for case in evidence
            if get_case_id(case)
        ]

        policy = policy_state.get("label", "")

        selected_case = normalize_case_ids(
            policy_state.get("case_id")
            or policy_state.get("selected_case")
        )

        topic_tree = format_topic_tree(
            self.memory[
                "evidence_state"
            ]["topic_structure"]
        )

        print(
            "\n╔══════════════ AGENT STATE ══════════════╗"
        )
        print(
            f" Query    : {query_mode} | {query_text}"
        )
        print(f" Trigger  : {trigger}")
        print(f" Evidence : {evidence_mode}")
        print(
            f" Cases    : "
            f"{', '.join(case_ids) if case_ids else 'None'}"
        )

        print("\n Topic Tree")
        for line in topic_tree.splitlines():
            print(f" {line}")

        print(f"\n Policy   : {policy}")
        print(
            f" Case     : "
            f"{selected_case if selected_case else 'None'}"
        )
        print(
            "╚══════════════════════════════════════════╝"
        )

    # ========================================================
    # Reply
    # ========================================================

    def reply(self, dialogue_history):
        """
        dialogue_history:
        [
            {"role": "user", "content": "..."},
            {"role": "assistant", "content": "..."},
            ...
            {"role": "user", "content": "..."},
        ]

        Returns:
            str: customer-service response
        """

        if not dialogue_history:
            raise ValueError("dialogue_history is empty")

        last_msg = dialogue_history[-1]

        if last_msg.get("role") != "user":
            raise ValueError(
                "The last message in dialogue_history "
                "must be from user."
            )

        user = str(
            last_msg.get("content", "")
        ).strip()

        if not user:
            raise ValueError(
                "Current user message is empty"
            )

        # ----------------------------------------------------
        # 1. Query
        # ----------------------------------------------------

        query_state = self.call_llm(
            self.query_prompt,
            {
                "issue_state": self.memory["issue_state"],
                "dialog_state": self.memory["dialog_state"],
                "current": {
                    "u": user
                },
            },
        )

        self.update_issue(query_state)

        # ----------------------------------------------------
        # 2. Trigger
        # ----------------------------------------------------

        trigger_state = self.call_llm(
            self.trigger_prompt,
            {
                "issue_state": self.memory["issue_state"],
                "evidence_state": {
                    "active_cases": self.memory[
                        "evidence_state"
                    ]["active_cases"]
                },
                "dialog_state": self.memory["dialog_state"],
            },
        )

        # ----------------------------------------------------
        # 3. RAG + Topic
        # ----------------------------------------------------

        if (
            trigger_state.get("evidence_mode")
            == "retrieve_new_evidence"
        ):
            query = (
                self.memory[
                    "issue_state"
                ]["current_issue"].strip()
                or user
            )

            evidence = batch_zefeng_rag(
                [history_to_text(dialogue_history)],
                [query],
            )[0]

            topic_structure = (
                self.generate_topic_structure(
                    query=query,
                    cases=evidence,
                )
            )

        else:
            evidence = self.memory[
                "evidence_state"
            ]["active_cases"]

            topic_structure = self.memory[
                "evidence_state"
            ]["topic_structure"]

        self.update_evidence(
            trigger_state,
            evidence,
            topic_structure,
        )

        # ----------------------------------------------------
        # 4. Policy
        # ----------------------------------------------------

        policy_state = self.call_llm(
            self.policy_prompt,
            {
                "issue_state": self.memory["issue_state"],
                "evidence_state": {
                    "active_cases": self.memory[
                        "evidence_state"
                    ]["active_cases"],
                    "topic_structure": self.memory[
                        "evidence_state"
                    ]["topic_structure"],
                },
                "interaction_state": self.memory[
                    "interaction_state"
                ],
                "dialog_state": self.memory[
                    "dialog_state"
                ],
                "current": {
                    "u": user
                },
            },
        )

        self.update_interaction(policy_state)

        # ----------------------------------------------------
        # 5. Response
        # ----------------------------------------------------

        response_state = self.call_llm(
            self.response_prompt,
            {
                "issue_state": self.memory["issue_state"],
                "evidence_state": {
                    "active_cases": self.memory[
                        "evidence_state"
                    ]["active_cases"]
                },
                "interaction_state": self.memory[
                    "interaction_state"
                ],
                "dialog_state": self.memory[
                    "dialog_state"
                ],
                "current": {
                    "u": user,
                    "policy_label": self.memory[
                        "interaction_state"
                    ]["last_action"],
                    "case_ids": self.memory[
                        "interaction_state"
                    ]["last_case_ids"],
                },
            },
        )

        response = (
            response_state.get("response")
            or response_state.get("text")
            or response_state.get("content")
            or ""
        )

        response = str(response).strip()

        self.update_dialogue(
            user,
            response,
        )

        # ----------------------------------------------------
        # Display
        # ----------------------------------------------------

        self.print_state(
            query_state=query_state,
            trigger_state=trigger_state,
            policy_state=policy_state,
        )

        return response

    def reset(self):
        self.memory = init_memory()
