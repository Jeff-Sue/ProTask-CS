def generate_topic_structures_batch(
    items: List[Dict],
    topic_prompt: str,
    max_depth: int = 8,
    max_retry: int = 2,
):
    """
    真正的 batch 层级 Topic Structure 生成。

    items:
    [
        {
            "query": "...",
            "cases": [...]
        },
        ...
    ]

    核心逻辑：
    1. 每个 query 作为一棵树的 root
    2. 当前所有待划分节点组成 frontier
    3. frontier 中所有节点一次 batch inference
    4. 每个节点完成一层划分
    5. 子节点中 case 数 > 1 的进入 next_frontier
    6. 重复直到所有叶节点只有一个 case

    注意：
    - 不改变单节点的输入格式
    - 不改变划分逻辑
    - 只是把多个独立节点并行执行
    """

    # ============================================================
    # 1. 初始化 Trees / Case Maps / Frontier
    # ============================================================

    trees = []
    case_maps = []

    frontier = []

    for tree_idx, item in enumerate(items):

        query = item.get("query", "")
        cases = item.get("cases", [])

        case_map = case_map_from_cases(cases)
        case_maps.append(case_map)

        case_ids = list(case_map.keys())

        root = {
            "query": query,
            "topic": query,
            "case_ids": case_ids,
            "leaf": len(case_ids) <= 1,
            "children": []
        }

        # 空 evidence
        if len(case_ids) == 0:
            trees.append(root)
            continue

        # 只有一个 case，直接 leaf
        if len(case_ids) == 1:
            root["case_id"] = case_ids[0]
            trees.append(root)
            continue

        trees.append(root)

        # 多个 cases，加入当前待划分节点
        frontier.append(
            {
                "tree_idx": tree_idx,
                "node": root,
                "case_ids": case_ids,
                "depth": 0
            }
        )

    # ============================================================
    # 2. Frontier-based Recursive Batch
    # ============================================================

    round_id = 0

    while frontier:

        print(
            f"[TopicStructure] Round {round_id}: "
            f"{len(frontier)} nodes"
        )

        # --------------------------------------------------------
        # 2.1 处理达到 max_depth 的节点
        # --------------------------------------------------------

        active_frontier = []

        for task in frontier:

            node = task["node"]
            case_ids = task["case_ids"]
            depth = task["depth"]

            if depth >= max_depth:

                print(
                    "[TopicStructure Warning] "
                    f"max_depth reached: {case_ids}"
                )

                fallback_to_case_leaves(
                    node,
                    case_ids
                )

                continue

            active_frontier.append(task)

        if not active_frontier:
            break

        # --------------------------------------------------------
        # 2.2 第一次 Batch Split
        # --------------------------------------------------------

        messages = []

        for task in active_frontier:

            tree_idx = task["tree_idx"]
            node = task["node"]
            case_ids = task["case_ids"]
            depth = task["depth"]

            query = items[
                tree_idx
            ].get(
                "query",
                ""
            )

            case_map = case_maps[
                tree_idx
            ]

            node_cases = [
                case_map[case_id]
                for case_id in case_ids
            ]

            split_input = (
                build_topic_split_input(
                    query=query,
                    cases=node_cases,
                    parent_topic=node["topic"],
                    depth=depth
                )
            )

            messages.append(
                build_messages(
                    topic_prompt,
                    split_input
                )
            )

        # 真正的一次 Batch
        outputs = parallel_inference(
            messages
        )

        # --------------------------------------------------------
        # 2.3 Parse + Validation
        # --------------------------------------------------------

        valid_results = {}

        retry_tasks = []

        for idx, (
            task,
            output
        ) in enumerate(
            zip(
                active_frontier,
                outputs
            )
        ):

            split_result = (
                parse_llm_output(
                    output
                )
            )

            valid_split = (
                normalize_split(
                    split_result,
                    task["case_ids"]
                )
            )

            if valid_split:

                valid_results[
                    idx
                ] = valid_split

            else:

                retry_tasks.append(
                    {
                        "frontier_idx": idx,
                        "task": task,
                        "attempt": 1
                    }
                )

        # ========================================================
        # 2.4 Invalid Nodes Batch Retry
        # ========================================================

        while (
            retry_tasks
            and
            retry_tasks[0]["attempt"]
            <= max_retry
        ):

            attempt = (
                retry_tasks[0][
                    "attempt"
                ]
            )

            print(
                f"[TopicStructure] Retry {attempt}: "
                f"{len(retry_tasks)} nodes"
            )

            retry_messages = []

            for retry_item in retry_tasks:

                task = retry_item[
                    "task"
                ]

                tree_idx = task[
                    "tree_idx"
                ]

                node = task[
                    "node"
                ]

                case_ids = task[
                    "case_ids"
                ]

                depth = task[
                    "depth"
                ]

                query = items[
                    tree_idx
                ].get(
                    "query",
                    ""
                )

                case_map = (
                    case_maps[
                        tree_idx
                    ]
                )

                node_cases = [
                    case_map[
                        case_id
                    ]
                    for case_id
                    in case_ids
                ]

                split_input = (
                    build_topic_split_input(
                        query=query,
                        cases=node_cases,
                        parent_topic=node[
                            "topic"
                        ],
                        depth=depth
                    )
                )

                split_input[
                    "retry_instruction"
                ] = (
                    "上一次划分结果不合法。"
                    "请重新划分，并严格保证："
                    "每个输入 case_id 必须且只能出现一次；"
                    "至少划分为两个有意义的子主题；"
                    "不得遗漏或重复任何 case_id。"
                )

                retry_messages.append(
                    build_messages(
                        topic_prompt,
                        split_input
                    )
                )

            retry_outputs = (
                parallel_inference(
                    retry_messages
                )
            )

            next_retry_tasks = []

            for retry_item, output in zip(
                retry_tasks,
                retry_outputs
            ):

                frontier_idx = (
                    retry_item[
                        "frontier_idx"
                    ]
                )

                task = (
                    retry_item[
                        "task"
                    ]
                )

                split_result = (
                    parse_llm_output(
                        output
                    )
                )

                valid_split = (
                    normalize_split(
                        split_result,
                        task[
                            "case_ids"
                        ]
                    )
                )

                if valid_split:

                    valid_results[
                        frontier_idx
                    ] = valid_split

                else:

                    next_attempt = (
                        retry_item[
                            "attempt"
                        ]
                        + 1
                    )

                    if (
                        next_attempt
                        <=
                        max_retry
                    ):

                        next_retry_tasks.append(
                            {
                                "frontier_idx":
                                    frontier_idx,

                                "task":
                                    task,

                                "attempt":
                                    next_attempt
                            }
                        )

            retry_tasks = (
                next_retry_tasks
            )

        # ========================================================
        # 2.5 Apply Results + Build Next Frontier
        # ========================================================

        next_frontier = []

        for idx, task in enumerate(
            active_frontier
        ):

            node = task["node"]
            case_ids = task[
                "case_ids"
            ]
            depth = task[
                "depth"
            ]
            tree_idx = task[
                "tree_idx"
            ]

            valid_split = (
                valid_results.get(
                    idx
                )
            )

            # ----------------------------------------------------
            # 最终仍失败：fallback
            # ----------------------------------------------------

            if not valid_split:

                print(
                    "[TopicStructure Warning] "
                    "failed split after retries: "
                    f"{case_ids}"
                )

                fallback_to_case_leaves(
                    node,
                    case_ids
                )

                continue

            # ----------------------------------------------------
            # 正常写入当前节点
            # ----------------------------------------------------

            node["leaf"] = False

            node["dimension"] = (
                valid_split[
                    "dimension"
                ]
            )

            node["children"] = []

            for child_info in (
                valid_split[
                    "children"
                ]
            ):

                child_case_ids = (
                    child_info[
                        "case_ids"
                    ]
                )

                child_node = {
                    "topic":
                        child_info[
                            "topic"
                        ],

                    "case_ids":
                        child_case_ids,

                    "leaf":
                        len(
                            child_case_ids
                        ) == 1,

                    "children": []
                }

                node[
                    "children"
                ].append(
                    child_node
                )

                # ------------------------------------------------
                # Case 数 = 1：叶节点
                # ------------------------------------------------

                if (
                    len(
                        child_case_ids
                    )
                    == 1
                ):

                    child_node[
                        "case_id"
                    ] = (
                        child_case_ids[
                            0
                        ]
                    )

                    continue

                # ------------------------------------------------
                # Case 数 > 1：进入下一轮 frontier
                # ------------------------------------------------

                next_frontier.append(
                    {
                        "tree_idx":
                            tree_idx,

                        "node":
                            child_node,

                        "case_ids":
                            child_case_ids,

                        "depth":
                            depth + 1
                    }
                )

        frontier = (
            next_frontier
        )

        round_id += 1

    return trees


def fallback_to_case_leaves(
    node: Dict,
    case_ids: List[str]
):
    """
    当某个节点多次无法获得合法 partition，
    或达到 max_depth 时，避免程序卡死。

    直接将剩余 cases 展开成独立叶节点。
    """

    node["leaf"] = False

    node["dimension"] = (
        "案例差异"
    )

    node["children"] = [
        make_leaf(
            case_id
        )
        for case_id
        in case_ids
    ]
