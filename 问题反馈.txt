import json
from pathlib import Path

from Agent.utils.scu_rag import batch_zefeng_rag
from Agent.utils.qwen_api_multi import single_inference, parallel_inference

def parse_json(text):
    if isinstance(text, dict):
        return text
    if not isinstance(text, str):
        return {}

    text = text.strip()
    if not text:
        return {}

    try:
        return json.loads(text)
    except Exception:
        pass

    if "```json" in text:
        try:
            text = text.split("```json", 1)[1].split("```", 1)[0].strip()
            return json.loads(text)
        except Exception:
            return {}

    if "```" in text:
        try:
            text = text.split("```", 1)[1].split("```", 1)[0].strip()
            return json.loads(text)
        except Exception:
            return {}

    return {}


def parse_llm_output(output):
    parsed = parse_json(output)
    if not parsed:
        return {}
    if "response" not in parsed:
        return parsed
    return parse_json(parsed.get("response", ""))


def prompt_read(path):
    path = Path(path)
    if not path.exists():
        raise FileNotFoundError(f"Prompt file not found: {path}")
    return path.read_text(encoding="utf-8").strip()


def build_messages(system_prompt, data):
    return [
        {"role": "system", "content": system_prompt},
        {"role": "user", "content": json.dumps(data, ensure_ascii=False)},
    ]


def history_to_text(history):
    lines = []
    for msg in history:
        if msg.get("role") == "user":
            lines.append(f"用户: {msg.get('content', '')}")
        elif msg.get("role") == "assistant":
            lines.append(f"客服: {msg.get('content', '')}")
    return "\n".join(lines)


def get_case_id(case):
    if not isinstance(case, dict):
        return ""
    return str(case.get("case_id") or case.get("id") or case.get("caseId") or "").strip()


def case_map_from_cases(cases):
    return {get_case_id(case): case for case in cases or [] if get_case_id(case)}


def get_policy_case_ids(policy_state):
    information = policy_state.get("information")
    if not isinstance(information, dict):
        return []

    value = information.get("case_id")
    if value is None:
        value = information.get("case_ids")

    if value is None:
        return []
    if isinstance(value, str):
        value = [value]
    if not isinstance(value, list):
        return []

    return list(dict.fromkeys(str(x).strip() for x in value if str(x).strip()))


def get_selected_evidence(policy_state, active_cases):
    if policy_state.get("label") not in {"CaseRecommendation", "ExplainedResponse"}:
        return None

    case_ids = get_policy_case_ids(policy_state)
    if not case_ids:
        return None

    case_map = case_map_from_cases(active_cases)
    selected = [case_map[cid] for cid in case_ids if cid in case_map]

    if not selected:
        return None
    if len(selected) == 1:
        return selected[0]
    return selected

def init_memory():
    return {
        "issue_state": {
            "current_issue": "",
            "query_history": [],
        },
        "evidence_state": {
            "active_cases": [],
            "topic_structure": {},
        },
        "interaction_state": {
            "stage": "Start",
            "last_policy": None,
            "policy_history": [],
        },
        "dialogue_state": {
            "recent_history": [],
        },
    }

# Topic Structure

def build_topic_split_input(query, cases, parent_path, depth):
    return {
        "query": query,
        "parent_path": parent_path,
        "depth": depth,
        "cases": cases,
    }


def make_leaf(case_id, topic=None):
    return {
        "topic": topic or case_id,
        "case_ids": [case_id],
        "leaf": True,
        "case_id": case_id,
        "children": [],
    }


def normalize_split(split_result, expected_case_ids):
    if not isinstance(split_result, dict):
        return None, "output is not a JSON object"

    dimension = str(split_result.get("dimension", "")).strip()
    children = split_result.get("children")

    if not dimension:
        return None, "dimension is empty"
    if not isinstance(children, list):
        return None, "children is not a list"

    expected_set = set(expected_case_ids)
    normalized_children = []
    observed = []
    unknown = []

    for child in children:
        if not isinstance(child, dict):
            continue

        topic = str(child.get("topic", "")).strip()
        case_ids = child.get("case_ids")

        if not topic or not isinstance(case_ids, list):
            continue

        clean_ids = []
        for case_id in case_ids:
            case_id = str(case_id).strip()
            if not case_id:
                continue
            if case_id not in expected_set:
                unknown.append(case_id)
                continue
            clean_ids.append(case_id)

        clean_ids = list(dict.fromkeys(clean_ids))
        if not clean_ids:
            continue

        normalized_children.append({
            "topic": topic,
            "case_ids": clean_ids,
        })
        observed.extend(clean_ids)

    if unknown:
        return None, f"unknown case_ids: {unknown}"

    if len(normalized_children) < 2:
        return None, "fewer than two valid children"

    duplicated = list(dict.fromkeys(cid for cid in observed if observed.count(cid) > 1))
    if duplicated:
        return None, f"duplicated case_ids: {duplicated}"

    missing = [cid for cid in expected_case_ids if cid not in observed]
    if missing:
        return None, f"missing case_ids: {missing}"

    return {
        "dimension": dimension,
        "children": normalized_children,
    }, ""


def fallback_to_case_leaves(node, case_ids):
    node["leaf"] = False
    node["dimension"] = "案例差异"
    node["children"] = [make_leaf(case_id) for case_id in case_ids]


def format_topic_tree(tree):
    if not tree:
        return "None"

    lines = []

    def node_text(node):
        topic = str(node.get("topic", "")).strip()
        dimension = str(node.get("dimension", "")).strip()
        case_id = str(node.get("case_id", "")).strip()

        if node.get("leaf") and case_id:
            return f"{topic}  [{case_id}]" if topic and topic != case_id else f"[{case_id}]"
        if dimension:
            return f"{topic}  <{dimension}>"
        return topic or case_id or "Unknown"

    def walk(node, prefix="", is_last=True, root=False):
        if root:
            lines.append(node_text(node))
        else:
            lines.append(prefix + ("└─ " if is_last else "├─ ") + node_text(node))

        children = node.get("children", [])
        if not children:
            return

        child_prefix = prefix if root else prefix + ("   " if is_last else "│  ")

        for i, child in enumerate(children):
            walk(child, child_prefix, i == len(children) - 1, False)

    walk(tree, root=True)
    return "\n".join(lines)

# Memory Agent

class MemoryAgent:
    """
    Stateful customer-service agent.

    One instance corresponds to one dialogue.
    Call reset() before starting a new dialogue if reusing it.
    """

    def __init__(
        self,
        prompt_dir="Agent/promptsv2_2",
        mode="single",
        max_topic_depth=8,
        max_topic_retry=2,
        show_state=True,
    ):
        prompt_dir = Path(prompt_dir)

        self.query_prompt = prompt_read(prompt_dir / "query.prompt")
        self.trigger_prompt = prompt_read(prompt_dir / "trigger.prompt")
        self.topic_prompt = prompt_read(prompt_dir / "topic.prompt")
        self.policy_prompt = prompt_read(prompt_dir / "policy.prompt")
        self.response_prompt = prompt_read(prompt_dir / "response.prompt")

        if mode not in {"single", "parallel"}:
            raise ValueError(f"Unsupported mode: {mode}")

        self.mode = mode
        self.max_topic_depth = max_topic_depth
        self.max_topic_retry = max_topic_retry
        self.show_state = show_state
        self.memory = init_memory()

    # ========================================================
    # LLM
    # ========================================================

    def call_llm(self, prompt, data):
        messages = build_messages(prompt, data)

        if self.mode == "single":
            output = single_inference(messages)
        else:
            output = parallel_inference([messages])[0]

        return parse_llm_output(output)

    def call_llm_batch(self, messages):
        if not messages:
            return []

        if self.mode == "single":
            outputs = [single_inference(message) for message in messages]
        else:
            outputs = parallel_inference(messages)

        return [parse_llm_output(output) for output in outputs]

    # ========================================================
    # Memory Update
    # ========================================================

    def update_issue(self, query_state):
        mode = str(query_state.get("query_mode", "none")).strip()
        query = str(query_state.get("query_text", "")).strip()

        if mode in {"rewrite", "refine"} and query:
            self.memory["issue_state"]["current_issue"] = query
            self.memory["issue_state"]["query_history"].append(query)

    def update_evidence(self, trigger_state, evidence, topic_structure=None):
        if trigger_state.get("evidence_mode") != "retrieve_new_evidence":
            return

        self.memory["evidence_state"]["active_cases"] = evidence
        self.memory["evidence_state"]["topic_structure"] = topic_structure or {}

    def update_interaction(self, policy_state):
        label = str(policy_state.get("label", "")).strip()
        information = policy_state.get("information")

        policy = {
            "label": label,
            "information": information,
        }

        interaction = self.memory["interaction_state"]
        interaction["last_policy"] = policy
        interaction["policy_history"].append(policy)

        if label in {"AskMissingSlot", "AskClarification"}:
            interaction["stage"] = "Clarification"
        elif label in {"ExplainedResponse", "CaseRecommendation"}:
            interaction["stage"] = "Resolution"
        elif label == "Handoff":
            interaction["stage"] = "End"

    def update_dialogue(self, user, response):
        history = self.memory["dialogue_state"]["recent_history"]

        history.append({
            "user": user,
            "assistant": response,
        })

        self.memory["dialogue_state"]["recent_history"] = history[-5:]

    # ========================================================
    # Topic Structure Generation
    # ========================================================

    def generate_topic_structure(self, query, cases):
        case_map = case_map_from_cases(cases)
        case_ids = list(case_map)

        root = {
            "query": query,
            "topic": query,
            "case_ids": case_ids,
            "leaf": len(case_ids) <= 1,
            "children": [],
        }

        if not case_ids:
            return root

        if len(case_ids) == 1:
            root["case_id"] = case_ids[0]
            return root

        frontier = [{
            "node": root,
            "case_ids": case_ids,
            "depth": 0,
            "parent_path": [],
        }]

        round_id = 0

        while frontier:
            print(f"[TopicStructure] Round {round_id}: {len(frontier)} nodes")

            active_frontier = []

            for task in frontier:
                if task["depth"] >= self.max_topic_depth:
                    print(f"[TopicStructure Warning] max_depth reached: {task['case_ids']}")
                    fallback_to_case_leaves(task["node"], task["case_ids"])
                else:
                    active_frontier.append(task)

            if not active_frontier:
                break

            messages = []

            for task in active_frontier:
                node_cases = [case_map[cid] for cid in task["case_ids"]]
                split_input = build_topic_split_input(
                    query=query,
                    cases=node_cases,
                    parent_path=task["parent_path"],
                    depth=task["depth"],
                )
                messages.append(build_messages(self.topic_prompt, split_input))

            outputs = self.call_llm_batch(messages)

            valid_results = {}
            retry_tasks = []

            for idx, (task, output) in enumerate(zip(active_frontier, outputs)):
                valid_split, error = normalize_split(output, task["case_ids"])

                if valid_split:
                    valid_results[idx] = valid_split
                else:
                    print(f"[TopicStructure Invalid] {error}")
                    print("output:", output, flush=True)
                    retry_tasks.append({
                        "idx": idx,
                        "task": task,
                        "error": error,
                    })

            for attempt in range(1, self.max_topic_retry + 1):
                if not retry_tasks:
                    break

                print(f"[TopicStructure] Retry {attempt}: {len(retry_tasks)} nodes")
                retry_messages = []

                for item in retry_tasks:
                    task = item["task"]
                    node_cases = [case_map[cid] for cid in task["case_ids"]]

                    split_input = build_topic_split_input(
                        query=query,
                        cases=node_cases,
                        parent_path=task["parent_path"],
                        depth=task["depth"],
                    )

                    split_input["retry_instruction"] = (
                        f"上一次输出未通过结构验证：{item['error']}。"
                        "请修正该问题并重新输出合法的当前层划分。"
                    )

                    retry_messages.append(build_messages(self.topic_prompt, split_input))

                retry_outputs = self.call_llm_batch(retry_messages)
                next_retry_tasks = []

                for item, output in zip(retry_tasks, retry_outputs):
                    valid_split, error = normalize_split(output, item["task"]["case_ids"])

                    if valid_split:
                        valid_results[item["idx"]] = valid_split
                    else:
                        print(f"[TopicStructure Invalid] {error}")
                        print("output:", output, flush=True)
                        next_retry_tasks.append({
                            "idx": item["idx"],
                            "task": item["task"],
                            "error": error,
                        })

                retry_tasks = next_retry_tasks

            next_frontier = []

            for idx, task in enumerate(active_frontier):
                node = task["node"]
                valid_split = valid_results.get(idx)

                if not valid_split:
                    print(f"[TopicStructure Warning] failed split after retries: {task['case_ids']}")
                    fallback_to_case_leaves(node, task["case_ids"])
                    continue

                node["leaf"] = False
                node["dimension"] = valid_split["dimension"]
                node["children"] = []

                for child_info in valid_split["children"]:
                    child_case_ids = child_info["case_ids"]
                    child_topic = child_info["topic"]

                    child_node = {
                        "topic": child_topic,
                        "case_ids": child_case_ids,
                        "leaf": len(child_case_ids) == 1,
                        "children": [],
                    }

                    node["children"].append(child_node)

                    if len(child_case_ids) == 1:
                        child_node["case_id"] = child_case_ids[0]
                        continue

                    next_frontier.append({
                        "node": child_node,
                        "case_ids": child_case_ids,
                        "depth": task["depth"] + 1,
                        "parent_path": task["parent_path"] + [child_topic],
                    })

            frontier = next_frontier
            round_id += 1

        return root

    # ========================================================
    # State Display
    # ========================================================

    def print_state(self, query_state, trigger_state, policy_state):
        if not self.show_state:
            return

        query_mode = query_state.get("query_mode", "")
        query_text = query_state.get("query_text", "")
        trigger = trigger_state.get("trigger", "")
        evidence_mode = trigger_state.get("evidence_mode", "")

        active_cases = self.memory["evidence_state"]["active_cases"]
        case_ids = [get_case_id(case) for case in active_cases if get_case_id(case)]

        policy = policy_state.get("label", "")
        information = policy_state.get("information")
        topic_tree = format_topic_tree(self.memory["evidence_state"]["topic_structure"])

        print("\n╔══════════════ AGENT STATE ══════════════╗")
        print(f" Query    : {query_mode} | {query_text}")
        print(f" Trigger  : {trigger}")
        print(f" Evidence : {evidence_mode}")
        print(f" Cases    : {', '.join(case_ids) if case_ids else 'None'}")

        print("\n Topic Tree")
        for line in topic_tree.splitlines():
            print(f" {line}")

        print(f"\n Policy   : {policy}")
        print(f" Info     : {information if information is not None else 'None'}")
        print("╚══════════════════════════════════════════╝")

    # ========================================================
    # Reply
    # ========================================================

    def reply(self, dialogue_history):
        """
        dialogue_history:
        [
            {"role": "user", "content": "..."},
            {"role": "assistant", "content": "..."},
            ...
            {"role": "user", "content": "..."},
        ]

        Returns:
            str: customer-service response
        """

        if not dialogue_history:
            raise ValueError("dialogue_history is empty")

        last_msg = dialogue_history[-1]

        if last_msg.get("role") != "user":
            raise ValueError("The last message in dialogue_history must be from user.")

        user = str(last_msg.get("content", "")).strip()

        if not user:
            raise ValueError("Current user message is empty")

        # ----------------------------------------------------
        # 1. Query
        # ----------------------------------------------------

        query_state = self.call_llm(
            self.query_prompt,
            {
                "issue_state": self.memory["issue_state"],
                "dialogue_state": self.memory["dialogue_state"],
                "current": {"u": user},
            },
        )

        self.update_issue(query_state)

        # ----------------------------------------------------
        # 2. Trigger
        # ----------------------------------------------------

        trigger_state = self.call_llm(
            self.trigger_prompt,
            {
                "query": {
                    "query_mode": query_state.get("query_mode", "none"),
                    "query_text": query_state.get("query_text", ""),
                },
                "active_cases": self.memory["evidence_state"]["active_cases"],
            },
        )

        # ----------------------------------------------------
        # 3. RAG + Topic
        # ----------------------------------------------------

        evidence_mode = trigger_state.get("evidence_mode", "")

        if evidence_mode == "retrieve_new_evidence":
            query = str(query_state.get("query_text", "")).strip()

            if not query:
                query = self.memory["issue_state"]["current_issue"].strip()

            if not query:
                raise ValueError("RAG query is empty")

            evidence = batch_zefeng_rag(
                [history_to_text(dialogue_history)],
                [query],
            )[0]

            topic_structure = self.generate_topic_structure(
                query=query,
                cases=evidence,
            )

        else:
            evidence = self.memory["evidence_state"]["active_cases"]
            topic_structure = self.memory["evidence_state"]["topic_structure"]

        self.update_evidence(trigger_state, evidence, topic_structure)

        # ----------------------------------------------------
        # 4. Policy
        # ----------------------------------------------------

        if evidence_mode == "no_evidence_needed":
            policy_evidence = {
                "active_cases": [],
                "topic_structure": {},
            }
        else:
            policy_evidence = {
                "active_cases": self.memory["evidence_state"]["active_cases"],
                "topic_structure": self.memory["evidence_state"]["topic_structure"],
            }

        policy_state = self.call_llm(
            self.policy_prompt,
            {
                "issue_state": self.memory["issue_state"],
                "evidence_state": policy_evidence,
                "interaction_state": self.memory["interaction_state"],
                "dialogue_state": self.memory["dialogue_state"],
                "current": {"u": user},
            },
        )

        self.update_interaction(policy_state)

        # ----------------------------------------------------
        # 5. Response
        # ----------------------------------------------------

        selected_evidence = get_selected_evidence(
            policy_state,
            self.memory["evidence_state"]["active_cases"],
        )

        response_dialogue = {
            "recent_history": self.memory["dialogue_state"]["recent_history"],
            "current": {"u": user},
        }

        response_policy = {
            "label": policy_state.get("label", ""),
            "information": policy_state.get("information"),
        }

        response_state = self.call_llm(
            self.response_prompt,
            {
                "dialogue_state": response_dialogue,
                "policy": response_policy,
                "evidence": selected_evidence,
            },
        )

        response = (
            response_state.get("response")
            or response_state.get("text")
            or response_state.get("content")
            or ""
        )

        response = str(response).strip()

        self.update_dialogue(user, response)

        # ----------------------------------------------------
        # Display
        # ----------------------------------------------------

        self.print_state(
            query_state=query_state,
            trigger_state=trigger_state,
            policy_state=policy_state,
        )

        return response

    def reset(self):
        self.memory = init_memory()
