# ============================================================
# Web / Interactive Agent
# ============================================================

class MemoryAgent:
    """
    Stateful wrapper for Web / Gradio interaction.

    Important:
    - 不修改原来的 batch pipeline
    - 不修改原来的 memory / query / trigger / RAG / topic / policy / response 逻辑
    - 每次 reply() 只执行一个用户 turn
    """

    def __init__(self):
        # --------------------------------------------------------
        # Prompts
        # --------------------------------------------------------
        self.query_prompt = prompt_read(
            Path("prompts/query.prompt")
        )

        self.trigger_prompt = prompt_read(
            Path("prompts/trigger.prompt")
        )

        self.topic_prompt = prompt_read(
            Path("prompts/topic.prompt")
        )

        # 与 run_memory_agent 保持一致
        self.policy_prompt = prompt_read(
            Path("prompts/policy_v2.prompt")
        )

        self.response_prompt = prompt_read(
            Path("prompts/response.prompt")
        )

        # --------------------------------------------------------
        # Runtime State
        # --------------------------------------------------------
        self.reset()


    # ========================================================
    # Reset
    # ========================================================

    def reset(self):
        self.memory = init_memory()

        # 与 batch 中 dialogue_history 的语义保持一致
        self.dialogue_history = ""

        # 网页显示用
        self.turns = []

        # 当前 turn
        self.turn_id = 0

        # 当前轮中间结果
        self.last_query = {}
        self.last_trigger = {}
        self.last_evidence = []
        self.last_topic_structure = {}
        self.last_policy = {}
        self.last_response = {}

        # 当前轮 latency
        self.last_latency = {}

        return self.memory


    # ========================================================
    # Load Existing Dialogue
    # ========================================================

    def load_history(self, history):
        """
        给网页端“加载已有对话”使用。

        history 支持：

        [
            {"user": "...", "assistant": "..."},
            ...
        ]

        注意：
        这里主要恢复 dialogue history。
        不重新推断历史 Query / Evidence / Policy 状态。
        """

        self.reset()

        if not history:
            return self.memory

        for item in history:
            user = str(item.get("user", "")).strip()
            assistant = str(item.get("assistant", "")).strip()

            if user:
                self.dialogue_history += (
                    f"用户: {user}\n"
                )

            if assistant:
                self.dialogue_history += (
                    f"客服: {assistant}\n"
                )

            if user or assistant:
                self.memory = update_dialogue_memory(
                    self.memory,
                    user,
                    assistant
                )

                self.turns.append({
                    "turn_id": self.turn_id,
                    "user": user,
                    "response": assistant
                })

                self.turn_id += 1

        return self.memory


    # ========================================================
    # One Interactive Turn
    # ========================================================

    def reply(self, user: str):
        """
        执行一个完整用户 turn：

        Query
          ↓
        Trigger
          ↓
        RAG
          ↓
        Topic Structure
          ↓
        Policy
          ↓
        Response
          ↓
        Memory Update

        与原始 run_memory_agent / batch pipeline 保持相同逻辑。
        """

        user = "" if user is None else str(user).strip()

        if not user:
            raise ValueError("User input is empty.")

        # ----------------------------------------------------
        # Total Timer
        # ----------------------------------------------------

        total_start = time.perf_counter()

        latency = {
            "query": 0.0,
            "trigger": 0.0,
            "rag": 0.0,
            "topic": 0.0,
            "policy": 0.0,
            "response": 0.0,
            "total": 0.0
        }

        # ----------------------------------------------------
        # Add Current User to Dialogue History
        # ----------------------------------------------------

        self.dialogue_history += (
            f"用户: {user}\n"
        )

        # ====================================================
        # 1. Query
        # ====================================================

        stage_start = time.perf_counter()

        query_input = build_query_input(
            self.memory,
            user
        )

        query_messages = build_messages(
            self.query_prompt,
            query_input
        )

        query_output = parallel_inference(
            [query_messages]
        )[0]

        query_state = parse_llm_output(
            query_output
        )

        self.memory = update_issue_memory(
            self.memory,
            query_state
        )

        latency["query"] = (
            time.perf_counter() - stage_start
        )

        # ====================================================
        # 2. Trigger
        # ====================================================

        stage_start = time.perf_counter()

        trigger_input = build_trigger_input(
            self.memory
        )

        trigger_messages = build_messages(
            self.trigger_prompt,
            trigger_input
        )

        trigger_output = parallel_inference(
            [trigger_messages]
        )[0]

        trigger_state = parse_llm_output(
            trigger_output
        )

        latency["trigger"] = (
            time.perf_counter() - stage_start
        )

        # ====================================================
        # 3. RAG
        # ====================================================

        evidence = (
            self.memory["evidence_state"]
            ["active_cases"]
        )

        topic_structure = copy.deepcopy(
            self.memory["evidence_state"]
            ["topic_structure"]
        )

        if (
            trigger_state.get("evidence_mode")
            == "retrieve_new_evidence"
        ):
            query = (
                self.memory["issue_state"]
                ["current_issue"]
            )

            debug_x = {
                "dialog_id": "web",
                "user": user,
                "query_raw": query_output,
                "query": query_state,
                "trigger": trigger_state,
                "dialogue_history":
                    self.dialogue_history
            }

            query = validate_rag_query(
                query,
                debug_x,
                self.memory,
                self.turn_id
            )

            # ----------------------------------------------
            # RAG timer
            # ----------------------------------------------

            stage_start = time.perf_counter()

            evidence = batch_zefeng_rag(
                [self.dialogue_history],
                [query]
            )[0]

            latency["rag"] = (
                time.perf_counter() - stage_start
            )

            # =================================================
            # 3.5 Topic Structure
            # =================================================

            stage_start = time.perf_counter()

            structures = (
                generate_topic_structures_batch(
                    [
                        {
                            "query": query,
                            "cases": evidence
                        }
                    ],
                    self.topic_prompt
                )
            )

            topic_structure = structures[0]

            latency["topic"] = (
                time.perf_counter() - stage_start
            )

        # ====================================================
        # Update Evidence Memory
        # ====================================================

        self.memory = update_evidence_memory(
            self.memory,
            trigger_state,
            evidence,
            topic_structure
        )

        # ====================================================
        # 4. Policy
        # ====================================================

        stage_start = time.perf_counter()

        policy_input = build_policy_input(
            self.memory,
            user
        )

        policy_messages = build_messages(
            self.policy_prompt,
            policy_input
        )

        policy_output = parallel_inference(
            [policy_messages]
        )[0]

        policy_state = parse_llm_output(
            policy_output
        )

        self.memory = update_interaction_memory(
            self.memory,
            policy_state
        )

        latency["policy"] = (
            time.perf_counter() - stage_start
        )

        # ====================================================
        # 5. Response
        # ====================================================

        stage_start = time.perf_counter()

        response_input = build_response_input(
            self.memory,
            user
        )

        response_messages = build_messages(
            self.response_prompt,
            response_input
        )

        response_output = parallel_inference(
            [response_messages]
        )[0]

        response = parse_llm_output(
            response_output
        )

        latency["response"] = (
            time.perf_counter() - stage_start
        )

        # ====================================================
        # Response Text
        # ====================================================

        # 保留原始 response dict
        # 同时提取网页需要显示的文字

        if isinstance(response, dict):
            response_text = str(
                response.get("response", "")
            ).strip()

            # 防止 response.prompt 返回的不是
            # {"response": "..."} 格式
            if not response_text:
                response_text = str(
                    response.get("content", "")
                ).strip()

            if not response_text:
                response_text = json.dumps(
                    response,
                    ensure_ascii=False
                )
        else:
            response_text = str(response).strip()

        # ====================================================
        # Dialogue Memory
        # ====================================================

        # 网页端没有 golden response，
        # 因此这里写入模型实际生成的 response。
        #
        # 这是交互模式和离线 evaluation 唯一必然的区别：
        # 离线使用 golden_response；
        # 在线只能使用真实 Agent response。

        self.memory = update_dialogue_memory(
            self.memory,
            user,
            response_text
        )

        self.dialogue_history += (
            f"客服: {response_text}\n"
        )

        # ====================================================
        # Total Latency
        # ====================================================

        latency["total"] = (
            time.perf_counter() - total_start
        )

        # ====================================================
        # Save Current Turn State
        # ====================================================

        self.last_query = copy.deepcopy(
            query_state
        )

        self.last_trigger = copy.deepcopy(
            trigger_state
        )

        self.last_evidence = copy.deepcopy(
            evidence
        )

        self.last_topic_structure = copy.deepcopy(
            self.memory["evidence_state"]
            ["topic_structure"]
        )

        self.last_policy = copy.deepcopy(
            policy_state
        )

        self.last_response = copy.deepcopy(
            response
        )

        # 这里就是之前 UI latency={} 的关键
        self.last_latency = copy.deepcopy(
            latency
        )

        # ====================================================
        # Save Turn
        # ====================================================

        turn_result = {
            "turn_id": self.turn_id,
            "user": user,

            "query": copy.deepcopy(
                query_state
            ),

            "trigger": copy.deepcopy(
                trigger_state
            ),

            "evidence": copy.deepcopy(
                evidence
            ),

            "topic_structure": copy.deepcopy(
                self.last_topic_structure
            ),

            "policy": copy.deepcopy(
                policy_state
            ),

            "response": copy.deepcopy(
                response
            ),

            "latency": copy.deepcopy(
                latency
            )
        }

        self.turns.append(
            turn_result
        )

        self.turn_id += 1

        # ====================================================
        # Debug
        # ====================================================

        print(
            "[MemoryAgent] latency =",
            json.dumps(
                self.last_latency,
                ensure_ascii=False,
                indent=2
            )
        )

        # ====================================================
        # Return
        # ====================================================

        return response_text


    # ========================================================
    # State for Web UI
    # ========================================================

    def get_current_state(self):
        """
        web_gradio.py 统一从这里拿数据，
        避免 UI 到处 getattr。
        """

        return {
            "query": copy.deepcopy(
                self.last_query
            ),

            "trigger": copy.deepcopy(
                self.last_trigger
            ),

            "evidence": copy.deepcopy(
                self.last_evidence
            ),

            "topic_structure": copy.deepcopy(
                self.last_topic_structure
            ),

            "policy": copy.deepcopy(
                self.last_policy
            ),

            "response": copy.deepcopy(
                self.last_response
            ),

            "latency": copy.deepcopy(
                self.last_latency
            ),

            "memory": copy.deepcopy(
                self.memory
            ),

            "turns": copy.deepcopy(
                self.turns
            ),

            "dialogue_history":
                self.dialogue_history
        }
