def call_llm(self, prompt, data):
    messages = build_messages(prompt, data)

    if self.mode == "single":
        output = single_inference(messages)
    else:
        output = parallel_inference([messages])[0]

    print("\n========== RAW LLM OUTPUT ==========")
    print("type:", type(output))
    print("output:", repr(output))

    parsed = parse_llm_output(output)

    print("---------- PARSED OUTPUT -----------")
    print("type:", type(parsed))
    print("parsed:", parsed)
    print("====================================\n")

    return parsed
