# ============================================================
# Utils

def now_iso():
    return datetime.now().astimezone().isoformat(
        timespec="seconds"
    )


def turn_choices(state):
    """
    返回开发者界面可选择的 Turn。
    """
    if not state:
        return []

    turns = int(state.get("turns", 0) or 0)

    return [
        f"Turn {i}"
        for i in range(1, turns + 1)
    ]


def parse_turn(value, state):
    """
    将 'Turn 3' 转成 3。
    如果没有选择，则默认当前最新 Turn。
    """
    if value:
        try:
            return int(
                str(value)
                .replace("Turn", "")
                .strip()
            )
        except Exception:
            pass

    if not state:
        return 0

    return int(
        state.get("turns", 0) or 0
    )

def new_session():

    return {
        "session_id":
            "web_" + uuid.uuid4().hex[:10],

        "created_at":
            now_iso(),

        "agent":
            MemoryAgent(
                mode="single",
                show_state=False,
            ),

        "history": [],

        "trace": [],

        "prefix_history": [],

        "turns": 0,

        "mode": "user",

        "feedback": {},
    }

# Agent Snapshot

def get_last_run(agent):

    turn_state = getattr(
        agent,
        "last_turn_state",
        None,
    )

    if turn_state:

        return {
            "query_state":
                turn_state.get("query", {}) or {},

            "trigger_state":
                turn_state.get("trigger", {}) or {},

            "evidence_state":
                turn_state.get("evidence", {}) or {},

            "policy_state":
                turn_state.get("policy", {}) or {},

            "response":
                turn_state.get("response", ""),

            "latency":
                turn_state.get("latency", {}) or {},
        }

    memory = getattr(
        agent,
        "memory",
        {},
    ) or {}

    return {
        "query_state": {},

        "trigger_state": {},

        "evidence_state":
            memory.get(
                "evidence_state",
                {},
            ) or {},

        "policy_state":
            (
                memory
                .get(
                    "interaction_state",
                    {},
                )
                .get(
                    "last_policy",
                    {},
                )
                or {}
            ),

        "response": "",

        "latency": {},
    }

# History Parser
# ============================================================

def parse_prefix(text):

    history = []

    if not text:
        return history

    for raw_line in text.splitlines():

        line = raw_line.strip()

        if not line:
            continue

        if line.startswith("用户:"):

            content = line.split(
                "用户:",
                1,
            )[1].strip()

            history.append({
                "role": "user",
                "content": content,
            })

        elif line.startswith("用户："):

            content = line.split(
                "用户：",
                1,
            )[1].strip()

            history.append({
                "role": "user",
                "content": content,
            })

        elif line.startswith("客服:"):

            content = line.split(
                "客服:",
                1,
            )[1].strip()

            history.append({
                "role": "assistant",
                "content": content,
            })

        elif line.startswith("客服："):

            content = line.split(
                "客服：",
                1,
            )[1].strip()

            history.append({
                "role": "assistant",
                "content": content,
            })

    return history

def session_html(state):

    return render_session(
        state["session_id"],
        state["turns"],
    )


def developer_chat_html(state):

    return render_chat(
        state["history"],
        user_mode=False,
    )


def user_chat_html(state):

    return render_chat(
        state["history"],
        user_mode=True,
    )


def current_decision_html(state):

    if not state["trace"]:
        return render_decision(None)

    return render_decision(
        state["trace"][-1]
    )


def trace_html(state):

    return render_trace(
        state["trace"],
        state.get("feedback", {}),
    )


def feedback_hint_html(state):

    if not state:
        return render_feedback_hint(None)

    turn = state.get(
        "turns",
        0,
    )

    if turn <= 0:
        return render_feedback_hint(None)

    return render_feedback_hint(turn)

def chat(
    user_input,
    state,
):

    if state is None:
        state = new_session()

    user_input = str(
        user_input or ""
    ).strip()

    if not user_input:

        choices = turn_choices(state)

        latest = (
            choices[-1]
            if choices
            else None
        )

        return (
            "",
            user_chat_html(state),
            developer_chat_html(state),
            current_decision_html(state),
            trace_html(state),
            session_html(state),
            state,
            feedback_hint_html(state),
            "",
            gr.update(visible=False),

            # developer feedback
            gr.update(
                choices=choices,
                value=latest,
            ),
            "",
            gr.update(visible=False),
        )

    history = list(
        state["history"]
    )

    history.append({
        "role": "user",
        "content": user_input,
    })

    try:

        response = state[
            "agent"
        ].reply(
            history
        )

    except Exception as exc:

        response = (
            "Agent 执行失败："
            f"{type(exc).__name__}: {exc}"
        )

    response = str(
        response or ""
    ).strip()

    history.append({
        "role": "assistant",
        "content": response,
    })

    state["history"] = history

    state["turns"] += 1

    snapshot = dict(
        get_last_run(
            state["agent"]
        )
    )

    snapshot["turn"] = state["turns"]

    snapshot["user"] = user_input

    snapshot["response"] = response

    snapshot["created_at"] = now_iso()

    state["trace"].append(
        snapshot
    )

    save_session(state)

    choices = turn_choices(state)

    latest = (
        choices[-1]
        if choices
        else None
    )

    return (
        "",
        user_chat_html(state),
        developer_chat_html(state),
        current_decision_html(state),
        trace_html(state),
        session_html(state),
        state,
        feedback_hint_html(state),
        "",
        gr.update(visible=False),

        # developer feedback
        gr.update(
            choices=choices,
            value=latest,
        ),
        "",
        gr.update(visible=False),
    )

def submit_positive(state):

    if (
        state is None
        or state.get("turns", 0) <= 0
    ):

        return (
            state,
            render_feedback_result(
                False,
                "当前没有可反馈的回复。",
            ),
            gr.update(visible=False),
            trace_html(state)
            if state
            else render_trace([], {}),
        )

    turn = state["turns"]

    state.setdefault(
        "feedback",
        {},
    )

    state["feedback"][turn] = {
        "rating": "positive",
        "reason": None,
        "reason_text": None,
        "comment": "",
        "created_at": now_iso(),
    }

    save_session(state)

    return (
        state,
        render_feedback_result(
            True,
            "感谢您的反馈。",
        ),
        gr.update(visible=False),
        trace_html(state),
    )

def submit_negative(
    reason_text,
    comment,
    state,
):

    if (
        state is None
        or state.get("turns", 0) <= 0
    ):

        return (
            state,
            render_feedback_result(
                False,
                "当前没有可反馈的回复。",
            ),
            gr.update(visible=True),
            reason_text,
            comment,
            trace_html(state)
            if state
            else render_trace([], {}),
        )

    reason_text = str(
        reason_text or ""
    ).strip()

    if not reason_text:

        return (
            state,
            render_feedback_result(
                False,
                "请选择一个不满意原因。",
            ),
            gr.update(visible=True),
            reason_text,
            comment,
            trace_html(state),
        )

    turn = state["turns"]

    reason_code = FEEDBACK_REASONS.get(
        reason_text,
        "other",
    )

    comment = str(
        comment or ""
    ).strip()

    state.setdefault(
        "feedback",
        {},
    )

    state["feedback"][turn] = {
        "rating": "negative",
        "reason": reason_code,
        "reason_text": reason_text,
        "comment": comment,
        "created_at": now_iso(),
    }

    save_session(state)

    return (
        state,
        render_feedback_result(
            True,
            "感谢您的反馈。",
        ),
        gr.update(visible=False),
        None,
        "",
        trace_html(state),
    )

# Load Prefix
# ============================================================

def load_prefix(
    prefix_text,
    state,
):

    if state is None:
        state = new_session()

    history = parse_prefix(
        prefix_text
    )

    if not history:

        return (
            developer_chat_html(state),
            current_decision_html(state),
            trace_html(state),
            session_html(state),
            state,

            render_prefix_result(
                False,
                "没有识别到有效对话。",
            ),

            gr.update(
                choices=turn_choices(state),
                value=None,
            ),
        )

    state["agent"] = MemoryAgent(
        mode="single",
        show_state=False,
    )

    state["history"] = history

    state["trace"] = []

    state["prefix_history"] = list(
        history
    )

    state["turns"] = 0

    state["feedback"] = {}

    save_session(state)

    return (
        developer_chat_html(state),
        current_decision_html(state),
        trace_html(state),
        session_html(state),
        state,

        render_prefix_result(
            True,
            f"已加载 {len(history)} 条历史消息。",
        ),

        gr.update(
            choices=[],
            value=None,
        ),
    )
