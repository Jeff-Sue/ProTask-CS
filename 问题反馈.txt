# ============================================================

AGENT_PROMPT_DIR = ROOT_DIR / "Agent" / "promptsv2_3"

DEFAULT_MODE = "single"
DEFAULT_MAX_TOPIC_DEPTH = 8


# ============================================================
# Session
# ============================================================

def new_session() -> Dict[str, Any]:
    """
    一个浏览器 session 对应一个独立 MemoryAgent。

    Project 不维护 Agent 内部 memory，
    只维护：
    - session id
    - agent instance
    - frontend dialogue history
    - turn records
    """

    session_id = f"web_{uuid.uuid4().hex[:10]}"

    agent = MemoryAgent(
        prompt_dir=AGENT_PROMPT_DIR,
        mode=DEFAULT_MODE,
        max_topic_depth=DEFAULT_MAX_TOPIC_DEPTH,
        show_state=False,
    )

    return {
        "session_id": session_id,
        "agent": agent,
        "dialogue_history": [],
        "turn_records": [],
    }


# ============================================================
# Utils
# ============================================================

def safe_state_copy(state: Dict[str, Any]) -> Dict[str, Any]:
    """
    last_turn_state 当前都是 JSON-compatible 数据，
    这里 deepcopy 一份，避免后续 memory 更新影响历史记录。
    """
    try:
        return copy.deepcopy(state)
    except Exception:
        return json.loads(
            json.dumps(state, ensure_ascii=False, default=str)
        )


def get_latest_state(session: Dict[str, Any]) -> Dict[str, Any]:
    records = session.get("turn_records", [])

    if not records:
        return {}

    return records[-1].get("state", {})


# ============================================================
# Main Interaction
# ============================================================

def chat(
    user_input: str,
    session: Dict[str, Any],
):
    if session is None:
        session = new_session()

    user_input = str(user_input or "").strip()

    if not user_input:
        return (
            "",
            render_chat(session["dialogue_history"]),
            render_state(get_latest_state(session)),
            render_history(session["turn_records"]),
            render_session_info(session),
            session,
        )

    agent: MemoryAgent = session["agent"]

    # --------------------------------------------------------
    # 1. Add user message
    # --------------------------------------------------------

    dialogue_history = session["dialogue_history"] + [
        {
            "role": "user",
            "content": user_input,
        }
    ]

    # --------------------------------------------------------
    # 2. Agent
    # --------------------------------------------------------

    try:
        response = agent.reply(dialogue_history)
        turn_state = safe_state_copy(agent.last_turn_state)

    except Exception as e:
        error_message = str(e)

        # 用户消息仍保留，但不伪造 assistant 回复
        session["dialogue_history"] = dialogue_history

        return (
            "",
            render_chat(dialogue_history),
            render_error(error_message),
            render_history(session["turn_records"]),
            render_session_info(session),
            session,
        )

    # --------------------------------------------------------
    # 3. Add assistant message
    # --------------------------------------------------------

    dialogue_history.append(
        {
            "role": "assistant",
            "content": response,
        }
    )

    session["dialogue_history"] = dialogue_history

    # --------------------------------------------------------
    # 4. Save turn record
    # --------------------------------------------------------

    turn_id = len(session["turn_records"]) + 1

    session["turn_records"].append(
        {
            "turn_id": turn_id,
            "user": user_input,
            "assistant": response,
            "state": turn_state,
        }
    )

    # --------------------------------------------------------
    # 5. Render
    # --------------------------------------------------------

    return (
        "",
        render_chat(dialogue_history),
        render_state(turn_state),
        render_history(session["turn_records"]),
        render_session_info(session),
        session,
    )


# ============================================================
# Reset
# ============================================================

def reset_chat():
    session = new_session()

    return (
        "",
        render_chat([]),
        render_state({}),
        render_history([]),
        render_session_info(session),
        session,
    )


# ============================================================
# Prefix
# ============================================================

def parse_prefix(prefix_text: str) -> List[Dict[str, str]]:
    """
    支持：

    用户: xxx
    客服: xxx

    用户：xxx
    客服：xxx
    """

    prefix_text = str(prefix_text or "").strip()

    if not prefix_text:
        return []

    history = []

    for raw_line in prefix_text.splitlines():
        line = raw_line.strip()

        if not line:
            continue

        if line.startswith("用户:"):
            history.append({
                "role": "user",
                "content": line[len("用户:"):].strip(),
            })

        elif line.startswith("用户："):
            history.append({
                "role": "user",
                "content": line[len("用户："):].strip(),
            })

        elif line.startswith("客服:"):
            history.append({
                "role": "assistant",
                "content": line[len("客服:"):].strip(),
            })

        elif line.startswith("客服："):
            history.append({
                "role": "assistant",
                "content": line[len("客服："):].strip(),
            })

        else:
            if not history:
                raise ValueError(
                    f"无法识别历史对话：{line}"
                )

            history[-1]["content"] += "\n" + line

    return history


def load_prefix(
    prefix_text: str,
    session: Dict[str, Any],
):
    if session is None:
        session = new_session()

    try:
        history = parse_prefix(prefix_text)

        if not history:
            raise ValueError("请输入历史对话。")

        # Prefix 只作为 dialogue history。
        # 不重放 Agent，因此重新创建 Agent，保证内部 memory 为空。
        new_agent = MemoryAgent(
            prompt_dir=AGENT_PROMPT_DIR,
            mode=DEFAULT_MODE,
            max_topic_depth=DEFAULT_MAX_TOPIC_DEPTH,
            show_state=False,
        )

        session["agent"] = new_agent
        session["dialogue_history"] = history
        session["turn_records"] = []

        status = f"""
        <div class="prefix-success">
            历史对话已加载，共 {len(history)} 条消息。
            下一条用户输入将基于该历史继续处理。
        </div>
        """

        return (
            render_chat(history),
            render_state({}),
            render_history([]),
            render_session_info(session),
            session,
            status,
        )

    except Exception as e:
        return (
            render_chat(session["dialogue_history"]),
            render_state(get_latest_state(session)),
            render_history(session["turn_records"]),
            render_session_info(session),
            session,
            f"""
            <div class="prefix-error">
                加载失败：{str(e)}
            </div>
            """,
        )
