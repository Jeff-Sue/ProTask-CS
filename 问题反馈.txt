import argparse
import json
import re


# ============================================================
# Config
# ============================================================

KNOWLEDGE_POLICY = {
    "ExplainedResponse",
    "CaseRecommendation"
}


# ============================================================
# Utils
# ============================================================

def load_json(path):
    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)


def extract_case_ids(text):
    if not text:
        return []
    return re.findall(r"KT\d+", str(text))


def get_policy_label(policy):
    if not isinstance(policy, dict):
        return ""
    return str(policy.get("label", "")).strip()


def get_pred_cases(policy):
    if not isinstance(policy, dict):
        return []

    label = get_policy_label(policy)
    if label not in KNOWLEDGE_POLICY:
        return []

    information = policy.get("information") or {}
    if not isinstance(information, dict):
        return []

    cases = information.get("case_id", [])

    if isinstance(cases, str):
        cases = [cases]

    if not isinstance(cases, list):
        return []

    return list(dict.fromkeys(
        str(case_id).strip()
        for case_id in cases
        if str(case_id).strip()
    ))


def get_retrieved_cases(turn):
    return [
        str(case.get("case_id", "")).strip()
        for case in turn.get("evidence", [])
        if isinstance(case, dict) and str(case.get("case_id", "")).strip()
    ]


def rate(values):
    return sum(values) / len(values) if values else 0


# ============================================================
# Case Turn Evaluation
# ============================================================

def evaluate_case_turn(data):
    total_case_turn = 0
    trigger_hit = 0
    top5_hit = 0
    top3_hit = 0
    top1_hit = 0
    policy_hit = 0
    selection_hit = 0

    for dialog in data.values():
        for turn in dialog.get("turns", []):
            golden_cases = extract_case_ids(turn.get("golden_response", ""))

            if not golden_cases:
                continue

            total_case_turn += 1
            golden_case = golden_cases[0]

            trigger_state = turn.get("trigger", {})
            trigger = trigger_state.get("trigger", False)

            if not trigger:
                continue

            trigger_hit += 1

            retrieved = get_retrieved_cases(turn)

            hit5 = golden_case in retrieved[:5]
            hit3 = golden_case in retrieved[:3]
            hit1 = golden_case in retrieved[:1]

            if hit5:
                top5_hit += 1
            if hit3:
                top3_hit += 1
            if hit1:
                top1_hit += 1

            if not hit5:
                continue

            policy = turn.get("policy", {})
            label = get_policy_label(policy)

            if label not in KNOWLEDGE_POLICY:
                continue

            policy_hit += 1

            pred_cases = get_pred_cases(policy)

            if golden_case in pred_cases:
                selection_hit += 1

    return {
        "案例轮数量": total_case_turn,
        "知识链路触发率": trigger_hit / total_case_turn if total_case_turn else 0,
        "Top-5有效案例召回率": top5_hit / trigger_hit if trigger_hit else 0,
        "Top-3有效案例召回率": top3_hit / trigger_hit if trigger_hit else 0,
        "Top-1有效案例召回率": top1_hit / trigger_hit if trigger_hit else 0,
        "知识型策略命中率": policy_hit / top5_hit if top5_hit else 0,
        "案例选择准确率": selection_hit / policy_hit if policy_hit else 0,
        "最终案例推荐准确率": selection_hit / total_case_turn if total_case_turn else 0
    }


# ============================================================
# Recommendation Timing
# ============================================================

def evaluate_timing(data):
    pred = []
    gold = []

    for dialog in data.values():
        for turn in dialog.get("turns", []):
            golden_case = bool(extract_case_ids(turn.get("golden_response", "")))

            policy = turn.get("policy", {})
            policy_label = get_policy_label(policy)
            predicted = policy_label in KNOWLEDGE_POLICY

            gold.append(golden_case)
            pred.append(predicted)

    tp = sum(p and g for p, g in zip(pred, gold))
    fp = sum(p and not g for p, g in zip(pred, gold))
    fn = sum((not p) and g for p, g in zip(pred, gold))
    tn = sum((not p) and (not g) for p, g in zip(pred, gold))

    precision = tp / (tp + fp) if tp + fp else 0
    recall = tp / (tp + fn) if tp + fn else 0
    f1 = 2 * precision * recall / (precision + recall) if precision + recall else 0

    return {
        "推荐时机TP": tp,
        "推荐时机FP": fp,
        "推荐时机FN": fn,
        "推荐时机TN": tn,
        "推荐时机Precision": precision,
        "推荐时机Recall": recall,
        "推荐时机F1": f1,
        "过早推荐率": fp / (tp + fp) if tp + fp else 0,
        "漏推荐率": fn / (tp + fn) if tp + fn else 0
    }


# ============================================================
# Dialogue Level
# ============================================================

def evaluate_dialogue(data):
    success = []
    golden_first = []
    pred_first = []

    for dialog in data.values():
        golden_cases = set()
        predicted_cases = set()

        golden_turn = None
        predicted_turn = None

        for turn in dialog.get("turns", []):
            turn_id = turn.get("turn_id")

            g_cases = extract_case_ids(turn.get("golden_response", ""))

            for case_id in g_cases:
                golden_cases.add(case_id)

                if golden_turn is None:
                    golden_turn = turn_id

            p_cases = get_pred_cases(turn.get("policy", {}))

            for case_id in p_cases:
                predicted_cases.add(case_id)

                if predicted_turn is None:
                    predicted_turn = turn_id

        success.append(bool(golden_cases & predicted_cases))

        if golden_turn is not None:
            golden_first.append(golden_turn)

        if predicted_turn is not None:
            pred_first.append(predicted_turn)

    return {
        "对话数量": len(success),
        "对话成功率": rate(success),
        "Golden案例首次推荐平均轮数": rate(golden_first),
        "Predicted案例首次推荐平均轮数": rate(pred_first)
    }


# ============================================================
# Policy Distribution
# ============================================================

def evaluate_policy_distribution(data):
    counts = {}
    total = 0
    empty = 0

    for dialog in data.values():
        for turn in dialog.get("turns", []):
            label = get_policy_label(turn.get("policy", {}))
            total += 1

            if not label:
                empty += 1
                label = "EMPTY"

            counts[label] = counts.get(label, 0) + 1

    result = {
        "总轮数": total,
        "空Policy数量": empty,
        "空Policy比例": empty / total if total else 0
    }

    for label, count in sorted(counts.items()):
        result[f"Policy_{label}"] = count

    return result


# ============================================================
# Output
# ============================================================

def print_result(title, result):
    print(f"\n========== {title} ==========")

    for key, value in result.items():
        if isinstance(value, float):
            print(f"{key}: {value:.4f}")
        else:
            print(f"{key}: {value}")


# ============================================================
# Main
# ============================================================

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--input", required=True)
    args = parser.parse_args()

    data = load_json(args.input)

    print_result("案例轮指标", evaluate_case_turn(data))
    print_result("所有轮推荐时机", evaluate_timing(data))
    print_result("对话级指标", evaluate_dialogue(data))
    print_result("Policy分布", evaluate_policy_distribution(data))


if __name__ == "__main__":
    main()
