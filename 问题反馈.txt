import argparse
import copy
import json
from pathlib import Path
from typing import Any

from utils.scu_rag import batch_zefeng_rag
from utils.qwen_api_multi import parallel_inference


# ============================================================
# Utils
# ============================================================

def parse_json(text):
    if isinstance(text, dict):
        return text
    if not isinstance(text, str):
        return {}
    text = text.strip()
    if not text:
        return {}
    try:
        return json.loads(text)
    except Exception:
        pass
    if "```json" in text:
        text = text.split("```json", 1)[1].split("```", 1)[0].strip()
    elif "```" in text:
        text = text.split("```", 1)[1].split("```", 1)[0].strip()
    try:
        return json.loads(text)
    except Exception:
        return {}


def parse_llm_output(output):
    parsed = parse_json(output)
    if not parsed:
        return {}
    if "response" not in parsed:
        return parsed
    return parse_json(parsed.get("response", ""))


def prompt_read(path: Path) -> str:
    if not path.exists():
        raise FileNotFoundError(f"Prompt file not found: {path}")
    return path.read_text(encoding="utf-8").strip()


def write_json(path: str | Path, data: Any) -> None:
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")


def build_messages(system_prompt, data):
    return [
        {"role": "system", "content": system_prompt},
        {"role": "user", "content": json.dumps(data, ensure_ascii=False)}
    ]


def get_case_ids(cases):
    return [x.get("case_id") for x in cases if isinstance(x, dict) and x.get("case_id")]


# ============================================================
# Memory
# ============================================================

def init_memory():
    return {
        "issue_state": {
            "current_issue": "",
            "query_history": []
        },
        "dialog_state": {
            "compressed_history": []
        },
        "evidence_state": {
            "active_cases": [],
            "topic_structure": {}
        },
        "interaction_state": {
            "stage": "Start",
            "last_policy": {
                "label": "",
                "information": None
            },
            "policy_history": []
        },
        "_runtime": {
            "structure_query": "",
            "structure_case_ids": []
        }
    }


# ============================================================
# Memory Update
# ============================================================

def update_issue_memory(memory, query_state):
    mode = query_state.get("query_mode", "none")
    query = query_state.get("query_text", "").strip()

    if mode in ["rewrite", "refine"] and query:
        old_query = memory["issue_state"]["current_issue"]
        memory["issue_state"]["current_issue"] = query
        if query != old_query:
            memory["issue_state"]["query_history"].append(query)

    return memory


def update_evidence_memory(memory, trigger_state, evidence):
    mode = trigger_state.get("evidence_mode", "no_evidence_needed")

    if mode == "retrieve_new_evidence":
        memory["evidence_state"]["active_cases"] = evidence or []

    return memory


def update_topic_structure_memory(memory, topic_structure):
    memory["evidence_state"]["topic_structure"] = topic_structure or {}
    return memory


def update_interaction_memory(memory, policy_state):
    label = policy_state.get("label", "")
    information = policy_state.get("information")

    policy = {
        "label": label,
        "information": information
    }

    memory["interaction_state"]["last_policy"] = policy
    memory["interaction_state"]["policy_history"].append(policy)

    if label in ["AskMissingSlot", "AskClarification"]:
        memory["interaction_state"]["stage"] = "Clarification"
    elif label in ["ExplainedResponse", "CaseRecommendation"]:
        memory["interaction_state"]["stage"] = "Resolution"
    elif label == "Handoff":
        memory["interaction_state"]["stage"] = "End"

    return memory


def update_dialogue_memory(memory, user, response):
    history = memory["dialog_state"]["compressed_history"]
    history.append({"user": user, "assistant": response})
    memory["dialog_state"]["compressed_history"] = history[-5:]
    return memory


# ============================================================
# Structure Cache
# ============================================================

def need_structure_update(memory):
    current_query = memory["issue_state"]["current_issue"]
    current_case_ids = get_case_ids(memory["evidence_state"]["active_cases"])

    old_query = memory["_runtime"]["structure_query"]
    old_case_ids = memory["_runtime"]["structure_case_ids"]

    if not memory["evidence_state"]["topic_structure"]:
        return True
    if current_query != old_query:
        return True
    if current_case_ids != old_case_ids:
        return True

    return False


def mark_structure_version(memory):
    memory["_runtime"]["structure_query"] = memory["issue_state"]["current_issue"]
    memory["_runtime"]["structure_case_ids"] = get_case_ids(memory["evidence_state"]["active_cases"])
    return memory


# ============================================================
# Prompt Input Builder
# ============================================================

def build_query_input(memory, user):
    return {
        "issue_state": memory["issue_state"],
        "dialog_state": memory["dialog_state"],
        "current": {"u": user}
    }


def build_trigger_input(memory):
    return {
        "issue_state": memory["issue_state"],
        "evidence_state": {
            "active_cases": memory["evidence_state"]["active_cases"],
            "topic_structure": memory["evidence_state"]["topic_structure"]
        },
        "dialog_state": memory["dialog_state"]
    }


def build_structure_input(memory):
    return {
        "query_text": memory["issue_state"]["current_issue"],
        "active_cases": memory["evidence_state"]["active_cases"]
    }


def build_policy_input(memory, user):
    return {
        "issue_state": memory["issue_state"],
        "evidence_state": {
            "topic_structure": memory["evidence_state"]["topic_structure"]
        },
        "interaction_state": memory["interaction_state"],
        "dialog_state": memory["dialog_state"],
        "current": {"u": user}
    }


def build_response_input(memory, user):
    return {
        "issue_state": memory["issue_state"],
        "evidence_state": {
            "active_cases": memory["evidence_state"]["active_cases"],
            "topic_structure": memory["evidence_state"]["topic_structure"]
        },
        "interaction_state": memory["interaction_state"],
        "dialog_state": memory["dialog_state"],
        "current": {
            "u": user,
            "policy": memory["interaction_state"]["last_policy"]
        }
    }


# ============================================================
# Topic Structure Validation
# ============================================================

def validate_topic_node(node, allowed_case_ids, is_root=False):
    if not isinstance(node, dict):
        return False

    case_ids = node.get("case_ids", [])
    children = node.get("children", [])

    if not isinstance(case_ids, list) or not case_ids:
        return False
    if not isinstance(children, list):
        return False

    case_set = set(case_ids)

    if len(case_set) != len(case_ids):
        return False
    if not case_set.issubset(allowed_case_ids):
        return False

    if not is_root:
        topic = node.get("topic", "")
        if not isinstance(topic, str) or not topic.strip():
            return False

    # Leaf: ideally one unique case
    if not children:
        return len(case_ids) == 1

    dimension = node.get("discriminative_dimension", "")
    if not isinstance(dimension, str) or not dimension.strip():
        return False

    child_sets = []

    for child in children:
        if not validate_topic_node(child, allowed_case_ids, is_root=False):
            return False

        child_set = set(child.get("case_ids", []))

        if not child_set.issubset(case_set):
            return False

        child_sets.append(child_set)

    # Sibling topics must partition the parent case set
    for i in range(len(child_sets)):
        for j in range(i + 1, len(child_sets)):
            if child_sets[i] & child_sets[j]:
                return False

    union_cases = set()
    for child_set in child_sets:
        union_cases |= child_set

    if union_cases != case_set:
        return False

    return True


def validate_topic_structure(structure, active_cases):
    if not isinstance(structure, dict):
        return False

    common_anchor = structure.get("common_anchor", "")
    root = structure.get("root")

    if not isinstance(common_anchor, str) or not common_anchor.strip():
        return False
    if not isinstance(root, dict):
        return False

    allowed_case_ids = set(get_case_ids(active_cases))
    root_case_ids = set(root.get("case_ids", []))

    if not allowed_case_ids:
        return False
    if root_case_ids != allowed_case_ids:
        return False

    return validate_topic_node(root, allowed_case_ids, is_root=True)


# ============================================================
# Topic Structure Generator Batch
# ============================================================

def run_structure_batch(items, memories, structure_prompt):
    messages = []
    structure_items = []

    for x in items:
        dialog_id = x["dialog_id"]
        memory = memories[dialog_id]

        if not x["structure_update"]:
            x["topic_structure"] = copy.deepcopy(memory["evidence_state"]["topic_structure"])
            continue

        if not memory["evidence_state"]["active_cases"]:
            x["topic_structure"] = {}
            memory["evidence_state"]["topic_structure"] = {}
            memories[dialog_id] = mark_structure_version(memory)
            continue

        messages.append(build_messages(structure_prompt, build_structure_input(memory)))
        structure_items.append(x)

    if not messages:
        return

    outputs = parallel_inference(messages)

    for x, out in zip(structure_items, outputs):
        dialog_id = x["dialog_id"]
        memory = memories[dialog_id]

        structure = parse_llm_output(out)

        if not validate_topic_structure(structure, memory["evidence_state"]["active_cases"]):
            x["structure_error"] = "invalid_topic_structure"
            x["raw_topic_structure"] = copy.deepcopy(structure)
            structure = {}

        x["topic_structure"] = structure
        memories[dialog_id] = update_topic_structure_memory(memory, structure)
        memories[dialog_id] = mark_structure_version(memories[dialog_id])


# ============================================================
# Batch Agent
# ============================================================

def run_memory_agent_batch(data):
    query_prompt = prompt_read(Path("prompts/query.prompt"))
    trigger_prompt = prompt_read(Path("prompts/trigger.prompt"))
    structure_prompt = prompt_read(Path("prompts/structure.prompt"))
    policy_prompt = prompt_read(Path("prompts/policy.prompt"))
    response_prompt = prompt_read(Path("prompts/response.prompt"))

    memories = {dialog_id: init_memory() for dialog_id in data}
    dialogue_history = {dialog_id: "" for dialog_id in data}
    results = {dialog_id: {"turns": []} for dialog_id in data}

    max_turn = max(len(item["text"]) for item in data.values())

    for t_id in range(max_turn):
        print(f"\n========== TURN {t_id} ==========")

        current_batch = []

        # ====================================================
        # Collect Current User Turns
        # ====================================================

        for dialog_id, item in data.items():
            if t_id >= len(item["text"]):
                continue

            msg = item["text"][t_id]

            if "用户" in msg:
                dialogue_history[dialog_id] += f"用户: {msg['用户']}\n"
            elif "客服" in msg:
                dialogue_history[dialog_id] += f"客服: {msg['客服']}\n"

            if "用户" not in msg:
                continue

            golden_response = ""
            if t_id + 1 < len(item["text"]) and "客服" in item["text"][t_id + 1]:
                golden_response = item["text"][t_id + 1]["客服"]

            current_batch.append({
                "dialog_id": dialog_id,
                "user": msg["用户"],
                "golden_response": golden_response,
                "dialogue_history": dialogue_history[dialog_id]
            })

        if not current_batch:
            continue

        # ====================================================
        # 1. Query Batch
        # ====================================================

        print(f"[Turn {t_id}] Query Batch: {len(current_batch)}")

        query_messages = [
            build_messages(
                query_prompt,
                build_query_input(memories[x["dialog_id"]], x["user"])
            )
            for x in current_batch
        ]

        query_outputs = parallel_inference(query_messages)

        for x, out in zip(current_batch, query_outputs):
            query_state = parse_llm_output(out)
            x["query"] = query_state
            dialog_id = x["dialog_id"]
            memories[dialog_id] = update_issue_memory(memories[dialog_id], query_state)

        # ====================================================
        # 2. Trigger Batch
        # ====================================================

        print(f"[Turn {t_id}] Trigger Batch")

        trigger_messages = [
            build_messages(trigger_prompt, build_trigger_input(memories[x["dialog_id"]]))
            for x in current_batch
        ]

        trigger_outputs = parallel_inference(trigger_messages)

        for x, out in zip(current_batch, trigger_outputs):
            x["trigger"] = parse_llm_output(out)

        # ====================================================
        # 3. RAG Batch
        # ====================================================

        print(f"[Turn {t_id}] RAG Batch")

        rag_dialogs = []
        rag_queries = []
        rag_indices = []

        for idx, x in enumerate(current_batch):
            dialog_id = x["dialog_id"]
            memory = memories[dialog_id]

            if x["trigger"].get("evidence_mode") == "retrieve_new_evidence":
                rag_dialogs.append(x["dialogue_history"])
                rag_queries.append(memory["issue_state"]["current_issue"])
                rag_indices.append(idx)
            else:
                x["evidence"] = copy.deepcopy(memory["evidence_state"]["active_cases"])

        if rag_dialogs:
            rag_outputs = batch_zefeng_rag(rag_dialogs, rag_queries)

            for idx, evidence in zip(rag_indices, rag_outputs):
                current_batch[idx]["evidence"] = evidence

        for x in current_batch:
            dialog_id = x["dialog_id"]
            memories[dialog_id] = update_evidence_memory(
                memories[dialog_id],
                x["trigger"],
                x.get("evidence", [])
            )

        # ====================================================
        # 4. Topic Structure Batch
        # ====================================================

        print(f"[Turn {t_id}] Topic Structure Batch")

        for x in current_batch:
            x["structure_update"] = need_structure_update(memories[x["dialog_id"]])

        run_structure_batch(
            items=current_batch,
            memories=memories,
            structure_prompt=structure_prompt
        )

        # ====================================================
        # 5. Policy Batch
        # ====================================================

        print(f"[Turn {t_id}] Policy Batch")

        policy_messages = [
            build_messages(
                policy_prompt,
                build_policy_input(memories[x["dialog_id"]], x["user"])
            )
            for x in current_batch
        ]

        policy_outputs = parallel_inference(policy_messages)

        for x, out in zip(current_batch, policy_outputs):
            policy_state = parse_llm_output(out)
            x["policy"] = policy_state
            dialog_id = x["dialog_id"]
            memories[dialog_id] = update_interaction_memory(memories[dialog_id], policy_state)

        # ====================================================
        # 6. Response Batch
        # ====================================================

        print(f"[Turn {t_id}] Response Batch")

        response_messages = [
            build_messages(
                response_prompt,
                build_response_input(memories[x["dialog_id"]], x["user"])
            )
            for x in current_batch
        ]

        response_outputs = parallel_inference(response_messages)

        for x, out in zip(current_batch, response_outputs):
            x["response"] = parse_llm_output(out)

            dialog_id = x["dialog_id"]
            memories[dialog_id] = update_dialogue_memory(
                memories[dialog_id],
                x["user"],
                x["golden_response"]
            )

            turn_result = {
                k: v for k, v in x.items()
                if k not in ["dialogue_history", "structure_update"]
            }

            turn_result["turn_id"] = int(t_id / 2)
            results[dialog_id]["turns"].append(turn_result)

    # ========================================================
    # Final Memory
    # ========================================================

    for dialog_id in data:
        final_memory = copy.deepcopy(memories[dialog_id])
        final_memory.pop("_runtime", None)
        results[dialog_id]["final_memory"] = final_memory

    return results


# ============================================================
# Main
# ============================================================

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--data_path", required=True)
    parser.add_argument("--output_path", required=True)
    parser.add_argument("--max_dialogues", type=int, default=10)
    args = parser.parse_args()

    with open(args.data_path, "r", encoding="utf-8") as f:
        data = json.load(f)

    data = dict(list(data.items())[:args.max_dialogues])

    outputs = run_memory_agent_batch(data)

    write_json(args.output_path, outputs)
    print(f"Saved {len(outputs)} dialogues to {args.output_path}")


if __name__ == "__main__":
    main()
