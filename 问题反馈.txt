from __future__ import annotations

import json
from datetime import datetime
from pathlib import Path
from typing import Any, Dict


# ============================================================
# Paths
# ============================================================

PROJECT_DIR = Path(__file__).resolve().parent
LOG_DIR = PROJECT_DIR / "logs"


# ============================================================
# Utils
# ============================================================

def now_iso():
    return datetime.now().astimezone().isoformat(timespec="seconds")


def ensure_log_dir():
    LOG_DIR.mkdir(
        parents=True,
        exist_ok=True,
    )


def get_log_path(session_id):
    ensure_log_dir()

    safe_session_id = str(
        session_id
    ).replace("/", "_").replace("\\", "_")

    return LOG_DIR / f"{safe_session_id}.json"


def json_safe(value):
    """
    将对象转换成可以 JSON 序列化的形式。

    正常情况下 trace 内部都是 dict/list/str，
    这里主要作为保险。
    """

    if value is None:
        return None

    if isinstance(
        value,
        (str, int, float, bool),
    ):
        return value

    if isinstance(value, dict):
        return {
            str(key): json_safe(item)
            for key, item in value.items()
        }

    if isinstance(value, (list, tuple)):
        return [
            json_safe(item)
            for item in value
        ]

    return str(value)


# ============================================================
# Session Log
# ============================================================

def build_session_log(state):
    """
    从 Gradio session state 构造最终保存的数据。

    注意：
    - agent 不保存
    - history 主要用于界面
    - trace 才是系统真实执行出来的 turn
    - feedback 根据 turn 合并到对应 turn
    """

    feedback_map = (
        state.get("feedback", {})
        or {}
    )

    turns = []

    for trace_item in (
        state.get("trace", [])
        or []
    ):
        item = dict(trace_item)

        turn_id = item.get("turn")

        feedback = (
            feedback_map.get(turn_id)
            or feedback_map.get(str(turn_id))
        )

        item["feedback"] = feedback

        turns.append(
            json_safe(item)
        )

    data = {
        "session_id": state.get(
            "session_id",
            "",
        ),

        "created_at": state.get(
            "created_at",
            "",
        ),

        "updated_at": now_iso(),

        "prefix_history": json_safe(
            state.get(
                "prefix_history",
                [],
            )
        ),

        "turns": turns,
    }

    return data


def save_session(state):
    """
    保存整个 session。

    每次：
    - Agent 完成一轮
    - 用户提交反馈

    都调用一次。

    使用临时文件 + replace，避免写一半时留下损坏 JSON。
    """

    if not state:
        return None

    session_id = state.get(
        "session_id"
    )

    if not session_id:
        return None

    path = get_log_path(
        session_id
    )

    data = build_session_log(
        state
    )

    temp_path = path.with_suffix(
        ".json.tmp"
    )

    temp_path.write_text(
        json.dumps(
            data,
            ensure_ascii=False,
            indent=2,
        ),
        encoding="utf-8",
    )

    temp_path.replace(path)

    return path
