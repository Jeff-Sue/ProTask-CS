import html
from typing import Any, Dict, List


# ============================================================
# Utils
# ============================================================

def esc(value: Any) -> str:
    return html.escape(str(value if value is not None else ""))


def badge(text: Any, kind: str = "default") -> str:
    return (
        f'<span class="badge badge-{kind}">'
        f'{esc(text)}'
        f'</span>'
    )


def ms(value: Any) -> str:
    try:
        return f"{int(value):,} ms"
    except Exception:
        return "0 ms"


# ============================================================
# Chat
# ============================================================

def render_chat(
    history: List[Dict[str, str]],
) -> str:

    if not history:
        return """
        <div class="empty-chat">

            <div class="empty-icon">
                ✦
            </div>

            <div class="empty-title">
                开始新的 IT 支持会话
            </div>

            <div class="empty-desc">
                输入用户问题后，
                Agent 将动态完成 Query、Retrieval、
                Topic Analysis、Policy 和 Response。
            </div>

        </div>
        """

    blocks = []

    for msg in history:

        role = msg.get("role")
        content = esc(msg.get("content", ""))

        if role == "user":

            blocks.append(
                f"""
                <div class="message-row user-row">

                    <div class="message-bubble user-message">
                        <div class="message-role">
                            User
                        </div>

                        <div>
                            {content}
                        </div>
                    </div>

                    <div class="avatar user-avatar">
                        U
                    </div>

                </div>
                """
            )

        else:

            blocks.append(
                f"""
                <div class="message-row">

                    <div class="avatar agent-avatar">
                        AI
                    </div>

                    <div class="message-bubble agent-message">
                        <div class="message-role">
                            IT Support
                        </div>

                        <div>
                            {content}
                        </div>
                    </div>

                </div>
                """
            )

    return (
        '<div class="chat-scroll">'
        + "".join(blocks)
        + "</div>"
    )


# ============================================================
# Evidence
# ============================================================

def render_cases(cases: List[Dict[str, Any]]) -> str:

    if not cases:
        return """
        <div class="empty-small">
            No active evidence
        </div>
        """

    items = []

    for case in cases:

        case_id = (
            case.get("case_id")
            or case.get("id")
            or case.get("caseId")
            or ""
        )

        title = case.get("title", "")

        items.append(
            f"""
            <div class="case-item">

                <div class="case-id">
                    {esc(case_id)}
                </div>

                <div class="case-title">
                    {esc(title)}
                </div>

            </div>
            """
        )

    return "".join(items)


# ============================================================
# Topic Tree
# ============================================================

def render_topic_tree(
    tree: Dict[str, Any],
) -> str:

    if not tree:
        return """
        <div class="empty-small">
            No topic structure
        </div>
        """

    def node(node_data, level=0):

        topic = esc(
            node_data.get("topic", "")
        )

        dimension = esc(
            node_data.get("dimension", "")
        )

        case_ids = node_data.get(
            "case_ids",
            [],
        )

        leaf = node_data.get(
            "leaf",
            False,
        )

        children = node_data.get(
            "children",
            [],
        )

        case_text = ""

        if case_ids:
            case_text = (
                '<span class="tree-case">'
                + esc(", ".join(case_ids))
                + "</span>"
            )

        dimension_text = ""

        if dimension:
            dimension_text = (
                '<span class="tree-dimension">'
                + dimension
                + "</span>"
            )

        leaf_class = (
            " tree-leaf"
            if leaf
            else ""
        )

        html_block = f"""
        <div
            class="tree-node{leaf_class}"
            style="margin-left:{level * 18}px"
        >

            <span class="tree-dot"></span>

            <span class="tree-topic">
                {topic}
            </span>

            {dimension_text}
            {case_text}

        </div>
        """

        for child in children:
            html_block += node(
                child,
                level + 1,
            )

        return html_block

    return (
        '<div class="topic-tree">'
        + node(tree)
        + "</div>"
    )


# ============================================================
# Current State
# ============================================================

def render_state(
    state: Dict[str, Any],
) -> str:

    if not state:
        return """
        <div class="state-empty">

            <div class="state-empty-title">
                Waiting for interaction
            </div>

            <div class="state-empty-desc">
                当前轮执行后，
                Query、Trigger、Evidence、
                Policy 和 Latency 将显示在这里。
            </div>

        </div>
        """

    query = state.get("query", {})
    trigger = state.get("trigger", {})
    evidence = state.get("evidence", {})
    policy = state.get("policy", {})
    latency = state.get("latency", {})

    query_mode = query.get(
        "query_mode",
        "none",
    )

    query_text = query.get(
        "query_text",
        "",
    )

    evidence_mode = trigger.get(
        "evidence_mode",
        "",
    )

    trigger_flag = trigger.get(
        "trigger",
        False,
    )

    policy_label = policy.get(
        "label",
        "",
    )

    information = policy.get(
        "information",
    )

    active_cases = evidence.get(
        "active_cases",
        [],
    )

    topic_structure = evidence.get(
        "topic_structure",
        {},
    )

    return f"""

    <div class="state-section">

        <div class="section-title">
            Query
        </div>

        <div class="kv-row">
            <span>Mode</span>
            {badge(query_mode, "blue")}
        </div>

        <div class="state-text">
            {esc(query_text) or "None"}
        </div>

    </div>


    <div class="state-section">

        <div class="section-title">
            Evidence Decision
        </div>

        <div class="kv-row">
            <span>Trigger</span>
            {badge(
                trigger_flag,
                "green" if trigger_flag else "gray"
            )}
        </div>

        <div class="kv-row">
            <span>Mode</span>
            {badge(evidence_mode, "blue")}
        </div>

    </div>


    <div class="state-section">

        <div class="section-title">
            Active Cases
        </div>

        {render_cases(active_cases)}

    </div>


    <div class="state-section">

        <div class="section-title">
            Topic Structure
        </div>

        {render_topic_tree(topic_structure)}

    </div>


    <div class="state-section">

        <div class="section-title">
            Policy
        </div>

        <div class="policy-main">
            {badge(policy_label, "green")}
        </div>

        <div class="state-label">
            Information
        </div>

        <pre class="json-small">{esc(information)}</pre>

    </div>


    <div class="state-section">

        <div class="section-title">
            Latency
        </div>

        <div class="latency-total">
            {ms(latency.get("total_ms", 0))}
        </div>

        <div class="latency-grid">

            <div>
                <span>Query</span>
                <b>{ms(latency.get("query_ms", 0))}</b>
            </div>

            <div>
                <span>Trigger</span>
                <b>{ms(latency.get("trigger_ms", 0))}</b>
            </div>

            <div>
                <span>RAG</span>
                <b>{ms(latency.get("rag_ms", 0))}</b>
            </div>

            <div>
                <span>Topic</span>
                <b>{ms(latency.get("topic_ms", 0))}</b>
            </div>

            <div>
                <span>Policy</span>
                <b>{ms(latency.get("policy_ms", 0))}</b>
            </div>

            <div>
                <span>Response</span>
                <b>{ms(latency.get("response_ms", 0))}</b>
            </div>

        </div>

    </div>

    """


# ============================================================
# Turn History
# ============================================================

def render_history(
    records: List[Dict[str, Any]],
) -> str:

    if not records:
        return """
        <div class="state-empty">
            暂无运行轨迹
        </div>
        """

    blocks = []

    for record in reversed(records):

        state = record.get(
            "state",
            {},
        )

        query = state.get(
            "query",
            {},
        )

        trigger = state.get(
            "trigger",
            {},
        )

        policy = state.get(
            "policy",
            {},
        )

        latency = state.get(
            "latency",
            {},
        )

        blocks.append(
            f"""
            <div class="history-item">

                <div class="history-head">

                    <b>
                        Turn {record.get("turn_id")}
                    </b>

                    <span>
                        {ms(latency.get("total_ms", 0))}
                    </span>

                </div>

                <div class="history-user">
                    {esc(record.get("user", ""))}
                </div>

                <div class="history-flow">

                    {badge(
                        query.get("query_mode", ""),
                        "blue"
                    )}

                    <span>→</span>

                    {badge(
                        trigger.get("evidence_mode", ""),
                        "gray"
                    )}

                    <span>→</span>

                    {badge(
                        policy.get("label", ""),
                        "green"
                    )}

                </div>

            </div>
            """
        )

    return "".join(blocks)


# ============================================================
# Session
# ============================================================

def render_session_info(
    session: Dict[str, Any],
) -> str:

    session_id = session.get(
        "session_id",
        "",
    )

    turns = len(
        session.get(
            "turn_records",
            [],
        )
    )

    return f"""
    <div class="session-bar">

        <div>
            <span class="status-dot"></span>
            Running
        </div>

        <div>
            Session
            <b>{esc(session_id)}</b>
        </div>

        <div>
            Turns
            <b>{turns}</b>
        </div>

    </div>
    """
