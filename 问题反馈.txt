# ============================================================
# Session
# ============================================================

def new_session():
    return {
        "session_id": "web_" + uuid.uuid4().hex[:10],

        "agent": MemoryAgent(
            mode="single",
            show_state=False,
        ),

        "history": [],
        "trace": [],
        "turns": 0,
        "mode": "user",
    }


# ============================================================
# Agent Snapshot
# ============================================================

def get_last_run(agent):
    """
    MemoryAgent.reply() 当前实际暴露的是：

    agent.last_turn_state = {
        "query": ...,
        "trigger": ...,
        "evidence": ...,
        "policy": ...,
        "response": ...,
        "latency": ...
    }

    UI 内部统一转换成：
        query_state
        trigger_state
        evidence_state
        policy_state
        response
        latency
    """

    turn_state = getattr(
        agent,
        "last_turn_state",
        None,
    )

    if turn_state:
        return {
            "query_state": turn_state.get(
                "query",
                {},
            ) or {},

            "trigger_state": turn_state.get(
                "trigger",
                {},
            ) or {},

            "evidence_state": turn_state.get(
                "evidence",
                {},
            ) or {},

            "policy_state": turn_state.get(
                "policy",
                {},
            ) or {},

            "response": turn_state.get(
                "response",
                "",
            ),

            "latency": turn_state.get(
                "latency",
                {},
            ) or {},
        }

    # fallback
    memory = getattr(
        agent,
        "memory",
        {},
    ) or {}

    return {
        "query_state": {},
        "trigger_state": {},

        "evidence_state": memory.get(
            "evidence_state",
            {},
        ) or {},

        "policy_state": (
            memory
            .get("interaction_state", {})
            .get("last_policy", {})
            or {}
        ),

        "response": "",
        "latency": {},
    }


# ============================================================
# History Parser
# ============================================================

def parse_prefix(text):
    history = []

    if not text:
        return history

    for raw_line in text.splitlines():
        line = raw_line.strip()

        if not line:
            continue

        if line.startswith("用户:"):
            content = line.split(
                "用户:",
                1,
            )[1].strip()

            history.append({
                "role": "user",
                "content": content,
            })

        elif line.startswith("用户："):
            content = line.split(
                "用户：",
                1,
            )[1].strip()

            history.append({
                "role": "user",
                "content": content,
            })

        elif line.startswith("客服:"):
            content = line.split(
                "客服:",
                1,
            )[1].strip()

            history.append({
                "role": "assistant",
                "content": content,
            })

        elif line.startswith("客服："):
            content = line.split(
                "客服：",
                1,
            )[1].strip()

            history.append({
                "role": "assistant",
                "content": content,
            })

    return history


# ============================================================
# Render Helpers
# ============================================================

def session_html(state):
    return render_session(
        state["session_id"],
        state["turns"],
    )


def developer_chat_html(state):
    return render_chat(
        state["history"],
        user_mode=False,
    )


def user_chat_html(state):
    return render_chat(
        state["history"],
        user_mode=True,
    )


def current_decision_html(state):
    if not state["trace"]:
        return render_decision(None)

    return render_decision(
        state["trace"][-1]
    )


def trace_html(state):
    return render_trace(
        state["trace"]
    )


# ============================================================
# Chat
# ============================================================

def chat(user_input, state):

    if state is None:
        state = new_session()

    user_input = str(
        user_input or ""
    ).strip()

    if not user_input:
        return (
            "",
            user_chat_html(state),
            developer_chat_html(state),
            current_decision_html(state),
            trace_html(state),
            session_html(state),
            state,
        )

    history = list(
        state["history"]
    )

    history.append({
        "role": "user",
        "content": user_input,
    })

    try:
        response = state["agent"].reply(
            history
        )

    except Exception as exc:
        response = (
            "Agent 执行失败："
            f"{type(exc).__name__}: {exc}"
        )

    response = str(
        response or ""
    ).strip()

    history.append({
        "role": "assistant",
        "content": response,
    })

    state["history"] = history
    state["turns"] += 1

    snapshot = get_last_run(
        state["agent"]
    )

    snapshot = dict(snapshot)

    snapshot["turn"] = state["turns"]
    snapshot["user"] = user_input
    snapshot["response"] = response

    state["trace"].append(
        snapshot
    )

    return (
        "",
        user_chat_html(state),
        developer_chat_html(state),
        current_decision_html(state),
        trace_html(state),
        session_html(state),
        state,
    )


# ============================================================
# Reset
# ============================================================

def reset_chat():

    state = new_session()

    return (
        "",
        user_chat_html(state),
        developer_chat_html(state),
        current_decision_html(state),
        trace_html(state),
        session_html(state),
        state,
        "",
    )


# ============================================================
# Load Prefix
# ============================================================

def load_prefix(prefix_text, state):

    if state is None:
        state = new_session()

    history = parse_prefix(
        prefix_text
    )

    if not history:
        return (
            developer_chat_html(state),
            current_decision_html(state),
            trace_html(state),
            session_html(state),
            state,
            render_prefix_result(
                False,
                "没有识别到有效对话。",
            ),
        )

    state["agent"] = MemoryAgent(
        mode="single",
        show_state=False,
    )

    state["history"] = history
    state["trace"] = []

    state["turns"] = sum(
        1
        for item in history
        if item.get("role") == "user"
    )

    return (
        developer_chat_html(state),
        current_decision_html(state),
        trace_html(state),
        session_html(state),
        state,
        render_prefix_result(
            True,
            f"已加载 {len(history)} 条历史消息。",
        ),
    )
