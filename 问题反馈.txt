import html
import json


# ============================================================
# Utils
# ============================================================

def escape(value):
    if value is None:
        return ""
    return html.escape(str(value))


def json_text(value):
    try:
        return json.dumps(
            value,
            ensure_ascii=False,
            indent=2,
        )
    except Exception:
        return str(value)


def badge(text, kind="gray"):
    return (
        f'<span class="badge badge-{kind}">'
        f'{escape(text)}'
        f'</span>'
    )

def render_empty_chat(user_mode=False):
    if user_mode:
        title = "您好，有什么可以帮您？"
        desc = "请描述您遇到的 IT 问题，我会尽力协助您解决。"
    else:
        title = "开始新的 IT 支持会话"
        desc = (
            "输入用户问题后，Agent 将执行 Query、Retrieval、"
            "Topic Analysis、Policy 和 Response。"
        )

    return f"""
    <div class="chat-surface">
        <div class="empty-chat">
            <div class="empty-symbol">✦</div>
            <div class="empty-title">{title}</div>
            <div class="empty-desc">{desc}</div>
        </div>
    </div>
    """


def render_chat(history, user_mode=False):
    if not history:
        return render_empty_chat(user_mode=user_mode)

    rows = []

    for message in history:
        role = message.get("role")
        content = escape(message.get("content", ""))

        if role == "user":
            rows.append(
                f"""
                <div class="message-row user-row">
                    <div class="message-wrap user-wrap">
                        <div class="message-role">用户</div>
                        <div class="message-bubble user-message">
                            {content}
                        </div>
                    </div>

                    <div class="avatar user-avatar">
                        U
                    </div>
                </div>
                """
            )

        elif role == "assistant":
            rows.append(
                f"""
                <div class="message-row agent-row">
                    <div class="avatar agent-avatar">
                        AI
                    </div>

                    <div class="message-wrap">
                        <div class="message-role">智能客服</div>
                        <div class="message-bubble agent-message">
                            {content}
                        </div>
                    </div>
                </div>
                """
            )

    return f"""
    <div class="chat-surface">
        <div class="chat-scroll">
            {''.join(rows)}
        </div>
    </div>
    """


# ============================================================
# Topic Tree
# ============================================================

def render_topic_tree(tree):
    if not tree:
        return """
        <div class="empty-small">
            暂无 Topic Structure
        </div>
        """

    def walk(node, depth=0):
        topic = escape(node.get("topic", ""))
        dimension = escape(node.get("dimension", ""))

        case_ids = node.get("case_ids") or []

        if isinstance(case_ids, str):
            case_ids = [case_ids]

        case_text = ", ".join(
            escape(x) for x in case_ids
        )

        leaf = bool(node.get("leaf"))

        node_class = (
            "tree-node tree-leaf"
            if leaf
            else "tree-node"
        )

        margin = depth * 18

        dimension_html = ""

        if dimension:
            dimension_html = (
                f'<span class="tree-dimension">'
                f'{dimension}'
                f'</span>'
            )

        case_html = ""

        if case_text:
            case_html = (
                f'<span class="tree-case">'
                f'{case_text}'
                f'</span>'
            )

        current = f"""
        <div
            class="{node_class}"
            style="margin-left:{margin}px"
        >
            <span class="tree-dot"></span>
            <span class="tree-topic">{topic}</span>
            {dimension_html}
            {case_html}
        </div>
        """

        children = node.get("children") or []

        return current + "".join(
            walk(child, depth + 1)
            for child in children
        )

    return f"""
    <div class="topic-tree">
        {walk(tree)}
    </div>
    """


# ============================================================
# Evidence
# ============================================================

def render_cases(cases):
    if not cases:
        return """
        <div class="empty-small">
            当前没有 Active Cases
        </div>
        """

    items = []

    for case in cases:
        if not isinstance(case, dict):
            continue

        case_id = (
            case.get("case_id")
            or case.get("id")
            or case.get("caseId")
            or ""
        )

        title = (
            case.get("title")
            or case.get("name")
            or case.get("question")
            or case.get("text")
            or ""
        )

        title = str(title)

        if len(title) > 110:
            title = title[:110] + "..."

        items.append(
            f"""
            <div class="case-item">
                <div class="case-id">
                    {escape(case_id)}
                </div>

                <div class="case-title">
                    {escape(title)}
                </div>
            </div>
            """
        )

    return "".join(items)


# ============================================================
# Latency
# ============================================================

def render_latency(latency):
    latency = latency or {}

    total = latency.get("total", 0)

    def get_time(*names):
        for name in names:
            if name in latency:
                return latency[name]
        return 0

    query = get_time("query")
    trigger = get_time("trigger")
    rag = get_time("rag", "retrieval")
    topic = get_time("topic", "topic_structure")
    policy = get_time("policy")
    response = get_time("response")

    return f"""
    <div class="latency-total">
        {float(total):.2f}
        <span>s</span>
    </div>

    <div class="latency-grid">
        <div>
            <span>Query</span>
            <b>{float(query):.2f}s</b>
        </div>

        <div>
            <span>Trigger</span>
            <b>{float(trigger):.2f}s</b>
        </div>

        <div>
            <span>RAG</span>
            <b>{float(rag):.2f}s</b>
        </div>

        <div>
            <span>Topic</span>
            <b>{float(topic):.2f}s</b>
        </div>

        <div>
            <span>Policy</span>
            <b>{float(policy):.2f}s</b>
        </div>

        <div>
            <span>Response</span>
            <b>{float(response):.2f}s</b>
        </div>
    </div>
    """

def render_decision(snapshot):
    if not snapshot:
        return """
        <div class="decision-surface">
            <div class="decision-empty">
                <div class="decision-empty-title">
                    Waiting for interaction
                </div>

                <div class="decision-empty-desc">
                    当前轮执行后，Query、Trigger、Evidence、
                    Topic、Policy 和 Latency 将显示在这里。
                </div>
            </div>
        </div>
        """

    query_state = snapshot.get("query_state", {})
    trigger_state = snapshot.get("trigger_state", {})
    policy_state = snapshot.get("policy_state", {})

    evidence_state = snapshot.get(
        "evidence_state",
        {},
    )

    active_cases = evidence_state.get(
        "active_cases",
        [],
    )

    topic_structure = evidence_state.get(
        "topic_structure",
        {},
    )

    latency = snapshot.get("latency", {})

    query_mode = query_state.get(
        "query_mode",
        "none",
    )

    query_text = query_state.get(
        "query_text",
        "",
    )

    trigger = trigger_state.get(
        "trigger",
        "",
    )

    evidence_mode = trigger_state.get(
        "evidence_mode",
        "",
    )

    policy = policy_state.get(
        "label",
        "",
    )

    information = policy_state.get(
        "information",
    )

    return f"""
    <div class="decision-surface">

        <div class="decision-section">
            <div class="section-heading">
                Query
            </div>

            <div class="decision-line">
                {badge(query_mode, "blue")}
            </div>

            <div class="decision-text">
                {escape(query_text) or "None"}
            </div>
        </div>


        <div class="decision-section">
            <div class="section-heading">
                Trigger
            </div>

            <div class="decision-row">
                <span>Trigger</span>
                {badge(trigger, "green")}
            </div>

            <div class="decision-row">
                <span>Evidence Mode</span>
                {badge(evidence_mode, "gray")}
            </div>
        </div>


        <div class="decision-section">
            <div class="section-heading">
                Evidence
            </div>

            {render_cases(active_cases)}
        </div>


        <div class="decision-section">
            <div class="section-heading">
                Topic Structure
            </div>

            {render_topic_tree(topic_structure)}
        </div>


        <div class="decision-section">
            <div class="section-heading">
                Policy
            </div>

            <div class="decision-line">
                {badge(policy, "blue")}
            </div>

            <div class="sub-label">
                Information
            </div>

            <pre class="json-view">{escape(json_text(information))}</pre>
        </div>


        <div class="decision-section latency-section">
            <div class="section-heading">
                Latency
            </div>

            {render_latency(latency)}
        </div>

    </div>
    """


# ============================================================
# Trace
# ============================================================

def render_trace(trace):
    if not trace:
        return """
        <div class="trace-surface">
            <div class="decision-empty">
                暂无运行轨迹
            </div>
        </div>
        """

    items = []

    for index, item in enumerate(
        reversed(trace),
        start=1,
    ):
        turn = item.get(
            "turn",
            len(trace) - index + 1,
        )

        user = escape(
            item.get("user", "")
        )

        query = item.get(
            "query_state",
            {},
        )

        trigger = item.get(
            "trigger_state",
            {},
        )

        policy = item.get(
            "policy_state",
            {},
        )

        latency = item.get(
            "latency",
            {},
        )

        total = latency.get(
            "total",
            0,
        )

        items.append(
            f"""
            <div class="trace-item">

                <div class="trace-head">
                    <b>Turn {turn}</b>
                    <span>{float(total):.2f}s</span>
                </div>

                <div class="trace-user">
                    {user}
                </div>

                <div class="trace-flow">
                    {badge(
                        query.get("query_mode", "none"),
                        "blue"
                    )}

                    <span class="flow-arrow">→</span>

                    {badge(
                        trigger.get(
                            "evidence_mode",
                            ""
                        ),
                        "gray"
                    )}

                    <span class="flow-arrow">→</span>

                    {badge(
                        policy.get("label", ""),
                        "green"
                    )}
                </div>

            </div>
            """
        )

    return f"""
    <div class="trace-surface">
        {''.join(items)}
    </div>
    """
