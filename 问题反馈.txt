def normalize_split(split_result, expected_case_ids):
    if not isinstance(split_result, dict):
        return None, "output is not a dict"

    # ========================================================
    # 1. Validate split
    # ========================================================

    split = split_result.get("split")

    if not isinstance(split, bool):
        return None, "missing or invalid split"

    # LLM 判断当前节点不应该继续划分
    if split is False:
        return {
            "split": False,
            "dimension": "",
            "children": []
        }, ""

    # ========================================================
    # 2. Validate dimension
    # ========================================================

    dimension = str(
        split_result.get("dimension", "")
    ).strip()

    if not dimension:
        return None, "missing dimension"

    # ========================================================
    # 3. Read assignments
    # ========================================================

    assignments = split_result.get(
        "assignments",
        []
    )

    if not isinstance(assignments, list):
        return None, "assignments is not a list"

    if not assignments:
        return None, "assignments is empty"

    expected_case_ids = [
        str(case_id).strip()
        for case_id in expected_case_ids
    ]

    expected_set = set(expected_case_ids)

    assignment_map = {}

    # ========================================================
    # 4. Validate every assignment
    # ========================================================

    for item in assignments:

        if not isinstance(item, dict):
            return None, "assignment is not a dict"

        case_id = str(
            item.get("case_id", "")
        ).strip()

        topic = str(
            item.get("topic", "")
        ).strip()

        if not case_id:
            return None, "assignment case_id is empty"

        if not topic:
            return None, f"topic of case '{case_id}' is empty"

        # 不允许生成输入之外的 case
        if case_id not in expected_set:
            return None, f"unexpected case_id: {case_id}"

        # 不允许同一个 case 出现两次
        if case_id in assignment_map:
            return None, f"duplicated case_id: {case_id}"

        assignment_map[case_id] = topic

    # ========================================================
    # 5. Check missing cases
    # ========================================================

    missing = [
        case_id
        for case_id in expected_case_ids
        if case_id not in assignment_map
    ]

    if missing:
        return None, f"missing case_ids: {missing}"

    # ========================================================
    # 6. Group cases by topic
    # ========================================================

    groups = {}

    # 按原始 case 顺序构建，避免 LLM 输出顺序影响结构
    for case_id in expected_case_ids:
        topic = assignment_map[case_id]

        if topic not in groups:
            groups[topic] = []

        groups[topic].append(case_id)

    # 至少需要两个 sibling topics
    if len(groups) < 2:
        return None, (
            "fewer than 2 valid topic groups: "
            f"{list(groups.keys())}"
        )

    # ========================================================
    # 7. Convert assignments -> children
    # ========================================================

    normalized_children = []

    for topic, case_ids in groups.items():
        normalized_children.append({
            "topic": topic,
            "case_ids": case_ids
        })

    return {
        "split": True,
        "dimension": dimension,
        "children": normalized_children
    }, ""
